\begin{frame}{Tahapan dan Peran}
\small
\begin{enumerate}
    \item \textbf{Penyuntingan.} Program ditulis sebagai berkas
          \file{.ASM} dengan editor teks.

    \vspace{0.15em}
    \item \textbf{Perakitan.} TASM menerjemahkan instruksi menjadi
          \file{.OBJ}; simbol dan label dicatat, alamat akhir belum
          ditetapkan.

    \vspace{0.15em}
    \item \textbf{Pengaitan.} TLINK menggabungkan berkas objek,
          menyelesaikan rujukan simbol, dan menghasilkan
          \file{.EXE} atau \file{.COM}.

    \vspace{0.15em}
    \item \textbf{Pemuatan.} DOS memuat program, menetapkan register
          segmen, lalu menyerahkan kendali ke instruksi pertama.
\end{enumerate}

\begin{frame}[fragile]{Kerangka Program EXE}
\begin{kodeasm}
.MODEL SMALL
.STACK 100h
.DATA
    ; deklarasi variabel dan konstanta
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    ; instruksi program
    MOV AH, 4Ch
    MOV AL, 00h
    INT 21h
END START
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Kerangka ini memisahkan kode, data, dan

% ---------------------------------------------------------
\begin{frame}[fragile]{Label dan Komentar}
\small
\begin{itemize}
    \item Label boleh dimulai huruf, \code{@}, \code{\_}, atau
          \code{?}; tidak boleh berspasi atau diawali angka.
    \item Panjang label maksimal 31 karakter; label instruksi
          diakhiri titik dua.
    \item Komentar ditulis setelah \code{;} dan diabaikan assembler.
\end{itemize}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV AH, 09h     ; siapkan fungsi cetak string
LEA DX, pesan   ; DX = alamat string
INT 21h         ; panggil layanan DOS
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Komentar menjelaskan maksud, bukan
mengulang bunyi instruksi.\par}
\Sumber{Bab VII, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Program COM dan ORG 100h}
\begin{columns}[t]
\begin{column}{0.50\textwidth}
\begin{kodeasm}
.MODEL TINY
.CODE
ORG 100h
START:
    MOV AH, 09h
    MOV DX, OFFSET pesan
    INT 21h
    INT 20h
pesan DB 'Halo, COM!', 0Dh, 0Ah, '$'
END START
\end{kodeasm}
\end{column}
\begin{column}{0.46\textwidth}
\footnotesize
\dir{ORG 100h} menetapkan program dimuat mulai offset
\hex{100}.

\vspace{0.3em}
Ruang 256 byte sebelum program dipakai DOS untuk
\eng{Program Segment Prefix} (PSP).

\vspace{0.3em}
Pada \file{.COM}, DOS sudah menetapkan register segmen; inisialisasi
DS tidak diperlukan.
\end{column}
\end{columns}
\Sumber{Bab VII, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Inisialisasi Register DS}
\small
Program \file{.EXE} tidak otomatis memiliki DS yang menunjuk segmen
data. Dua instruksi berikut perlu ditulis di awal program.

\vspace{0.3em}
\begin{kodeasm}[numbers=none]
MOV AX, @DATA       ; alamat segmen data
MOV DS, AX          ; pasang ke DS
\end{kodeasm}

\vspace{0.2em}
\code{@DATA} dihasilkan oleh \dir{.MODEL} dan berisi nomor segmen
tempat \dir{.DATA} berada.

\begin{alertblock}{Bila dihilangkan}
\small Instruksi yang mengakses variabel akan membaca alamat yang
salah.
\end{alertblock}
\Sumber{Bab VII, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item Kerangka program memuat \dir{.MODEL}, \dir{.STACK},
          \dir{.DATA}, \dir{.CODE}, label titik masuk, dan
          \dir{END}.

    \vspace{0.15em}
    \item \dir{.MODEL} menentukan susunan segmen; \code{SMALL} untuk
          \file{.EXE} sederhana dan \code{TINY} untuk \file{.COM}.

    \vspace{0.15em}
    \item Label menandai lokasi instruksi atau data; komentar
          ditulis setelah \code{;}.

    \vspace{0.15em}
    \item \dir{ORG 100h} memberi ruang PSP 256 byte pada program
          \file{.COM}.

    \vspace{0.15em}
    \item Program \file{.EXE} perlu menginisialisasi DS melalui
          \code{MOV AX, @DATA} dan \code{MOV DS, AX}.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Direktif Bukan Instruksi}
\small
\begin{itemize}
    \item \term{Instruksi} dijalankan prosesor, misalnya
          \code{MOV} dan \code{ADD}.

    \vspace{0.18em}
    \item \term{Direktif} adalah perintah kepada assembler,
          misalnya \dir{.MODEL}, \dir{DB}, dan \dir{EQU}.

    \vspace{0.18em}
    \item Direktif tidak diterjemahkan menjadi instruksi mesin.

    \vspace{0.18em}
    \item \dir{.MODEL} menggantikan penulisan \dir{SEGMENT} dan
          \dir{ASSUME} pada program sederhana.

    \vspace{0.18em}
    \item \dir{ORG 100h} hanya untuk \file{.COM}; \file{.EXE}
          memakai alamat dari linker.
\end{itemize}

\vspace{0.25em}
\begin{block}{Inti}
\small Direktif mengatur bagaimana assembler menyusun program dan
data.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Alur Translasi dan Berkas Hasil}
\small
\begin{enumerate}
    \item \file{program.asm} ditulis di editor.

    \vspace{0.15em}
    \item TASM menghasilkan \file{program.obj}.

    \vspace{0.15em}
    \item TLINK menghasilkan \file{program.exe} atau, dengan
          \code{/t}, \file{program.com}.

    \vspace{0.15em}
    \item DOS memuat berkas hasil dan menjalankannya.
\end{enumerate}

\vspace{0.25em}
\begin{block}{Mengapa \file{.OBJ} belum dapat dijalankan?}
\small \file{.OBJ} masih memuat rujukan simbol dan alamat yang belum
final; TLINK menyelesaikannya.
\end{block}
\Sumber{Bab VII, Bagian 3}
\end{frame}

\begin{frame}{Kesalahan Logika}
\small
\begin{itemize}
    \item TASM melaporkan kesalahan penulisan beserta nomor baris.

    \vspace{0.18em}
    \item Kesalahan logika tidak selalu dilaporkan: salah memilih
          register, lupa inisialisasi DS, atau salah menghitung
          offset.

    \vspace{0.18em}
    \item Kesalahan seperti itu harus dicari dengan
          \eng{debugger}.
\end{itemize}

\vspace{0.3em}
\begin{alertblock}{Pesan assembler bukan jaminan}
\small Program yang berhasil dirakit belum tentu berjalan benar.
\end{alertblock}
\Sumber{Bab VII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item \dir{DB}, \dir{DW}, dan \dir{DD} menentukan ukuran data;
          \dir{EQU} memberi nama konstanta; \dir{DUP} mengulang
          pola.

    \vspace{0.15em}
    \item Direktif adalah perintah kepada assembler, bukan
          instruksi yang dijalankan prosesor.

    \vspace{0.15em}
    \item Alur translasi: \file{.ASM} $\rightarrow$ \file{.OBJ}
          $\rightarrow$ \file{.EXE}/\file{.COM} $\rightarrow$
          memori.

    \vspace{0.15em}
    \item \file{.OBJ} belum dapat dijalankan karena alamat dan
          rujukan simbol belum final.

    \vspace{0.15em}
    \item TASM hanya melaporkan kesalahan penulisan; kesalahan
          logika dicari dengan \eng{debugger}.
\end{itemize}
\end{frame}

    \item Berapa byte yang disediakan \code{nilai DB 25},

    \item Sebutkan penyebab dan perbaikan: \file{Undefined symbol},

          \file{.EXE}, dan berkas tanpa \dir{END}.
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item Rantai proses: \file{.ASM} $\rightarrow$ TASM
          $\rightarrow$ \file{.OBJ} $\rightarrow$ TLINK
          $\rightarrow$ \file{.EXE}/\file{.COM} $\rightarrow$
          memori.

    \vspace{0.15em}
    \item TASM menerjemahkan instruksi; TLINK menyelesaikan rujukan
          dan menetapkan alamat segmen.

    \vspace{0.15em}
    \item \file{.COM}: satu segmen, maksimum sekitar 64 KB,
          \dir{ORG 100h}, tanpa inisialisasi DS.

    \vspace{0.15em}
    \item \file{.EXE}: beberapa segmen, titik masuk dari \dir{END},
          perlu inisialisasi DS.

    \vspace{0.15em}
    \item Kerangka program: \dir{.MODEL}, \dir{.STACK}, \dir{.DATA},
          \dir{.CODE}, label, dan \dir{END}.

    \vspace{0.15em}
    \item \dir{DB}/\dir{DW}/\dir{DD} menentukan ukuran;
          \dir{EQU} konstanta; \dir{DUP} mengulang pola.
\end{itemize}
\end{frame}
