\begin{frame}{Capaian dan Tujuan Pembelajaran}
\begin{block}{Sub-CPMK 1.3}
\small Mengidentifikasi register umum (AX--DX), register segmen
(CS, DS, SS, ES), register \eng{pointer}/\eng{index} (SP, BP, SI,
DI), IP, dan \eng{flag} beserta pengaruh \eng{flag} terhadap
keputusan program.
\end{block}

\vspace{0.1em}
{\small\textbf{CPMK yang didukung:} \CPMK{01}\quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{02}\par}

\vspace{0.15em}
\begin{exampleblock}{Setelah pertemuan ini, mahasiswa mampu:}
\footnotesize
\begin{enumerate}
    \item mengidentifikasi keempat belas register 8086 dan
          kelompoknya;
    \item menjelaskan register umum AX--DX serta akses byte tinggi
          dan byte rendah;
    \item menjelaskan register segmen dan pasangan alamat yang
          dibentuknya;
    \item menjelaskan SP, BP, SI, DI, dan IP;
    \item menjelaskan sembilan \eng{flag} dan pengaruhnya terhadap
          keputusan program.
\end{enumerate}
\end{exampleblock}
\end{frame}

% 3. PETA MATERI

\begin{frame}{Peta Materi}

    \vspace{0.25em}
    \item \textbf{Register Pointer, Index, dan Instruction Pointer}\\

% ---------------------------------------------------------
\begin{frame}{Peran Khas Register Umum}
\small
\begin{itemize}
    \item \textbf{AX sebagai akumulator.} Banyak operasi aritmatika
          dan layanan DOS memakai AX atau AL sebagai operand utama.

    \vspace{0.18em}
    \item \textbf{BX sebagai penunjuk alamat.} BX sering menyimpan
          alamat \eng{offset} dari suatu segmen.

    \vspace{0.18em}
    \item \textbf{CX sebagai pencacah.} CX menentukan berapa kali
          pengulangan dijalankan; \code{LOOP} memakainya.

    \vspace{0.18em}
    \item \textbf{DX sebagai data pendukung.} DX menampung sisa
          pembagian 16 bit dan data pendukung operasi tertentu.
\end{itemize}

\vspace{0.25em}
\begin{block}{Bukan larangan mutlak}
\small Peran khas menunjukkan pemakaian yang paling sering dan paling
aman; register umum tetap dapat menampung nilai lain.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Pasangan Alamat Segmen}
\small
\begin{itemize}
    \item \reg{CS}:\reg{IP} --- instruksi berikutnya.

    \vspace{0.18em}
    \item \reg{DS}:\reg{SI}, \reg{DS}:\reg{BX}, atau \reg{DS} dengan
          \eng{offset} --- data umum.

    \vspace{0.18em}
    \item \reg{SS}:\reg{SP} atau \reg{SS}:\reg{BP} ---
          \eng{stack}.

    \vspace{0.18em}
    \item \reg{ES}:\reg{DI} --- tujuan operasi string.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{Hati-hati}
\small Mengubah CS, IP, SS, atau SP tanpa alasan yang benar dapat
membuat prosesor mengambil instruksi dari alamat salah atau membuat
\code{RET} melompat ke alamat acak.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item 8086 memiliki empat belas register 16 bit yang
          dikelompokkan menjadi umum, segmen, pointer/index, dan
          \eng{flag}.

    \vspace{0.15em}
    \item AX, BX, CX, dan DX dapat diakses sebagai dua byte 8 bit:
          AH/AL, BH/BL, CH/CL, dan DH/DL.

    \vspace{0.15em}
    \item Peran khas: AX akumulator, BX penunjuk offset, CX
          pencacah, DX data pendukung.

    \vspace{0.15em}
    \item CS, DS, SS, dan ES menyimpan nomor segmen; pasangan
          alamatnya adalah CS:IP, DS:SI/BX, SS:SP/BP, dan ES:DI.

    \vspace{0.15em}
    \item Register segmen tidak menyimpan hasil perhitungan,
          melainkan letak segmen.
\end{itemize}
\end{frame}

\PemisahBagian{Register Pointer, Index, dan Instruction Pointer}

% ---------------------------------------------------------
\begin{frame}{Stack Pointer dan Base Pointer}
\small
\begin{itemize}
    \item \eng{Stack} bekerja secara \eng{last in, first out}.
          \code{PUSH} menaruh data, \code{POP} mengambilnya.

    \vspace{0.18em}
    \item \reg{SP} menunjuk puncak \eng{stack} aktif; alamatnya
          ditulis \reg{SS}:\reg{SP}. Nilainya berubah saat
          \code{PUSH} dan \code{POP}.

    \vspace{0.18em}
    \item \reg{BP} menjadi titik acuan di dalam \eng{stack};
          alamatnya \reg{SS}:\reg{BP}. BP tidak terpengaruh
          perubahan SP.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{Akibat perubahan SP}
\small Mengubah SP secara sembarangan berarti memindahkan seluruh
tumpukan; \code{RET} dapat melompat ke alamat yang salah.
\end{alertblock}
\Sumber{Bab V, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Source Index dan Destination Index}
\small
\begin{itemize}
    \item \reg{SI} (\eng{Source Index}) menunjuk alamat sumber pada
          operasi string; pasangannya \reg{DS}:\reg{SI}.

    \vspace{0.18em}
    \item \reg{DI} (\eng{Destination Index}) menunjuk alamat tujuan;
          pasangannya \reg{ES}:\reg{DI}.

    \vspace{0.18em}
    \item Keduanya dipakai untuk mengakses deretan data seperti
          string atau \eng{array}.

    \vspace{0.18em}
    \item Pada operasi string, SI dan DI dapat berubah otomatis
          mengikuti arah pemrosesan.
\end{itemize}

\vspace{0.25em}
\begin{block}{Sebelum operasi string}
\small Isi SI dan DI dengan alamat yang benar lebih dahulu.
\end{block}
\Sumber{Bab V, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Instruction Pointer}
\small
\begin{itemize}
    \item \reg{IP} menyimpan \eng{offset} instruksi berikutnya yang
          akan diambil dan dieksekusi.

    \vspace{0.18em}
    \item Pasangan \reg{CS}:\reg{IP} menentukan alamat instruksi
          tersebut.

    \vspace{0.18em}
    \item IP bertambah sendiri setiap kali instruksi diambil.

    \vspace{0.18em}
    \item Programmer hampir tidak pernah menulis IP secara langsung;
          perubahannya terjadi melalui \code{JMP}, \code{CALL},
          \code{RET}, dan \code{INT}.
\end{itemize}

\vspace{0.25em}
\begin{block}{Hubungan dengan BIU}
\small BIU memakai \reg{CS}:\reg{IP} untuk mengambil instruksi
berikutnya.
\end{block}
\Sumber{Bab V, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Mengapa 7FFF + 0001 Menghasilkan OF = 1}
\small
\begin{itemize}
    \item \hex{7FFF} = $+32767$ dan \hex{0001} = $+1$.

    \vspace{0.18em}
    \item Hasil 16 bit adalah \hex{8000}, yang sebagai bilangan
          bertanda berarti $-32768$.

    \vspace{0.18em}
    \item Dua bilangan positif menghasilkan bilangan negatif,
          sehingga OF = 1.

    \vspace{0.18em}
    \item CF = 0 karena tidak ada simpanan keluar dari 16 bit untuk
          operasi tak bertanda.

    \vspace{0.18em}
    \item SF = 1 karena bit tanda hasil \hex{8000} menyala.
\end{itemize}

\vspace{0.2em}
\begin{block}{Inti}
\small CF dan OF menjawab pertanyaan yang berbeda; keduanya tidak
boleh dipertukarkan.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{CF vs OF}
\small
\begin{columns}[t]
\begin{column}{0.47\textwidth}
    {\color{obeTeal}\bfseries CF --- tak bertanda}\\[0.25em]
    \begin{itemize}
        \item Menandai simpanan keluar dari bit tertinggi.
        \item Dipakai dengan \code{JC}, \code{JB}, \code{JNC},
              \code{JAE}.
        \item Contoh: \hex{FFFF} + \hex{0001}.
    \end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
    {\color{obeMaroon}\bfseries OF --- bertanda}\\[0.25em]
    \begin{itemize}
        \item Menandai hasil di luar jangkauan bertanda.
        \item Dipakai dengan \code{JO}, \code{JNO}.
        \item Contoh: \hex{7FFF} + \hex{0001}.
    \end{itemize}
\end{column}
\end{columns}

\vspace{0.5em}
\begin{alertblock}{Kesalahan umum}
\small Memakai \code{JC} untuk memeriksa hasil bilangan bertanda
dapat memberi kesimpulan yang salah.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item FLAGS memiliki sembilan penanda: CF, PF, AF, ZF, SF, OF
          sebagai status serta TF, IF, DF sebagai kendali.

    \vspace{0.15em}
    \item CF menandai simpanan keluar untuk operasi tak bertanda;
          OF menandai \eng{overflow} untuk operasi bertanda.

    \vspace{0.15em}
    \item \hex{7FFF} + \hex{0001} menghasilkan \hex{8000} dengan
          OF = 1 dan SF = 1, tetapi CF = 0.

    \vspace{0.15em}
    \item \hex{8000} + \hex{8000} menghasilkan \hex{0000} dengan
          CF = 1, OF = 1, dan ZF = 1.

    \vspace{0.15em}
    \item \eng{Flag} dibaca oleh lompatan bersyarat untuk memilih
          jalur program.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Contoh 1: Memprediksi Flag}
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\begin{kodeasm}
.MODEL SMALL
.DATA
    hasil1 DW 0
    hasil2 DW 0
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AX, 7FFFh
    MOV BX, 0001h
    ADD AX, BX
    MOV hasil1, AX
    MOV AX, 8000h
    MOV BX, 8000h
    ADD AX, BX
\end{kodeasm}
\end{column}
\begin{column}{0.50\textwidth}
\footnotesize
{\color{obeTeal}\bfseries Kasus 1: \hex{7FFF} + \hex{0001}}\\[0.15em]
Hasil \hex{8000}; CF = 0, OF = 1, SF = 1, ZF = 0.

\vspace{0.35em}
{\color{obeMaroon}\bfseries Kasus 2: \hex{8000} + \hex{8000}}\\[0.15em]
Hasil \hex{0000}; CF = 1, OF = 1, SF = 0, ZF = 1.

\vspace{0.35em}
\begin{block}{Perhatikan}
\small \code{MOV hasil1, AX} tidak mengubah \eng{flag}; keadaan
hasil penjumlahan masih dapat diamati.
\end{block}

\vspace{0.2em}
{\color{obeGray}Prediksikan sebelum menjalankan \eng{single step}.
\par}
\end{column}
\end{columns}
\Sumber{Bab V, Contoh 1}
\end{frame}

\begin{frame}[fragile]{Contoh 2: Memilih Pesan Berdasarkan Flag}
\begin{columns}[t]
\begin{column}{0.56\textwidth}
\begin{kodeasm}
.MODEL SMALL
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AL, 0F0h
    MOV BL, 10h
    ADD AL, BL
    JNC tanpa_carry
    LEA DX, pesan_carry
    MOV AH, 09h
    INT 21h
tanpa_carry:
    JNZ bukan_nol
...
\end{kodeasm}
\end{column}
\begin{column}{0.40\textwidth}
\footnotesize
\code{ADD AL, BL} menghasilkan AL = \hex{00}, CF = 1, ZF = 1,
SF = 0.

\vspace{0.25em}
Karena CF = 1 dan ZF = 1, \code{JNC} dan \code{JNZ} tidak diambil;
pesan carry lalu pesan hasil nol dicetak.

\vspace{0.25em}
{\color{obeTeal}\bfseries Keluaran}\\[0.15em]
\begin{keluaran}
Terjadi carry, CF = 1.
Hasil nol, ZF = 1.
\end{keluaran}
\end{column}
\end{columns}

% =========================================================
% LATIHAN
% =========================================================
\begin{frame}{Latihan}
\small
\begin{enumerate}
    \item Sebutkan keempat belas register 8086 dan kelompokkan
          menjadi umum, segmen, pointer/index, kendali, dan status.

    \vspace{0.15em}
    \item AX semula berisi \hex{1234}. Tentukan AH dan AL. Setelah
          \code{MOV AL, 20h}, tentukan AX dan jelaskan mengapa AH
          tidak berubah.

    \vspace{0.15em}
    \item Jelaskan peran CS, DS, SS, dan ES beserta pasangan alamat
          yang dibentuk masing-masing.

    \vspace{0.15em}
    \item Jelaskan arti CF, PF, AF, ZF, SF, OF, TF, IF, dan DF.
          Kelompokkan menjadi \eng{flag} status dan kendali.

    \vspace{0.15em}
    \item Untuk \hex{7FFF} + \hex{0001} dan \hex{8000} + \hex{8000},
          tentukan hasil 16 bit serta CF, OF, SF, dan ZF.
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item Empat belas register 8086 berukuran 16 bit dan terbagi
          menjadi umum, segmen, pointer/index, serta \eng{flag}.

    \vspace{0.15em}
    \item AX--DX dapat diakses per byte; peran khasnya adalah
          akumulator, offset, pencacah, dan data pendukung.

    \vspace{0.15em}
    \item Register segmen membentuk pasangan CS:IP, DS:SI/BX,
          SS:SP/BP, dan ES:DI.

    \vspace{0.15em}
    \item SP menunjuk puncak \eng{stack}, BP menjadi acuan, SI/DI
          menunjuk sumber/tujuan, dan IP menunjuk instruksi
          berikutnya.

    \vspace{0.15em}
    \item Sembilan \eng{flag}: CF, PF, AF, ZF, SF, OF, TF, IF, dan
          DF; CF untuk tak bertanda dan OF untuk bertanda.

    \vspace{0.15em}
    \item \eng{Flag} dibaca oleh lompatan bersyarat untuk memilih
          jalur program.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Setelah \code{MOV AL, 20h} pada AX yang semula
          \hex{1234}, isi AX menjadi \dots
          \begin{enumerate}
              \item \hex{1220} \quad \item \hex{2034} \quad
              \item \hex{0020} \quad \item \hex{2020}
          \end{enumerate}

    \vspace{0.2em}
    \item Register yang berpasangan dengan CS untuk menentukan
          alamat instruksi berikutnya adalah \dots
          \begin{enumerate}
              \item SP \quad \item IP \quad
              \item DS \quad \item DI
          \end{enumerate}

    \vspace{0.2em}
    \item \code{ADD AL, BL} dengan AL = \hex{7F} dan BL = \hex{01}
          menghasilkan \dots
          \begin{enumerate}
              \item CF = 1 dan OF = 0
              \item CF = 0 dan OF = 1
              \item CF = 1 dan OF = 1
              \item CF = 0 dan OF = 0
          \end{enumerate}
\end{enumerate}
\end{frame}
