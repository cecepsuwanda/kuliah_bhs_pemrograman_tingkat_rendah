\begin{frame}{Mengapa Interupsi Diperlukan}
\small
\begin{itemize}
    \item Prosesor berjalan jauh lebih cepat daripada perangkat di
          sekitarnya. Bila program terus-menerus memeriksa status
          keyboard, sebagian besar waktunya terbuang untuk pengulangan
          yang hampir selalu sia-sia --- cara ini disebut
          \eng{polling}.

    \vspace{0.25em}
    \item \term{Interupsi} adalah mekanisme yang memaksa prosesor
          menghentikan sementara alur program, memindahkan kendali ke
          rutin khusus yang disebut \term{penangan interupsi}
          (\eng{interrupt handler}), lalu melanjutkan kembali program
          semula tepat pada instruksi berikutnya.

    \vspace{0.25em}
    \item Interupsi juga menjadi sarana baku untuk meminta layanan
          sistem: program cukup memanggil \code{INT 10h} tanpa menulis
          sendiri rutin pengendali video.
\end{itemize}

\vspace{0.3em}
\begin{exampleblock}{Kotak hitam yang dibuka}
\small Sejak Bab III, \code{INT 21h} dipakai sebagai kotak hitam.
Pertanyaannya kini bukan lagi ``bagaimana memakai'' layanan,
melainkan ``apa yang sesungguhnya terjadi'' ketika \asm{INT}
dijalankan.
\end{exampleblock}
\Sumber{Bab XIV, Mengapa Interupsi Diperlukan}
\end{frame}

\begin{frame}[fragile]{Mekanisme \asm{INT} dan \asm{IRET}}
\begin{columns}[t]
\begin{column}{0.56\textwidth}
{\footnotesize\asm{INT} \code{n} bekerja seperti pemanggilan prosedur
jarak jauh, dengan tambahan penyimpanan \eng{flag}:}
\begin{enumerate}\footnotesize
    \item \reg{FLAGS} disimpan ke stack;
    \item \reg{CS} disimpan ke stack;
    \item \reg{IP} --- sudah menunjuk instruksi sesudah \asm{INT}
          \code{n} --- disimpan ke stack;
    \item \flag{IF} dan \flag{TF} dikosongkan;
    \item alamat penangan dibaca dari entri vektor pada alamat
          $n \times 4$: dua byte pertama ke \reg{IP}, dua byte
          berikutnya ke \reg{CS}.
\end{enumerate}
{\footnotesize Ketiga penyimpanan itu menurunkan \reg{SP} sebanyak
\dec{6} byte.\par}
\end{column}

\begin{alertblock}{Penangan wajib diakhiri \asm{IRET}, bukan \asm{RET}}
\footnotesize \asm{IRET} mengambil \reg{IP}, \reg{CS}, lalu
\reg{FLAGS} --- kebalikan urutan penyimpanan; setara \asm{RET}
ditambah \asm{POPF}. Bila \asm{RET} yang dipakai, \reg{FLAGS}
terambil sebagai alamat kembali sehingga prosesor melompat ke alamat
acak.
\end{alertblock}
\Sumber{Bab XIV, Mekanisme INT dan IRET}
\end{frame}

\begin{frame}{Flag \flag{TF} dan Eksekusi Satu Langkah}
\small
\begin{itemize}
    \item Bit ke-8 register \reg{FLAGS} adalah \term{trap flag}
          \flag{TF}. Bila bit ini bernilai satu, prosesor menjalankan
          satu instruksi lalu segera membangkitkan interupsi vektor
          \hex{01}.

    \vspace{0.25em}
    \item Mekanisme inilah yang dipakai \eng{debugger} untuk
          menyediakan tombol \file{Single Step} pada Emu8086 atau
          perintah \file{T} pada DEBUG.

    \vspace{0.25em}
    \item Peran \flag{TF} menjelaskan langkah keempat mekanisme
          \asm{INT}: bila flag itu tetap bernilai satu, penangan akan
          memicu vektor \hex{01} setiap kali satu instruksinya selesai
          sehingga prosesor terjebak.
\end{itemize}

\vspace{0.3em}
\begin{block}{\file{T} dan \file{P} pada DEBUG}
\small Perintah \file{T} (\eng{trace}) menjalankan satu instruksi dan
\emph{masuk} ke rutin penangan, sedangkan \file{P} (\eng{proceed})
menjalankannya sampai selesai tanpa masuk ke penangan.
\end{block}
\Sumber{Bab XIV, Flag TF dan Eksekusi Satu Langkah}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item Interupsi lebih hemat daripada \eng{polling}: prosesor
          dipanggil hanya ketika ada kejadian, lalu melanjutkan
          program tepat pada instruksi berikutnya. Interupsi juga
          menjadi sarana baku meminta layanan sistem.

    \vspace{0.25em}
    \item Berdasarkan sumbernya ada tiga jenis: perangkat keras
          (\file{INTR} \eng{maskable} dan \file{NMI}), perangkat lunak
          (\asm{INT} \code{n}), dan pengecualian yang dibangkitkan
          prosesor.

    \vspace{0.25em}
    \item Tabel vektor menempati \hex{0000}--\hex{03FF} pada segmen
          nol; setiap vektor 4 byte (offset lalu segmen), sehingga
          alamat entri vektor ke-$n$ adalah $n \times 4$.

    \vspace{0.25em}
    \item \asm{INT} \code{n} menyimpan \reg{FLAGS}, \reg{CS}, lalu
          \reg{IP} (SP turun \dec{6} byte), mengosongkan \flag{IF} dan
          \flag{TF}, lalu memuat \reg{CS}:\reg{IP} dari vektor.
          \asm{IRET} mengembalikan ketiganya secara terbalik.

    \vspace{0.25em}
    \item \flag{TF} adalah dasar eksekusi satu langkah pada
          \eng{debugger}: satu instruksi, lalu interupsi vektor
          \hex{01}.
\end{itemize}
\end{frame}

\PemisahBagian{Interupsi BIOS: Video, Keyboard, dan Printer}

% ---------------------------------------------------------
\begin{frame}{Kedudukan BIOS}
\small
\begin{itemize}
    \item \term{BIOS} (\eng{Basic Input/Output System}) adalah
          kumpulan rutin yang tersimpan pada cip memori baca-saja dan
          dipetakan ke alamat \hex{F000} pada ruang memori.

    \vspace{0.25em}
    \item Rutin ini berbicara langsung dengan pengendali video,
          pengendali keyboard, dan port printer, lalu menyajikan
          antarmuka yang seragam bagi program di atasnya.

    \vspace{0.25em}
    \item Karena perbedaan perangkat keras disembunyikan di sini,
          program tidak perlu mengenal nomor port perangkat.
\end{itemize}

\vspace{0.35em}
\begin{exampleblock}{Konvensi seluruh layanan BIOS}
\small \reg{AH} berisi nomor fungsi, register lain berisi argumen, dan
hasil dikembalikan pada register.
\end{exampleblock}
\Sumber{Bab XIV, Kedudukan BIOS}
\end{frame}

\end{tikzpicture}
\end{center}

\vspace{0.3em}
{\small\reg{BL} = \hex{0E} berarti \bin{00001110}: bit 3--0 bernilai
\hex{E} (kuning terang) dan bit 6--4 bernilai \hex{0} (latar
hitam).\par}

\vspace{0.25em}
\begin{block}{Atribut yang sering dipakai}
\small \hex{07} abu-abu terang pada latar hitam (bawaan DOS),
\hex{0E} kuning, \hex{1F} putih pada latar biru, \hex{4E} kuning pada
latar merah.
\end{block}

\vspace{0.2em}
{\footnotesize\color{obeGray}Pemakaian \code{INT 10h} lebih aman
daripada menulis langsung ke memori video pada alamat \hex{B800},
karena BIOS sudah mengurus alamat karakter dan atributnya.\par}
\Sumber{Bab XIV, Byte Atribut Warna}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{\code{INT 16h}: Layanan Keyboard}
\small Interupsi \hex{16} menyediakan tiga fungsi utama.

\vspace{0.3em}
\begin{enumerate}
    \item \textbf{Fungsi \hex{00} --- baca tombol.} Prosesor menunggu
          sampai ada tombol ditekan, lalu mengembalikan \reg{AH} =
          \eng{scan code} dan \reg{AL} = kode ASCII. Untuk tombol
          panah, kode ASCII bernilai nol sehingga \reg{AH} yang harus
          diperiksa.

    \vspace{0.25em}
    \item \textbf{Fungsi \hex{01} --- periksa status tombol.} Fungsi
          ini tidak menunggu. Bila ada tombol pada penyangga keyboard,
          \flag{ZF} bernilai nol dan \reg{AX} berisi tombol tersebut
          tanpa mengeluarkannya.

    \vspace{0.25em}
    \item \textbf{Fungsi \hex{02} --- status tombol pengubah.}
          Mengembalikan keadaan \file{Shift}, \file{Ctrl},
          \file{Alt}, dan \file{Caps Lock} pada \reg{AL}.
\end{enumerate}

\vspace{0.3em}
\begin{alertblock}{BIOS tidak melakukan \eng{echo}}
\small Masukan DOS melakukan \eng{echo} ke layar dan menghormati
tombol \file{Ctrl}+\file{C}, sedangkan masukan BIOS tidak melakukan
keduanya. Program yang memakai BIOS bertanggung jawab mencetak sendiri
karakter yang dibaca.
\end{alertblock}
\Sumber{Bab XIV, INT 16h: Layanan Keyboard}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{\code{INT 17h}: Layanan Printer}
\small
\begin{itemize}
    \item Interupsi \hex{17} mengirim satu byte ke printer melalui
          \eng{port} paralel.
    \item Fungsi \hex{00} mencetak karakter pada \reg{AL}, fungsi
          \hex{01} menginisialisasi printer, dan fungsi \hex{02}
          membaca statusnya.
\end{itemize}

\vspace{0.4em}
\begin{exampleblock}{Bit status pada \reg{AH}}
\small
\begin{itemize}
    \item bit 1 --- waktu tunggu habis
    \item bit 3 --- kesalahan masukan/keluaran
    \item bit 5 --- kertas habis
    \item bit 7 --- printer sibuk
\end{itemize}
\end{exampleblock}
\Sumber{Bab XIV, INT 17h: Layanan Printer}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item BIOS adalah kumpulan rutin pada cip ROM yang dipetakan ke
          \hex{F000} dan menyembunyikan perbedaan perangkat keras;
          seluruh layanannya memakai \reg{AH} sebagai nomor fungsi.

    \vspace{0.25em}
    \item \code{INT 10h} melayani video: \hex{00} menetapkan mode,
          \hex{02} memindahkan kursor, \hex{03} membacanya, \hex{09}
          menulis karakter berulang tanpa menggeser kursor, dan
          \hex{13} menulis seluruh string pada \reg{ES}:\reg{BP}.

    \vspace{0.25em}
    \item Warna ditentukan byte atribut: bit 3--0 huruf, bit 6--4
          latar, bit 7 kedip; \hex{0E} kuning pada latar hitam.

    \vspace{0.25em}
    \item \code{INT 16h} melayani keyboard (fungsi \hex{00},
          \hex{01}, \hex{02}) tanpa \eng{echo} dan tanpa menghormati
          \file{Ctrl}+\file{C}.

    \vspace{0.25em}
    \item \code{INT 17h} mengirim satu byte ke printer melalui
          \eng{port} paralel dan mengembalikan bit status pada
          \reg{AH}.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Penyangga Fungsi \hex{0A}}
\small Fungsi \hex{0A} berbeda dari fungsi lain karena pengguna juga
menyediakan penyangga (\eng{buffer}) yang tata letaknya ditentukan
DOS.

\vspace{0.15em}
\begin{kodeasm}[numbers=none]
.DATA
    nama    DB 20           ; byte 0: kapasitas maksimum,
                            ;   termasuk tombol Enter
            DB 0            ; byte 1: jumlah karakter yang diketik,
                            ;   diisi oleh DOS
            DB 20 DUP(0)    ; byte 2 dan seterusnya: isi yang diketik
\end{kodeasm}

\vspace{0.15em}
{\footnotesize Byte pertama harus diisi \emph{sebelum} pemanggilan,
byte kedua diisi DOS dengan jumlah karakter yang benar-benar diketik.
DOS juga menambahkan \code{0Dh}, tetapi \emph{tidak} menambahkan tanda
\code{'\$'}.\par}

\vspace{0.15em}
\begin{alertblock}{Tanda \code{'\$'} harus disisipkan program}
\footnotesize Karena fungsi \hex{09} berhenti hanya ketika menemukan
\code{'\$'}, tanda itu dituliskan pada posisi
\code{penyangga + 2 + jumlah karakter}. Bila tidak, fungsi \hex{09}
terus mencetak isi memori sesudah penyangga sampai menemukan byte
\hex{24} --- layar dipenuhi karakter sampah.
\end{alertblock}
\Sumber{Bab XIV, Penyangga Fungsi 0Ah}
\end{frame}

\begin{frame}[fragile]{Memasang Penangan Interupsi Sendiri}
\scriptsize Mengganti isi sebuah entri vektor berarti mengganti
seluruh perilaku interupsi tersebut. Contoh memakai vektor \hex{60}
yang tidak dipakai perangkat keras mana pun.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.5\textwidth}
\begin{kodeasm}[numbers=none, basicstyle=\ttfamily\fontsize{7}{7.8}\selectfont]
MOV AH, 35h                ; baca vektor lama
MOV AL, 60h
INT 21h                    ; ES:BX = penangan lama
PUSH ES
PUSH BX                    ; simpan di stack
PUSH DS                    ; simpan DS program
MOV DX, OFFSET handler60   ; DX = offset baru
MOV AX, SEG handler60      ; AX = segmen baru
MOV DS, AX                 ; DS:DX = penangan baru
MOV AH, 25h                ; pasang vektor
MOV AL, 60h
INT 21h
POP DS
INT 60h                    ; uji penangan baru
\end{kodeasm}
\end{column}
\begin{column}{0.48\textwidth}
\begin{kodeasm}[numbers=none, basicstyle=\ttfamily\fontsize{7}{7.8}\selectfont]
handler60 PROC FAR
    PUSH AX                ; simpan register
    PUSH DX
    MOV AH, 09h            ; cetak pesan
    LEA DX, pesan
    INT 21h
    POP DX                 ; pulihkan register
    POP AX
    IRET                   ; pop IP, CS, FLAGS
handler60 ENDP
\end{kodeasm}
\end{column}
\end{columns}

\vspace{0.15em}
{\footnotesize\color{obeGray}Lima langkah: simpan penangan lama
(fungsi \hex{35}), siapkan alamat baru pada \reg{DS}:\reg{DX}, pasang
(fungsi \hex{25}), uji dengan \asm{INT} \code{n}, lalu pulihkan
penangan lama. Penangan adalah prosedur jauh yang diakhiri \asm{IRET};
karena dipanggil lewat interupsi, register yang dipakainya harus
disimpan dan dipulihkan sendiri.\par}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Memulihkan Vektor dan Risikonya}
\begin{columns}[t]
\begin{column}{0.42\textwidth}
{\footnotesize\color{obeBlue}\bfseries Memulihkan vektor lama}\\[0.15em]
\begin{kodeasm}[numbers=none]
POP DX          ; offset penangan lama
POP DS          ; segmen penangan lama
MOV AH, 25h     ; pasang kembali
MOV AL, 60h
INT 21h
\end{kodeasm}

\vspace{0.2em}
{\footnotesize\color{obeBlue}\bfseries Interupsi berantai}\\[0.1em]
{\footnotesize Penangan yang menggantikan interupsi perangkat keras
melanjutkan kendali ke penangan lama dengan \emph{melompat} ke
alamatnya, bukan dengan \asm{IRET}.\par}
\end{column}
\begin{column}{0.56\textwidth}
\begin{alertblock}{Tiga kesalahan yang paling sering terjadi}
\footnotesize
\begin{itemize}
    \item penangan tidak diakhiri \asm{IRET}, sehingga prosesor
          melompat ke alamat acak;
    \item vektor lama tidak dipulihkan, sehingga program berikutnya
          menerima penangan yang sudah tidak ada;
    \item penangan tidak menyimpan register yang dipakainya.
\end{itemize}
\end{alertblock}

\vspace{0.15em}
{\footnotesize\color{obeGray}Vektor \hex{08} dan \hex{09} sebaiknya
tidak dijadikan percobaan pertama.\par}
\end{column}
\end{columns}
\Sumber{Bab XIV, Memasang Penangan Interupsi Sendiri}
\end{frame}

\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item BIOS menggerakkan perangkat, DOS mengatur pemakaian sistem;
          keduanya memakai \reg{AH} sebagai nomor fungsi.

    \vspace{0.25em}
    \item Fungsi \code{INT 21h} yang penting: masukan (\hex{01},
          \hex{0A}), keluaran (\hex{02}, \hex{09}), pengelolaan vektor
          (\hex{25}, \hex{35}), dan terminasi (\hex{4C}).

    \vspace{0.25em}
    \item Penyangga fungsi \hex{0A}: byte 0 kapasitas, byte 1 jumlah

          penyangga akan dicetak dengan fungsi \hex{09}.

    \vspace{0.25em}
    \item Fungsi \hex{4C} lebih umum dan lebih aman daripada
          \code{INT 20h}, sehingga menjadi pilihan baku program
          \file{.EXE}.

    \vspace{0.25em}
    \item Penangan baru wajib menyimpan register yang dipakainya dan
          diakhiri \asm{IRET}; vektor lama wajib dipulihkan sesudah
          pengujian.
\end{itemize}

\begin{frame}[fragile]{Contoh 1: Menyapa Pengguna dengan \code{INT 21h}}
\scriptsize Meminta nama pengguna dengan fungsi \hex{0A}, menyisipkan
tanda \code{'\$'}, lalu menyapa pengguna dengan nama tersebut.

\vspace{0.1em}
\begin{columns}[t]
\begin{column}{0.46\textwidth}
\begin{kodeasm}[numbers=none, basicstyle=\ttfamily\fontsize{7.2}{8}\selectfont]
.MODEL SMALL
.STACK 100h
.DATA
    tanya  DB 'Siapa nama Anda? $'
    sapaan DB 0Dh,0Ah,'Halo, $'
    nama   DB 20        ; kapasitas
           DB 0         ; jumlah terisi
           DB 20 DUP(0) ; isi nama
    penutup DB '! Selamat belajar assembly.',0Dh,0Ah,'$'
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX          ; DS -> .DATA
    MOV AH, 09h
    LEA DX, tanya
    INT 21h             ; cetak pertanyaan
\end{kodeasm}
\end{column}
\begin{column}{0.52\textwidth}
\begin{kodeasm}[numbers=none, basicstyle=\ttfamily\fontsize{7.2}{8}\selectfont]
    MOV AH, 0Ah         ; baca nama
    LEA DX, nama
    INT 21h             ; DOS isi byte 1 dan 2..
    LEA SI, nama
    MOV BL, [SI+1]      ; jumlah karakter
    MOV BH, 0           ; BX = jumlah karakter
    ADD SI, 2           ; SI -> nama pertama
    ADD SI, BX          ; SI -> sesudah terakhir
    MOV BYTE PTR [SI], '$'
    MOV AH, 09h
    LEA DX, sapaan
    INT 21h             ; cetak "Halo, "
    LEA DX, nama+2      ; cetak nama pengguna
    INT 21h
    LEA DX, penutup
    INT 21h
    MOV AH, 4Ch
    INT 21h             ; terminasi program
\end{kodeasm}
\end{column}
\end{columns}

{\scriptsize\color{obeGray}Keluaran: \code{Siapa nama Anda? Andi},
lalu \code{Halo, Andi! Selamat belajar assembly.} Fungsi \hex{0A}
sudah menampilkan (\eng{echo}) karakter yang diketik. Sumber:
\file{input-nama.asm}.\par}
\end{frame}

\begin{kodeasm}[numbers=none, basicstyle=\ttfamily\fontsize{7}{7.8}\selectfont]

    LEA BP, judul

    MOV DH, 07h       ; baris 7

    \item Sebutkan tiga jenis interupsi pada prosesor 8086. Untuk
          setiap jenis, berikan satu contoh beserta nomor vektornya
          dan siapa yang memicunya.

    \vspace{0.28em}
    \item Dengan memakai rumus alamat vektor$_n = n \times 4$,
          tentukan alamat entri vektor \hex{0A}, \hex{13}, dan
          \hex{21}. Sebutkan pula alamat byte pertama (offset) dan
          byte ketiga (segmen) setiap entri.

    \vspace{0.28em}
    \item Sebuah program berjalan dengan \reg{SP} = \hex{FF00} ketika
          \asm{INT 21h} dijalankan. Berapa \reg{SP} sesudah ketiga
          penyimpanan selesai, nilai apa yang berada pada
          \reg{SS}:\reg{SP}, \reg{SS}:\reg{SP}+2, dan
          \reg{SS}:\reg{SP}+4, dan mengapa \asm{RET} tidak dapat
          menggantikan \asm{IRET}?

    \vspace{0.28em}
    \item Tentukan warna huruf dan warna latar untuk atribut \hex{1E},
          \hex{4F}, dan \hex{07}, dengan menyebut nilai bit 3--0, bit
          6--4, dan bit 7.
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\footnotesize
\begin{itemize}
    \item Interupsi memaksa prosesor menghentikan sementara alur
          program untuk melayani suatu kejadian, lalu melanjutkannya
          kembali; mekanisme ini lebih hemat daripada \eng{polling}.

    \vspace{0.18em}
    \item Berdasarkan sumbernya: interupsi perangkat keras
          (\file{INTR} dan \file{NMI}), interupsi perangkat lunak
          (\asm{INT} \code{n}), dan pengecualian yang dibangkitkan
          prosesor, misalnya pembagian dengan nol.

    \vspace{0.18em}
    \item Tabel vektor menempati \hex{0000}--\hex{03FF} pada segmen
          nol; setiap vektor 4 byte (2 byte offset dan 2 byte segmen),
          sehingga alamat entri vektor ke-$n$ adalah $n \times 4$.

    \vspace{0.18em}
    \item \asm{INT} \code{n} menyimpan \reg{FLAGS}, \reg{CS}, lalu
          \reg{IP} ke stack, mengosongkan \flag{IF} dan \flag{TF},
          kemudian memuat \reg{CS}:\reg{IP} dari entri vektor.
          \asm{IRET} mengembalikan ketiganya secara terbalik, setara
          \asm{RET} ditambah \asm{POPF}.

    \vspace{0.18em}
    \item BIOS melayani perangkat keras (\code{INT 10h} video,
          \code{INT 16h} keyboard, \code{INT 17h} printer), sedangkan
          \code{INT 21h} melayani keperluan tingkat program: masukan
          (\hex{01}, \hex{0A}), keluaran (\hex{02}, \hex{09}),
          pengelolaan vektor (\hex{25}, \hex{35}), dan terminasi
          (\hex{4C}).

    \vspace{0.18em}
    \item Fungsi \hex{0A} memakai penyangga dengan tata letak dari
          DOS, dan isi vektor dapat diganti melalui fungsi \hex{25};
          penangan baru wajib menyimpan register dan diakhiri
          \asm{IRET}, sedangkan vektor lama wajib dipulihkan.
\end{itemize}
\end{frame}

\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Setiap entri pada tabel vektor interupsi 8086 berukuran
          \dots
          \begin{enumerate}\scriptsize
              \item 2 byte berisi offset penangan \quad
              \item 4 byte: 2 byte offset dan 2 byte segmen \\
              \item 4 byte berisi nomor vektor \quad
              \item 8 byte: \reg{IP}, \reg{CS}, dan \reg{FLAGS}
          \end{enumerate}

    \vspace{0.2em}
    \item Alamat entri untuk vektor \hex{10} adalah \dots
          \begin{enumerate}\scriptsize
              \item \hex{0010} \quad \item \hex{0040} \quad
              \item \hex{0100} \quad \item \hex{0160}
          \end{enumerate}

    \vspace{0.2em}
    \item Urutan penyimpanan ke stack ketika \asm{INT} \code{n}
          dijalankan adalah \dots
          \begin{enumerate}\scriptsize
              \item \reg{IP}, \reg{CS}, \reg{FLAGS} \quad
              \item \reg{CS}, \reg{IP}, \reg{FLAGS} \\
              \item \reg{FLAGS}, \reg{CS}, \reg{IP} \quad
              \item \reg{FLAGS}, \reg{IP}, \reg{CS}
          \end{enumerate}
\end{enumerate}
\end{frame}
