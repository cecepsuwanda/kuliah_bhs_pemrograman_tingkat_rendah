\PemisahBagian{Array Satu dan Multi Dimensi}

% ---------------------------------------------------------
\begin{frame}[fragile]{Array Satu Dimensi}
\small
Array dideklarasikan dengan DB, DW, atau DD, lalu diisi nilai yang
dipisahkan koma.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
deret   DW 10, 20, 30, 40, 50
BANYAK  EQU ($ - deret) / 2
\end{kodeasm}

\vspace{0.15em}
\begin{itemize}
    \item Setiap elemen word menempati dua byte.
    \item Elemen pertama di offset \code{deret}, kedua di
          \code{deret+2}, dan seterusnya.
    \item \code{BANYAK} dihitung dari selisih alamat.
\end{itemize}
\Sumber{Bab XV, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Menjumlahkan Array}
\small
\reg{SI} menunjuk elemen saat ini; setiap iterasi maju dua byte
karena satu word berukuran dua byte.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
LEA SI, deret
MOV CX, BANYAK
XOR AX, AX
JUMLAH:
    ADD AX, [SI]
    ADD SI, 2
    LOOP JUMLAH
MOV jumlah, AX
\end{kodeasm}
\Sumber{Bab XV, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Array Multi Dimensi}
\small
Array multi dimensi disimpan baris demi baris. Untuk matriks
\code{BARIS} $\times$ \code{KOLOM}:

\[
\text{offset} = (r \times \text{KOLOM} + c) \times
\text{ukuran elemen}
\]

\vspace{0.1em}
\begin{kodeasm}[numbers=none]
matriks DW 1, 2, 3, 4
        DW 5, 6, 7, 8
        DW 9,10,11,12
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Prosesor hanya melihat deretan byte;
program menghitung baris dan kolom lebih dahulu.\par}
\Sumber{Bab XV, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Offset Matriks 3 $\times$ 4}
\small
\begin{center}
\begin{tabular}{@{}cccc@{}}
    \toprule
    \textbf{Baris} & \textbf{Kolom} & \textbf{Perhitungan} &
    \textbf{Offset} \\
    \midrule
    0 & 0 & $(0 \times 4 + 0) \times 2$ & 0 \\
    \addlinespace
    0 & 3 & $(0 \times 4 + 3) \times 2$ & 6 \\
    \addlinespace
    1 & 0 & $(1 \times 4 + 0) \times 2$ & 8 \\
    \addlinespace
    1 & 2 & $(1 \times 4 + 2) \times 2$ & 12 \\
    \addlinespace
    2 & 3 & $(2 \times 4 + 3) \times 2$ & 22 \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XV, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item Array satu dimensi disimpan berurutan; elemen word
          berjarak dua byte.

    \vspace{0.15em}
    \item Akses array memakai register penunjuk dan LOOP.

    \vspace{0.15em}
    \item Array multi dimensi disimpan baris demi baris.

    \vspace{0.15em}
    \item Offset = (baris $\times$ kolom + kolom) $\times$ ukuran
          elemen.

    \vspace{0.15em}
    \item Prosesor tidak mengenal struktur array; program menghitung
          alamatnya.
\end{itemize}
\end{frame}

\PemisahBagian{String}

% ---------------------------------------------------------
\begin{frame}[fragile]{String sebagai Array Byte}
\small
String adalah deretan karakter ASCII yang disimpan sebagai byte
berurutan.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
sumber  DB 'ASSEMBLY$'
tujuan  DB 12 DUP(?)
panjang DB ?
\end{kodeasm}

\vspace{0.15em}
\begin{itemize}
    \item Akhir string ditandai \code{'\$'} atau panjang tersimpan.
    \item Buffer tujuan harus cukup untuk salinan dan penanda akhir.
\end{itemize}
\Sumber{Bab XV, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Menyalin String}
\small
Salin byte per byte sampai penanda akhir ditemukan.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
LEA SI, sumber
LEA DI, tujuan
XOR CX, CX
SALIN:
    MOV AL, [SI]
    MOV [DI], AL
    CMP AL, '$'
    JE  SELESAI
    INC SI
    INC DI
    INC CX
    JMP SALIN
SELESAI:
\end{kodeasm}
\Sumber{Bab XV, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Membandingkan String}
\small
Perbandingan dilakukan karakter demi karakter; bila berbeda,
perbandingan dihentikan.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
LEA SI, string1
LEA DI, string2
MOV CX, panjang
BANDING:
    MOV AL, [SI]
    CMP AL, [DI]
    JNE BERBEDA
    INC SI
    INC DI
    LOOP BANDING
    JMP SELESAI
BERBEDA:
\end{kodeasm}
\Sumber{Bab XV, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Operasi String}
\footnotesize
Pencarian memakai \code{CMP} di dalam perulangan; bila ditemukan,
\reg{SI} menunjuk karakter tersebut.

\vspace{0.2em}
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.8cm}
    >{\raggedright\arraybackslash}p{4.2cm}
    >{\raggedright\arraybackslash}p{5.8cm}@{}}
    \toprule
    \textbf{Operasi} & \textbf{Instruksi utama} &
    \textbf{Keterangan} \\
    \midrule
    Menyalin & MOV, INC, CMP, JMP & salin sampai penanda akhir \\
    \addlinespace
    Membandingkan & CMP, JNE, LOOP & hentikan bila berbeda \\
    \addlinespace
    Mencari & CMP, JE, INC, JMP & hasil berupa alamat di SI \\
    \addlinespace
    Menghitung panjang & CMP, INC, JMP & hitung byte sampai akhir \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XV, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item String adalah array byte berisi kode ASCII.

    \vspace{0.15em}
    \item Akhir string ditandai \code{'\$'} atau panjang tersimpan.

    \vspace{0.15em}
    \item Penyalinan dilakukan byte per byte sampai penanda akhir.

    \vspace{0.15em}
    \item Perbandingan memakai CMP dan berhenti saat berbeda.

    \vspace{0.15em}
    \item Pencarian menghasilkan alamat di SI.
\end{itemize}
\end{frame}

\PemisahBagian{Mencetak Bilangan Desimal dan Heksadesimal}

% ---------------------------------------------------------
\begin{frame}{Konversi ke Digit Desimal}
\small
\begin{enumerate}
    \item Bagi bilangan dengan 10.
    \item Simpan sisa pembagian sebagai satu digit.
    \item Ulangi sampai hasil bagi nol.
    \item Ambil digit dari yang terakhir disimpan, lalu cetak.
\end{enumerate}

\vspace{0.25em}
\begin{block}{Mengapa stack?}
\small Sisa pembagian disimpan pada \eng{stack} agar urutan cetak
terbalik menjadi benar.
\end{block}

\vspace{0.15em}
{\footnotesize\color{obeGray}Bilangan nol tetap menghasilkan satu
digit \code{'0'}.\par}
\Sumber{Bab XV, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Kode CETAK\_DESIMAL}
\begin{kodeasm}
MOV CX, 0
MOV BX, 10
BAGI:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    CMP AX, 0
    JNE BAGI
CETAK_DIGIT:
    POP DX
    ADD DL, '0'
    MOV AH, 02h
    INT 21h
    LOOP CETAK_DIGIT
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Simpan AX, BX, CX, dan DX dengan PUSH
di awal prosedur dan pulihkan dengan POP di akhir.\par}
\Sumber{Bab XV, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Konversi ke Digit Heksadesimal}
\small
\begin{itemize}
    \item Heksadesimal memakai empat bit per digit.

    \vspace{0.18em}
    \item Digit diambil dengan \code{ROL AX, 4}, lalu disaring
          dengan \code{AND DL, 0Fh}.

    \vspace{0.18em}
    \item Nilai 0--9 ditambah \code{'0'}; nilai 10--15 ditambah 7
          lalu \code{'0'}, sehingga menjadi \code{'A'}--\code{'F'}.

    \vspace{0.18em}
    \item Pencetakan selalu menghasilkan empat digit.
\end{itemize}

\vspace{0.25em}
\begin{block}{Contoh}
\small \hex{3039} = 12345 desimal.
\end{block}
\Sumber{Bab XV, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Kode CETAK\_HEX dan Tabel ASCII}
\begin{columns}[t]
\begin{column}{0.55\textwidth}
\begin{kodeasm}
MOV CX, 4
HEX_ULANG:
    ROL AX, 4
    MOV DL, AL
    AND DL, 0Fh
    CMP DL, 9
    JBE HEX_ANGKA
    ADD DL, 7
HEX_ANGKA:
    ADD DL, '0'
    MOV AH, 02h
    INT 21h
    LOOP HEX_ULANG
\end{kodeasm}
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
\begin{tabular}{@{}p{1.4cm}p{1.6cm}p{1.6cm}@{}}
    \toprule
    \textbf{Nilai} & \textbf{Hex} & \textbf{Karakter} \\
    \midrule
    0--9 & 30--39 & '0'--'9' \\
    \addlinespace
    10--15 & 41--46 & 'A'--'F' \\
    \bottomrule
\end{tabular}

\vspace{0.3em}
{\color{obeGray}Nilai 10--15 ditambah 7 sebelum ditambah
\code{'0'}.\par}
\end{column}
\end{columns}
\Sumber{Bab XV, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item Desimal: bagi dengan 10, simpan sisa, cetak terbalik.

    \vspace{0.15em}
    \item Sisa disimpan pada stack agar urutan benar.

    \vspace{0.15em}
    \item Heksadesimal: ambil empat bit dengan ROL dan AND 0Fh.

    \vspace{0.15em}
    \item Digit 0--9 menjadi '0'--'9'; 10--15 menjadi 'A'--'F'.

    \vspace{0.15em}
    \item Pencetakan memakai INT 21h fungsi 02h.
\end{itemize}
\end{frame}

\begin{frame}{Latihan}
\small
\begin{enumerate}
    \item Jelaskan cara mendeklarasikan array word lima elemen.
          Tentukan offset elemen pertama, kedua, dan kelima.

    \vspace{0.15em}
    \item Diketahui matriks word $3 \times 4$. Tentukan offset
          elemen baris 2 kolom 1 dan baris 1 kolom 3.

    \vspace{0.15em}
    \item Jelaskan representasi string di memori. Mengapa string
          fungsi \hex{09} harus diakhiri \code{'\$'}?

    \vspace{0.15em}
    \item Tuliskan perulangan menyalin string dari SI ke DI sampai
          \code{'\$'}. Kapan perbandingan dua string dapat
          dihentikan?

    \vspace{0.15em}
    \item Jelaskan algoritme konversi desimal. Mengapa sisa
          pembagian disimpan pada stack?
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN

\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item Array satu dimensi disimpan berurutan; word berjarak dua
          byte.

    \vspace{0.15em}
    \item Array multi dimensi disimpan baris demi baris dengan
          offset $(r \times K + c) \times$ ukuran.

    \vspace{0.15em}
    \item String adalah array byte ASCII dengan penanda akhir.

    \vspace{0.15em}
    \item Salin dan bandingkan string dilakukan byte per byte.

    \vspace{0.15em}
    \item Desimal memakai pembagian 10 dan stack; heksadesimal
          memakai ROL/AND 0Fh.

    \vspace{0.15em}
    \item Pencetakan memakai layanan DOS fungsi 02h.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Elemen ketiga array word \code{deret} berada pada offset
          \dots
          \begin{enumerate}
              \item \code{deret+1} \quad \item \code{deret+2} \quad
              \item \code{deret+4} \quad \item \code{deret+6}
          \end{enumerate}

    \vspace{0.2em}
    \item Offset elemen baris $r$ kolom $c$ pada matriks word dengan
          $K$ kolom adalah \dots
          \begin{enumerate}
              \item $(r+c)\times2$
              \item $(r\times K+c)\times2$
              \item $(r\times c+K)\times2$
              \item $(K\times c+r)\times2$
          \end{enumerate}

    \vspace{0.2em}
    \item Konversi desimal memakai pembagian berulang oleh \dots
          \begin{enumerate}
              \item 2 \quad \item 8 \quad
              \item 10 \quad \item 16
          \end{enumerate}
\end{enumerate}
\end{frame}
