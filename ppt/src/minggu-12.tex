\PemisahBagian{Label, JMP, dan CMP}

% ---------------------------------------------------------
\begin{frame}{Label}
\small
\begin{itemize}
    \item Nama label boleh dimulai huruf, \code{@}, \code{\_}, atau
          \code{?}; tidak boleh berspasi atau diawali angka.

    \vspace{0.18em}
    \item Panjang maksimal 31 karakter dan harus unik.

    \vspace{0.18em}
    \item Label didefinisikan dengan titik dua, misalnya
          \code{ULANG:}.

    \vspace{0.18em}
    \item Saat menjadi tujuan lompatan, label ditulis tanpa titik dua.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{Nama ganda}
\small Label yang sama didefinisikan dua kali menghasilkan pesan
\file{Symbol already defined}.
\end{alertblock}
\Sumber{Bab XIII, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Lompatan Tanpa Syarat JMP}
\small
\code{JMP} memindahkan kendali ke label tujuan tanpa memeriksa
\eng{flag}.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
    JMP SELESAI         ; langsung menuju SELESAI
    MOV AX, 1           ; dilewati
SELESAI:
    ; eksekusi berlanjut di sini
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Dipakai untuk melewati blok, mengakhiri
percabangan, atau membentuk perulangan tanpa syarat.\par}
\Sumber{Bab XIII, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{CMP dan Flag}
\footnotesize
\code{CMP} mengurangkan operand kedua dari operand pertama tanpa
menyimpan hasil; hanya \eng{flag} yang berubah.

\vspace{0.2em}
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.4cm}
    >{\raggedright\arraybackslash}p{5.2cm}
    >{\raggedright\arraybackslash}p{6.2cm}@{}}
    \toprule
    \textbf{Flag} & \textbf{Makna} & \textbf{Dipakai oleh} \\
    \midrule
    ZF & Hasil operasi nol & JE, JNE, JZ, JNZ \\
    \addlinespace
    CF & Pinjaman/simpanan tak bertanda & JB, JAE, JC, JNC \\
    \addlinespace
    SF & Bit tanda hasil & JS, JNS \\
    \addlinespace
    OF & Luapan bertanda & JO, JNO \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XIII, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Percabangan if}
\small
Pola \code{if} menjadi \code{CMP} diikuti lompatan bersyarat.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
; if (AX > BX) { AX = BX }
CMP AX, BX
JLE TIDAK_LEBIH
MOV AX, BX
TIDAK_LEBIH:
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Blok yang dijalankan diletakkan di antara
lompatan dan label akhir.\par}
\Sumber{Bab XIII, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{if dengan Blok Panjang}
\small
Lompatan bersyarat 8086 hanya menjangkau jarak terbatas. Bila blok
\code{if} sangat panjang, balik syarat dan gunakan \code{JMP} sebagai
perantara.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
JLE LANJUT
JMP BLOK_PANJANG
LANJUT:
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Pola ini menghindari kesalahan jangkauan
lompatan.\par}
\Sumber{Bab XIII, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item Label menandai lokasi kode; definisi memakai titik dua,
          tujuan lompatan tanpa titik dua.

    \vspace{0.15em}
    \item JMP memindahkan kendali tanpa syarat.

    \vspace{0.15em}
    \item CMP memperbarui flag tanpa menyimpan hasil.

    \vspace{0.15em}
    \item Pola if memakai CMP dan satu lompatan bersyarat.

    \vspace{0.15em}
    \item Blok panjang dapat memakai JMP perantara.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Contoh: FF vs 1}
\small
\hex{FF} dapat dibaca sebagai 255 (tidak bertanda) atau $-1$
(bertanda).

\vspace{0.25em}
\begin{itemize}
    \item Sebagai bilangan tidak bertanda, \hex{FF} $>$ 1: gunakan
          \code{JA}.
    \item Sebagai bilangan bertanda, \hex{FF} $<$ 1: gunakan
          \code{JL}.
\end{itemize}

\vspace{0.25em}
\begin{block}{Inti}
\small Pola bit sama, tafsir berbeda; pilihan Jcc harus mengikuti
jenis bilangan.
\end{block}
\Sumber{Bab XIII, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Percabangan if--else}
\small
Pola \code{if--else} memerlukan satu \code{JMP} untuk melewati blok
\code{else}.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
; if (AX == BX) { hasil = 1 } else { hasil = 0 }
CMP AX, BX
JNE BLOK_ELSE
MOV hasil, 1
JMP SELESAI
BLOK_ELSE:
MOV hasil, 0
SELESAI:
\end{kodeasm}
\Sumber{Bab XIII, Bagian 2}
\end{frame}

\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item Jcc memeriksa flag; syarat terpenuhi berarti lompat.

    \vspace{0.15em}
    \item JG/JGE/JL/JLE untuk bilangan bertanda.

    \vspace{0.15em}
    \item JA/JAE/JB/JBE untuk bilangan tidak bertanda.

    \vspace{0.15em}
    \item if--else memerlukan JMP untuk melewati blok else.

    \vspace{0.15em}
    \item Jangan menyisipkan instruksi yang mengubah flag antara CMP
          dan Jcc.
\end{itemize}
\end{frame}

\PemisahBagian{Perulangan}

% ---------------------------------------------------------
\begin{frame}[fragile]{Instruksi LOOP}
\small
\code{LOOP} mengurangi CX dengan satu, lalu melompat bila CX belum
nol.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV CX, 5
ULANG:
    ; badan perulangan
    LOOP ULANG      ; CX--; lompat bila CX <> 0
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Badan selalu dijalankan minimal sekali.
Bila CX = 0 di awal, LOOP mengulang 65.536 kali; gunakan JCXZ untuk
mencegahnya.\par}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Varian LOOP}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.8cm}
    >{\raggedright\arraybackslash}p{5.0cm}
    >{\raggedright\arraybackslash}p{5.0cm}@{}}
    \toprule
    \textbf{Instruksi} & \textbf{Aksi} & \textbf{Kegunaan} \\
    \midrule
    LOOP & CX $\leftarrow$ CX $-1$; lompat bila CX $\neq0$ &
    Cacah diketahui \\
    \addlinespace
    LOOPE/LOOPZ & Seperti LOOP, tetapi hanya bila ZF = 1 &
    Mencari ketidakcocokan \\
    \addlinespace
    LOOPNE/LOOPNZ & Seperti LOOP, tetapi hanya bila ZF = 0 &
    Berhenti saat nilai ditemukan \\
    \addlinespace
    JCXZ & Lompat bila CX = 0; CX tidak berubah &
    Menangani cacah nol \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{DEC/JNZ dan while}
\small
\code{LOOP} setara dengan \code{DEC CX} lalu \code{JNZ}, tetapi
pasangan itu lebih luwes karena pencacah tidak harus CX.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
; while (AX < 10) { AX = AX + 1 }
ULANG_WHILE:
    CMP AX, 10
    JGE SELESAI_WHILE
    INC AX
    JMP ULANG_WHILE
SELESAI_WHILE:
\end{kodeasm}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pola for}
\small
Perulangan \code{for} dengan cacah tetap lebih ringkas memakai
\code{LOOP}.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
; for (CX = 8; CX > 0; CX--) { proses }
MOV CX, 8
ULANG_FOR:
    ; badan perulangan
    LOOP ULANG_FOR
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Periksa CX bila cacah perulangan mungkin
nol.\par}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Perulangan Bersarang}
\small
\code{LOOP} selalu memakai CX. Pencacah luar harus disimpan sebelum
perulangan dalam dimulai.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV CX, 3
LUAR:
    PUSH CX
    MOV CX, 4
DALAM:
    ; badan perulangan dalam
    LOOP DALAM
    POP CX
    LOOP LUAR
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeMaroon}Kesalahan umum: lupa memulihkan
pencacah luar.\par}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pola Nilai Maksimum}
\small
Kandidat maksimum disimpan di AX; setiap elemen dibandingkan dengan
kandidat.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
LEA SI, deret
MOV CX, BANYAK - 1
MOV AX, [SI]
ADD SI, 2
CARI:
    CMP AX, [SI]
    JAE LANJUT
    MOV AX, [SI]
LANJUT:
    ADD SI, 2
    LOOP CARI
\end{kodeasm}
\Sumber{Bab XIII, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item LOOP mengurangi CX dan melompat bila CX belum nol.

    \vspace{0.15em}
    \item LOOPE/LOOPNE menambahkan syarat ZF.

    \vspace{0.15em}
    \item JCXZ menangani cacah nol.

    \vspace{0.15em}
    \item while memeriksa syarat sebelum badan; for memakai LOOP.

    \vspace{0.15em}
    \item Perulangan bersarang menyimpan pencacah luar dengan
          PUSH/POP.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Contoh 1: Mencari Nilai Maksimum}
\begin{columns}[t]
\begin{column}{0.53\textwidth}
\begin{kodeasm}
.CODE
START:

    LEA SI, deret

CARI:
    ADD SI, 2

LANJUT:

\end{kodeasm}

\end{keluaran}

\vspace{0.3em}
\begin{itemize}
    \item AX menyimpan kandidat maksimum.
    \item JAE memakai tafsiran tidak bertanda.
    \item LOOP mengulang sampai semua elemen diperiksa.
\end{itemize}

\vspace{0.2em}
{\footnotesize\color{obeGray}\code{deret} dan \code{BANYAK}

\end{column}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh 2: Pengurutan Gelembung}
\begin{columns}[t]
\begin{column}{0.53\textwidth}
\begin{kodeasm}
LINTASAN:
    PUSH CX
    MOV SI, 0
    MOV CX, BANYAK-1
BANDING:
    MOV AL, deret[SI]
    MOV BL, deret[SI+1]
    CMP AL, BL
    JBE TIDAK_TUKAR
    MOV deret[SI], BL
    MOV deret[SI+1], AL
TIDAK_TUKAR:
    INC SI
    LOOP BANDING
\end{kodeasm}
\end{column}
\begin{column}{0.44\textwidth}
\footnotesize
{\color{obeTeal}\bfseries Keluaran}\\[0.15em]
\begin{keluaran}

\end{keluaran}

\vspace{0.3em}
\begin{itemize}
    \item PUSH CX menyimpan pencacah luar.
    \item JBE melewati penukaran bila sudah urut.
    \item Setelah LOOP BANDING, POP CX dan LOOP LINTASAN
          melanjutkan lintasan berikutnya.
\end{itemize}
\end{column}
\end{columns}

\begin{frame}{Latihan}
\small
\begin{enumerate}
    \item Jelaskan aturan penulisan label. Mengapa definisi memakai
          titik dua, sedangkan tujuan lompatan tidak?

    \vspace{0.15em}
    \item Jelaskan perbedaan JMP dan Jcc. Kapan masing-masing
          dipakai?

    \vspace{0.15em}
    \item Jelaskan cara kerja CMP. Mengapa CMP tidak mengubah operand
          tetapi dapat dipakai mengambil keputusan?

    \vspace{0.15em}
    \item Setelah CMP AX, BX, tentukan Jcc untuk AX $>$ BX
          (bertanda), AX $<$ BX (tidak bertanda), dan AX $=$ BX.

    \vspace{0.15em}
    \item Jelaskan cara kerja LOOP. Apa yang terjadi bila CX = 0 di
          awal, dan bagaimana mencegahnya?
\end{enumerate}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Instruksi yang memindahkan kendali tanpa memeriksa flag
          adalah \dots
          \begin{enumerate}
              \item JE \quad \item JNE \quad
              \item JMP \quad \item LOOP
          \end{enumerate}

    \vspace{0.2em}
    \item \code{CMP AX, BX} \dots
          \begin{enumerate}
              \item mengubah AX dan BX
              \item hanya mengubah flag
              \item menyimpan hasil di AX
              \item menyimpan hasil di memori
          \end{enumerate}

    \vspace{0.2em}
    \item LOOP memakai register \dots{} sebagai pencacah.
          \begin{enumerate}
              \item AX \quad \item BX \quad
              \item CX \quad \item DX
          \end{enumerate}
\end{enumerate}
\end{frame}
