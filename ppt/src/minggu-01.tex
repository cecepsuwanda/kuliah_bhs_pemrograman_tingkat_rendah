\begin{block}{Inti perbedaannya}
\small \term{Tingkat abstraksi} adalah seberapa jauh bahasa
menyembunyikan detail perangkat keras dari programmer.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Kapan Pemrograman Tingkat Rendah Digunakan}
\small
\begin{itemize}
    \item \textbf{Kendali penuh atas perangkat keras.}
          \eng{Device driver}, \eng{bootloader}, dan \eng{firmware}
          memerlukan akses langsung ke register dan port I/O.

    \vspace{0.3em}
    \item \textbf{Kinerja dan ukuran kode.}
          Pada sistem tertanam dengan memori terbatas, program assembly
          dapat jauh lebih kecil dan lebih cepat.

    \vspace{0.3em}
    \item \textbf{Analisis dan keamanan.}
          Membaca hasil \eng{disassembly}, menganalisis
          \eng{malware}, dan menguji keamanan perangkat lunak.

    \vspace{0.3em}
    \item \textbf{Pendidikan.}
          Memperlihatkan secara eksplisit bagaimana prosesor mengambil
          instruksi, memindahkan data, dan mengambil keputusan
          berdasarkan \eng{flag}.
\end{itemize}

\vspace{0.3em}
\begin{alertblock}{Yang perlu diingat}
\small Tingkat rendah tidak berarti ``lebih baik''. Keduanya alat
dengan tujuan berbeda.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item Tingkat abstraksi ditentukan oleh seberapa jauh sebuah
          bahasa menyembunyikan detail perangkat keras. Urutannya dari
          bawah ke atas: perangkat keras, bahasa mesin, assembly,
          bahasa tingkat tinggi.

    \vspace{0.25em}
    \item Operasi yang sama dapat dituliskan pada setiap tingkat:
          \hex{B8 05 00} (mesin), \code{MOV AX, 5} (assembly), dan
          \code{x = 5;} (tingkat tinggi).

    \vspace{0.25em}
    \item Hubungan assembly dan bahasa mesin bersifat
          \eng{one-to-one}, sedangkan satu baris bahasa tingkat tinggi
          dapat mewakili banyak instruksi mesin.

    \vspace{0.25em}
    \item Semakin rendah tingkatnya, semakin penuh kendali atas
          register, memori, dan port I/O, tetapi portabilitas,
          produktivitas, dan kemudahan pemeliharaan semakin rendah.

    \vspace{0.25em}
    \item Pemrograman tingkat rendah dipilih ketika kendali perangkat
          keras, ukuran atau kecepatan kode, atau analisis keamanan
          menjadi pertimbangan utama.
\end{itemize}
\end{frame}

\vspace{0.3em}
\small
\begin{enumerate}
    \item \textbf{Penyuntingan} — menulis kode \file{.asm}.
    \item \textbf{Perakitan} — assembler menerjemahkan \eng{mnemonic}
          menjadi kode mesin $\rightarrow$ \file{.obj}.
    \item \textbf{Pengaitan} — linker menyelesaikan alamat dan
          menggabungkan pustaka $\rightarrow$ \file{.exe}.
    \item \textbf{Pemuatan} — \eng{loader} memuat ke memori dan
          menyerahkan kendali ke instruksi pertama.
\end{enumerate}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Assembler, Linker, dan Loader}
\small
\begin{description}
    \item[Assembler] Menerjemahkan kode sumber assembly menjadi kode
          mesin. Karena setiap instruksi assembly bersesuaian dengan
          satu instruksi mesin, penerjemahan ini berjalan
          \eng{one-to-one} ditambah penanganan direktif dan simbol.

    \vspace{0.35em}
    \item[Linker] Menyelesaikan rujukan antarberkas. Bila program
          memanggil prosedur di berkas lain atau di pustaka,
          linker-lah yang menyambungkannya dan menetapkan alamat akhir.

    \vspace{0.35em}
    \item[Loader] Bagian sistem operasi yang memuat \file{.exe} ke
          memori, membuat \eng{Program Segment Prefix} (PSP),
          menetapkan register segmen, dan melompat ke titik masuk.
\end{description}

\vspace{0.4em}
\begin{block}{Pembagian tugas}
\small Assembler $\rightarrow$ kode mesin per instruksi \quad\textbullet\quad
Linker $\rightarrow$ alamat akhir \quad\textbullet\quad
Loader $\rightarrow$ memuat ke memori.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Program \file{.COM} dan \file{.EXE}}
\small
\begin{center}
\begin{tabular}{@{}p{2.9cm}p{5.0cm}p{5.0cm}@{}}

    \textbf{Aspek} & \textbf{Program \file{.COM}} &
    \textbf{Program \file{.EXE}} \\
    \midrule
    Ukuran maksimum & 64 KB dikurangi 256 byte &
    Tidak dibatasi satu segmen \\
    \addlinespace

    \addlinespace

    Harus sendiri: \code{MOV AX, @DATA} lalu \code{MOV DS, AX} \\
    \addlinespace
    Cara keluar & \code{INT 20h} &
    \code{MOV AH, 4Ch} lalu \code{INT 21h} \\
    \bottomrule
\end{tabular}
\end{center}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Jebakan Paling Sering: Register \reg{DS}}
\begin{alertblock}{Kesalahan klasik pada program \file{.EXE}}
\small Lupa menginisialisasi register \reg{DS}. Akibatnya instruksi
yang mengakses variabel membaca alamat yang salah dan program mencetak
karakter sampah.
\end{alertblock}

\vspace{0.5em}
\begin{block}{Mengapa \file{.COM} tidak perlu?}
\small DOS sudah menetapkan seluruh register segmen ke segmen program,
sehingga variabel dapat langsung dialamatkan.
\end{block}

\vspace{0.6em}
{\small\color{obeGray}Dua instruksi wajib pada awal program \file{.EXE}:\par}
\vspace{0.3em}
\begin{center}
\code{MOV AX, @DATA} \quad\textbullet\quad \code{MOV DS, AX}
\end{center}
\end{frame}

\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item Empat tahap rantai proses: penyuntingan \file{.asm},
          perakitan \file{.obj}, pengaitan \file{.exe}, lalu pemuatan
          ke memori oleh \eng{loader}.

    \vspace{0.25em}
    \item Pembagian tugas: assembler menerjemahkan \eng{mnemonic}
          menjadi kode mesin, linker menyelesaikan rujukan antarberkas
          dan menetapkan alamat akhir, \eng{loader} membuat PSP dan
          melompat ke titik masuk.

    \vspace{0.25em}
    \item \file{.COM}: satu segmen, maksimum 64 KB dikurangi 256 byte,
          wajib \dir{ORG 100h}, keluar dengan \code{INT 20h}, dan
          \reg{DS} sudah benar dengan sendirinya.

    \vspace{0.25em}
    \item \file{.EXE}: segmen kode, data, dan stack terpisah, keluar
          dengan \code{MOV AH, 4Ch} + \code{INT 21h}, dan \reg{DS}
          wajib diinisialisasi lebih dahulu.

    \vspace{0.25em}
    \item Empat pesan kesalahan assembler yang sering muncul:
          \file{Undefined symbol}, \file{Syntax error},
          \file{Operand types do not match}, dan
          \file{Improper operand type}.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Struktur Program Assembly Minimal}
\small Kerangka baku program assembly 8086 untuk lingkungan DOS:

\vspace{0.3em}
\begin{kodeasm}[numbers=none]
.MODEL SMALL        ; tentukan model memori
.STACK 100h         ; sediakan ruang stack

.DATA               ; ------- segmen data -------
    ; deklarasi variabel dan konstanta di sini

.CODE               ; ------- segmen kode -------
START:              ; label titik masuk program
    MOV AX, @DATA   ; inisialisasi register DS
    MOV DS, AX

    ; instruksi program di sini

    MOV AH, 4Ch     ; keluar dari program
    MOV AL, 00h
    INT 21h

END START           ; akhiri dan tunjuk titik masuk
\end{kodeasm}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Peran Setiap Bagian Kerangka}
\small
\begin{center}
\begin{tabular}{@{}p{3.0cm}p{10.3cm}@{}}
    \toprule
    \textbf{Direktif / label} & \textbf{Peran} \\
    \midrule
    \dir{.MODEL SMALL} & Model memori: satu segmen kode + satu segmen
    data, masing-masing maks.\ 64 KB \\
    \addlinespace
    \dir{.STACK 100h} & Menyediakan \hex{100} byte (\dec{256} byte)
    ruang stack \\
    \addlinespace
    \dir{.DATA} & Awal segmen data; variabel dideklarasikan dengan
    \dir{DB}, \dir{DW}, atau \dir{DD} \\
    \addlinespace
    \dir{.CODE} & Awal segmen kode; seluruh instruksi yang dieksekusi \\
    \addlinespace
    \file{START:} & Label instruksi pertama; harus sama dengan nama
    setelah \dir{END} \\
    \addlinespace
    \dir{END START} & Menutup berkas dan menunjuk titik masuk
    eksekusi \\
    \bottomrule
\end{tabular}
\end{center}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Inisialisasi Register \reg{DS}}
\small Program \file{.EXE} tidak otomatis memiliki \reg{DS} yang benar.
Dua instruksi berikut \textbf{wajib} ada di awal program:

\vspace{0.4em}
\begin{kodeasm}[numbers=none]
MOV AX, @DATA       ; salin alamat segmen data ke AX
MOV DS, AX          ; lalu pindahkan ke DS
\end{kodeasm}

\vspace{0.7em}
\begin{block}{Mengapa harus lewat \reg{AX}?}
\small Arsitektur 8086 melarang pemindahan nilai segera
(\eng{immediate}) langsung ke register segmen. Pemindahan ke register
segmen hanya boleh berasal dari register umum.
\end{block}

\vspace{0.4em}
{\footnotesize\color{obeGray}\code{@DATA} adalah simbol bawaan hasil
\dir{.MODEL}, berisi nomor segmen tempat \dir{.DATA} berada.\par}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Alur Eksekusi Program Pertama}
\small
\begin{enumerate}
    \item DOS memuat \file{.exe}, membuat \eng{Program Segment Prefix},
          dan menetapkan \reg{CS} serta \reg{IP} ke titik masuk.

    \vspace{0.35em}
    \item Instruksi pertama menginisialisasi \reg{DS} agar penunjukan
          variabel pada segmen data benar.

    \vspace{0.35em}
    \item Instruksi program dijalankan satu per satu sesuai urutan.

    \vspace{0.35em}
    \item Layanan DOS \code{INT 21h} fungsi \hex{09} mencetak string
          ke layar.

    \vspace{0.35em}
    \item Layanan DOS \code{INT 21h} fungsi \hex{4C} mengakhiri
          program dan mengembalikan kendali ke DOS.
\end{enumerate}

\vspace{0.4em}
\begin{block}{Pola yang akan dipakai berulang}
\small Siapkan argumen pada register $\rightarrow$ panggil layanan
\code{INT 21h} $\rightarrow$ akhiri program.
\end{block}
\end{frame}

\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item Tiga lingkungan kerja yang dapat dipasang berdampingan:
          Emu8086 (emulator dengan tampilan register, memori, dan
          \eng{flag}), GUI Turbo Assembler (TASM/TLINK bergrafis),
          dan DOSBox (lingkungan DOS asli dengan \file{DEBUG}).

    \vspace{0.25em}
    \item Kerangka minimal program: \dir{.MODEL SMALL},
          \dir{.STACK 100h}, \dir{.DATA}, \dir{.CODE}, label titik
          masuk \file{START:}, dan \dir{END START}.

    \vspace{0.25em}
    \item \dir{.MODEL} menentukan model memori, \dir{.STACK}
          menyediakan \dec{256} byte ruang stack, dan nama setelah
          \dir{END} harus sama dengan label titik masuk.

    \vspace{0.25em}
    \item Pada \file{.EXE}, inisialisasi \reg{DS} harus melalui
          \reg{AX} (\code{MOV AX, @DATA} lalu \code{MOV DS, AX})
          karena nilai segera tidak boleh langsung masuk ke register
          segmen.

    \vspace{0.25em}
    \item Alur eksekusi: DOS memuat \file{.exe} dan membuat PSP,
          \reg{DS} diinisialisasi, instruksi dijalankan berurutan,
          lalu \code{INT 21h} fungsi \hex{09} mencetak dan fungsi
          \hex{4C} mengakhiri program.
\end{itemize}
\end{frame}

% =========================================================
% CONTOH PROGRAM
% =========================================================
\begin{frame}[fragile]{Contoh 1: Mencetak Teks dengan Layanan DOS}
\begin{columns}[t]
\begin{column}{0.58\textwidth}
\begin{kodeasm}
.MODEL SMALL
.STACK 100h
.DATA
    pesan DB 'Halo, dunia!', 0Dh, 0Ah, '$'
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AH, 09h
    LEA DX, pesan
    INT 21h
    MOV AH, 4Ch
    MOV AL, 00h
    INT 21h
END START
\end{kodeasm}
\end{column}
\begin{column}{0.38\textwidth}
{\color{obeTeal}\bfseries Keluaran}\\[0.3em]
\begin{keluaran}
Halo, dunia!
\end{keluaran}

\vspace{0.5em}
{\footnotesize
\code{AH} = nomor fungsi layanan DOS.\\[0.25em]
\code{DX} = alamat string.\\[0.25em]
String diakhiri \code{'\$'}.\par}

\vspace{0.4em}
{\footnotesize\color{obeGray}\file{code/asm/hello.asm}\par}
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh 2: Program \file{.COM} dengan \dir{ORG 100h}}
\small Bentuk lebih ringkas: seluruh kode, data, dan stack dalam satu
segmen sehingga \reg{DS} tidak perlu diinisialisasi.

\vspace{0.3em}
\begin{kodeasm}
.MODEL TINY
.CODE
ORG 100h            ; program .COM dimulai pada offset 100h

START:
    MOV AH, 09h     ; cetak string
    MOV DX, OFFSET pesan
    INT 21h

    MOV AH, 4Ch     ; keluar
    MOV AL, 0
    INT 21h

pesan DB 'Selamat datang di assembly!', 0Dh, 0Ah, '$'
END START
\end{kodeasm}

\vspace{0.2em}
{\footnotesize\color{obeGray}Perhatikan: tidak ada \code{MOV AX, @DATA}
maupun \code{MOV DS, AX}.\par}
\end{frame}

\begin{frame}[fragile]{Contoh 3: Menampilkan Karakter Satu per Satu}
\small Layanan DOS fungsi \hex{02} mencetak satu karakter pada register
\reg{DL}.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.55\textwidth}
\begin{kodeasm}
.MODEL SMALL
.STACK 100h
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AH, 02h     ; cetak karakter pada DL
    MOV DL, 'A'
    INT 21h
    MOV DL, 'B'
    INT 21h
    MOV DL, 'C'
    INT 21h
    MOV AH, 4Ch
    MOV AL, 0
    INT 21h
END START
\end{kodeasm}
\end{column}
\begin{column}{0.41\textwidth}
{\color{obeTeal}\bfseries Keluaran}\\[0.3em]
\begin{keluaran}
ABC
\end{keluaran}

\vspace{0.6em}
{\footnotesize Ketiga contoh memakai pola yang sama: siapkan argumen
pada register, panggil \code{INT 21h}, lalu akhiri program.\par}
\end{column}
\end{columns}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item Bahasa pemrograman dibedakan menurut tingkat abstraksi:
          bahasa mesin, assembly, dan tingkat tinggi. Assembly berada
          tepat di atas bahasa mesin dan bersifat \eng{one-to-one}.

    \vspace{0.25em}
    \item Pemrograman tingkat rendah dipakai untuk kendali penuh
          perangkat keras, ukuran/kecepatan kode, serta analisis dan
          keamanan.

    \vspace{0.25em}
    \item Rantai proses: \file{.asm} $\rightarrow$ assembler
          $\rightarrow$ \file{.obj} $\rightarrow$ linker
          $\rightarrow$ \file{.exe} $\rightarrow$ \eng{loader}
          $\rightarrow$ memori.

    \vspace{0.25em}
    \item Struktur minimal program memuat \dir{.MODEL}, \dir{.STACK},
          \dir{.DATA}, \dir{.CODE}, label titik masuk, dan \dir{END}.

    \vspace{0.25em}
    \item \file{.EXE} wajib menginisialisasi \reg{DS}; \file{.COM}
          tidak.

    \vspace{0.25em}
    \item Pola dasar: siapkan argumen pada register, panggil
          \code{INT 21h}, akhiri dengan fungsi \hex{4C}.
\end{itemize}
\end{frame}

              \item satu assembly menghasilkan tepat satu instruksi mesin

              \item \file{.exe} \quad \item \file{.lst}
          \end{enumerate}

    \vspace{0.25em}
    \item Fungsi layanan DOS untuk mengakhiri program adalah \dots
          \begin{enumerate}\footnotesize
              \item \hex{09} \quad \item \hex{02} \quad
              \item \hex{4C} \quad \item \hex{01}
          \end{enumerate}
\end{enumerate}
