\begin{frame}{Capaian dan Tujuan Pembelajaran}
\begin{block}{Sub-CPMK 3.1}
\footnotesize Menulis program dengan instruksi ADD, ADC, INC, SUB,
SBB, DEC, MUL, dan DIV serta menjelaskan pengaruhnya terhadap
\eng{flag}.
\end{block}

\vspace{0.1em}
{\small\textbf{CPMK yang didukung:} \CPMK{03}\quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{03}\par}

\vspace{0.15em}
\begin{exampleblock}{Setelah pertemuan ini, mahasiswa mampu:}
\footnotesize
\begin{enumerate}
    \item menggunakan ADD, ADC, INC, SUB, SBB, dan DEC;
    \item menjelaskan pengaruh operasi terhadap CF, OF, ZF, SF,
          AF, dan PF;
    \item membedakan CF untuk bilangan tidak bertanda dan OF untuk
          bilangan bertanda;
    \item menyusun aritmatika lebih dari 16 bit dengan ADC dan SBB;
    \item menggunakan MUL dan DIV serta menghindari
          \eng{divide overflow}.
\end{enumerate}
\end{exampleblock}
\end{frame}

\PemisahBagian{Penjumlahan dan Pengurangan}

% ---------------------------------------------------------
\begin{frame}[fragile]{ADD, ADC, dan INC}
\small
\begin{itemize}
    \item \code{ADD tujuan, sumber}: menjumlahkan sumber ke tujuan.
    \item \code{ADC tujuan, sumber}: menjumlahkan sumber, tujuan,
          dan CF.
    \item \code{INC tujuan}: menambah tujuan dengan satu.
\end{itemize}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV AX, 25
ADD AX, 17          ; AX = 42
ADC AX, BX          ; AX = AX + BX + CF
INC AX              ; AX = AX + 1
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}ADC dipakai untuk membawa carry dari
word rendah ke word tinggi.\par}
\Sumber{Bab X, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{SUB, SBB, dan DEC}
\small
\begin{itemize}
    \item \code{SUB tujuan, sumber}: mengurangi tujuan dengan
          sumber.
    \item \code{SBB tujuan, sumber}: mengurangi tujuan dengan
          sumber dan CF (pinjaman).
    \item \code{DEC tujuan}: mengurangi tujuan dengan satu.
\end{itemize}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV AX, 100
SUB AX, 45          ; AX = 55
SBB AX, BX          ; AX = AX - BX - CF
DEC AX              ; AX = AX - 1
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}SBB diperlukan agar pinjaman dari word
rendah ikut dikurangkan pada word tinggi.\par}
\Sumber{Bab X, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh Penjumlahan 32 Bit}
\small
\begin{kodeasm}[numbers=none]
MOV AX, rendahA
ADD AX, rendahB     ; hasil rendah, CF mungkin 1
MOV hasilRendah, AX
MOV AX, tinggiA
ADC AX, tinggiB     ; CF ikut dijumlahkan
MOV hasilTinggi, AX
\end{kodeasm}

\vspace{0.2em}
\begin{itemize}
    \item \hex{0001FFFF} + \hex{00020001} = \hex{0004}:\hex{0000}.
    \item Bila tahap kedua memakai ADD, carry tahap pertama hilang.
    \item INC dan DEC tidak mengubah CF.
\end{itemize}
\Sumber{Bab X, Bagian 1}
\end{frame}

\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item ADD menjumlahkan; ADC ikut menambahkan CF; INC menambah
          satu.

    \vspace{0.15em}
    \item SUB mengurangi; SBB ikut mengurangkan CF; DEC mengurangi
          satu.

    \vspace{0.15em}
    \item Operasi aritmatika memperbarui CF, OF, ZF, SF, AF, dan PF.

    \vspace{0.15em}
    \item Aritmatika 32 bit memakai ADD pada word rendah dan ADC
          pada word tinggi.

    \vspace{0.15em}
    \item INC dan DEC tidak mengubah CF, sehingga tidak dipakai
          untuk meneruskan carry.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Contoh DIV 8 dan 16 Bit}
\small
\begin{kodeasm}[numbers=none]
; 8 bit: 1000 / 7 = 142 sisa 6
MOV AX, 1000
MOV BL, 7
DIV BL              ; AL = 142, AH = 6

; 16 bit: 100000 / 1000 = 100 sisa 0
MOV DX, 1
MOV AX, 86A0h       ; DX:AX = 100000
MOV BX, 1000
DIV BX              ; AX = 100, DX = 0
\end{kodeasm}

register yang sudah ditentukan prosesor.\par}

% ---------------------------------------------------------
\begin{frame}{Divide Overflow}
\small
\begin{itemize}
    \item Hasil bagi harus muat pada register tujuan.

    \vspace{0.18em}
    \item Pembagian 8 bit: AH harus lebih kecil daripada pembagi.

    \vspace{0.18em}
    \item Pembagian 16 bit: DX harus lebih kecil daripada pembagi.

    \vspace{0.18em}
    \item Pembagi tidak boleh nol.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{Akibat}
\small Bila syarat tidak dipenuhi, terjadi \eng{divide overflow}
dan program dapat berhenti mendadak.
\end{alertblock}
\Sumber{Bab X, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pemeriksaan Sebelum DIV}
\small
\begin{kodeasm}[numbers=none]
MOV AX, 1000
MOV BL, 7
CMP AH, BL          ; AH = 03h, BL = 07h
JAE luapan          ; lompat bila AH >= BL
DIV BL              ; aman: AL = 142, AH = 6
luapan:
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}AH = \hex{03} lebih kecil daripada
BL = \hex{07}, sehingga hasil bagi muat di AL.\par}
\Sumber{Bab X, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item MUL 8 bit: AL $\times$ operand, hasil di AX.

    \vspace{0.15em}
    \item MUL 16 bit: AX $\times$ operand, hasil di DX:AX.

    \vspace{0.15em}
    \item DIV 8 bit: AX $\div$ operand; bagi di AL, sisa di AH.

    \vspace{0.15em}
    \item DIV 16 bit: DX:AX $\div$ operand; bagi di AX, sisa di DX.

    \vspace{0.15em}
    \item Periksa AH/DX dan pembagi nol sebelum DIV.
\end{itemize}
\end{frame}

\PemisahBagian{Flag, Tanda Bilangan, dan Verifikasi Hasil}

% ---------------------------------------------------------
\begin{frame}{CF untuk Tak Bertanda, OF untuk Bertanda}
\small
\begin{columns}[t]
\begin{column}{0.47\textwidth}
    {\color{obeTeal}\bfseries CF --- tak bertanda}\\[0.25em]
    \begin{itemize}
        \item Hasil melewati batas 8/16 bit.
        \item \hex{FF} + \hex{01} = \hex{00}: CF = 1, OF = 0.
    \end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
    {\color{obeMaroon}\bfseries OF --- bertanda}\\[0.25em]
    \begin{itemize}
        \item Hasil keluar dari jangkauan bertanda.
        \item \hex{7F} + \hex{01} = \hex{80}: CF = 0, OF = 1.
    \end{itemize}
\end{column}
\end{columns}

\vspace{0.35em}
\begin{block}{Keduanya dapat terjadi sekaligus}
\small \hex{80} + \hex{80} = \hex{00} menghasilkan CF = 1 dan
OF = 1.
\end{block}
\Sumber{Bab X, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Flag Lain}
\small
\begin{itemize}
    \item ZF = 1 bila hasil nol; dipakai memeriksa kesamaan operand.

    \vspace{0.18em}
    \item SF = 1 bila bit tanda hasil menyala (hasil bertanda
          negatif).

    \vspace{0.18em}
    \item AF menandai simpanan dari empat bit rendah, dipakai pada
          operasi BCD.

    \vspace{0.18em}
    \item PF menandai paritas byte rendah hasil operasi.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{INC dan DEC}
\small Keduanya memperbarui ZF, SF, OF, AF, dan PF, tetapi tidak
mengubah CF.
\end{alertblock}
\Sumber{Bab X, Bagian 3}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item ADD menjumlahkan; ADC ikut menambahkan CF; INC menambah
          satu.

    \vspace{0.15em}
    \item SUB mengurangi; SBB ikut mengurangkan CF; DEC mengurangi
          satu.

    \vspace{0.15em}
    \item CF untuk luapan tak bertanda; OF untuk luapan bertanda.

    \vspace{0.15em}
    \item Aritmatika 32 bit memakai ADD lalu ADC, atau SUB lalu SBB.

    \vspace{0.15em}
    \item MUL 8 bit menghasilkan AX; MUL 16 bit menghasilkan DX:AX.

    \vspace{0.15em}
    \item DIV 8 bit menghasilkan AL/AH; DIV 16 bit menghasilkan
          AX/DX; periksa overflow lebih dahulu.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Perbedaan utama ADC dari ADD adalah \dots
          \begin{enumerate}
              \item ADC hanya bekerja pada 8 bit
              \item ADC ikut menambahkan CF
              \item ADC tidak mengubah flag
              \item ADC menyimpan hasil di memori
          \end{enumerate}

    \vspace{0.2em}
    \item Instruksi yang tidak mengubah CF adalah \dots
          \begin{enumerate}
              \item ADD \quad \item SUB \quad
              \item INC \quad \item ADC
          \end{enumerate}

    \vspace{0.2em}
    \item Operasi \hex{7F} + \hex{01} pada 8 bit menghasilkan \dots
          \begin{enumerate}
              \item CF = 1, OF = 0
              \item CF = 0, OF = 1
              \item CF = 1, OF = 1
              \item CF = 0, OF = 0
          \end{enumerate}
\end{enumerate}
