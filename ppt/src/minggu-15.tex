\begin{frame}{Empat Keterbatasan Disassembly}
\footnotesize
\begin{itemize}
    \item \textbf{Kode dan data sulit dibedakan.} Byte \hex{48} dapat
          berarti instruksi \asm{DEC AX} atau huruf \code{'H'};
          \eng{disassembler} bekerja dengan asumsi semua byte adalah
          instruksi.

    \vspace{0.2em}
    \item \textbf{Alamat relatif dan displacement.} \code{LOOP 0109}
          di dalam memori sesungguhnya berisi byte \hex{E2} \hex{F7},
          yaitu ``mundur 9 byte''; hasil hitungan ulang hanya benar
          bila instruksi berada di posisi yang tepat.

    \vspace{0.2em}
    \item \textbf{Nama simbolik hilang.} Variabel \code{larik} dan
          label \code{ULANG} tidak tersimpan di berkas biner; keduanya
          muncul sebagai \asm{[SI+01F4]}.

    \vspace{0.2em}
    \item \textbf{Sulit dibaca dan dipelihara.} Komentar hilang, makro
          sudah dijabarkan, dan tidak ada satu pun nama yang bermakna.
\end{itemize}

\vspace{0.1em}
{\footnotesize\important{Bongkar hanya daerah yang benar-benar berisi
kode} --- periksa batas segmen dengan mencatat \reg{CS} dan \reg{IP}
saat program dimulai.\par}
\Sumber{Bab XVI, Keterbatasan Disassembly}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item \eng{Disassembly} mengubah kode mesin kembali menjadi
          \eng{mnemonic}. Pekerjaannya searah dan tidak dapat
          membalikkan \eng{assembly} sepenuhnya, karena nama simbolik,
          komentar, dan batas data hilang saat perakitan.

    \vspace{0.22em}
    \item Perintah \code{U} pada DEBUG menampilkan alamat, byte mesin,
          dan \eng{mnemonic} sekaligus; byte mesin adalah acuan yang
          benar.

    \vspace{0.22em}
    \item Perkakas lain: \eng{disassembler} Emu8086, GUI Turbo
          Assembler, \code{ndisasm}, dan \code{objdump}.

    \vspace{0.22em}
    \item Keterbatasan utama: kode dan data sulit dibedakan,
          \eng{displacement} menggantikan nama variabel, dan semua
          penanda yang berguna bagi manusia hilang.
\end{itemize}
\end{frame}

\begin{frame}{Empat Metrik Optimasi}
\small
\begin{enumerate}
    \item \textbf{Jumlah instruksi.} Banyaknya instruksi yang
          \emph{dieksekusi}, bukan yang tertulis di kode. Metrik ini
          diperoleh dengan menghitung iterasi.

    \vspace{0.25em}
    \item \textbf{Perkiraan jumlah siklus.} Dihitung dari tabel siklus
          tiap instruksi, bukan diukur langsung pada mesin.

    \vspace{0.25em}
    \item \textbf{Ukuran kode dalam byte.} Panjang kode mesin yang
          dihasilkan; penting pada sistem tertanam dengan memori
          terbatas.

    \vspace{0.25em}
    \item \textbf{Jumlah akses memori.} Akses memori jauh lebih mahal
          daripada akses register, sehingga pengurangan satu akses
          saja sudah berarti.
\end{enumerate}

\vspace{0.2em}
\begin{block}{Ukuran keberhasilan ditetapkan lebih dahulu}
\small Sebelum memperbaiki sesuatu, metrik yang dipakai harus dipilih
agar perbandingan sebelum dan sesudah dapat dipercaya.
\end{block}
\Sumber{Bab XVI, Metrik Optimasi}
\end{frame}

\begin{frame}{Tanpa Pengukuran, Optimasi Hanyalah Dugaan}
\small
\begin{itemize}
    \item \textbf{Hambatan yang berbeda.} Perbaikan hanya berguna bila
          menyerang bagian yang paling banyak dieksekusi; mempercepat
          instruksi yang dijalankan seratus kali tidak berarti apa-apa
          bila ada perulangan yang berjalan sejuta kali.

    \vspace{0.25em}
    \item \textbf{Metrik yang bertentangan.} Suatu perubahan dapat
          mempercepat program tetapi memperbesar kode; bila target
          program adalah ukuran, perubahan itu justru merugikan.

    \vspace{0.25em}
    \item \textbf{Lingkungan yang berbeda.} Hasil yang cepat pada
          emulator belum tentu cepat pada perangkat keras
          sesungguhnya, dan sebaliknya.
\end{itemize}

\vspace{0.2em}
\begin{exampleblock}{Aktivitas}
\small Tuliskan dugaan Anda tentang bagian program yang paling lambat,
bongkar program itu dengan \code{U}, lalu bandingkan dugaan dengan
kenyataan pada jurnal praktikum.
\end{exampleblock}
\Sumber{Bab XVI, Tanpa Pengukuran, Optimasi Hanyalah Dugaan}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item Pola hasil \eng{disassembly} dapat dikenali: pembandingan dan
          percabangan, perulangan, \eng{prologue}/\eng{epilogue},
          inisialisasi segmen, dan pemanggilan interupsi.

    \vspace{0.22em}
    \item Ciri perulangan yang paling andal adalah lompatan mundur ke
          alamat yang lebih kecil daripada alamat instruksi lompatan.

    \vspace{0.22em}
    \item Empat metrik optimasi: jumlah instruksi yang dieksekusi,
          perkiraan siklus, ukuran kode dalam byte, dan jumlah akses
          memori.

    \vspace{0.22em}
    \item Jarak biaya antarinstrusi sangat lebar: \asm{MOV reg, reg} 2
          siklus, sedangkan \asm{DIV reg16} sekitar 150 siklus.

    \vspace{0.22em}
    \item Angka perkiraan hanya berlaku pada kondisi ideal; tanpa
          pengukuran pada mesin yang dipakai, optimasi hanyalah dugaan.
\end{itemize}
\end{frame}

\PemisahBagian{Strategi Optimasi pada Level Instruksi}

% ---------------------------------------------------------
\begin{frame}[fragile]{Memilih Register daripada Memori}
\small Teknik paling mendasar adalah menyimpan nilai yang sering dipakai
di dalam register, bukan di memori.

\vspace{0.15em}
\begin{kodeasm}[numbers=none]
    ; --- kurang baik: nilai diambil dari memori setiap iterasi ---
    MOV CX, 10
ULANG:
    ADD AX, jumlah      ; akses memori: 9 + 6 = 15 siklus
    LOOP ULANG

    ; --- lebih baik: nilai sudah ada di register ---
    MOV CX, 10
    MOV BX, jumlah      ; sekali saja: 8 + 6 = 14 siklus
ULANG2:
    ADD AX, BX          ; akses register: 3 siklus
    LOOP ULANG2
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Sepuluh iterasi versi pertama
$10 \times 15 = 150$ siklus; versi kedua
$14 + 10 \times 3 = 44$ siklus.\par}
\Sumber{Bab XVI, Memilih Register daripada Memori}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Strength Reduction: Penggantian Operasi}
\scriptsize
\begin{center}
\begin{tabular}{@{}p{3.5cm}p{3.5cm}p{5.9cm}@{}}
    \toprule
    \textbf{Operasi asal} & \textbf{Pengganti} & \textbf{Syarat
    berlaku} \\
    \midrule
    \asm{MUL BX}, \reg{BX} = 2 & \asm{ADD AX, AX} & Pengali tepat
    dua; hasil tidak memerlukan \reg{DX} \\
    \addlinespace
    \asm{MUL BX}, \reg{BX} = $2^n$ & \asm{SHL AX, n} & Pengali pangkat
    dua dan tak bertanda \\
    \addlinespace
    \asm{DIV BX}, \reg{BX} = $2^n$ & \asm{SHR AX, n} & Pembagi pangkat
    dua, nilai tak bertanda, sisa tidak diperlukan \\
    \addlinespace
    \asm{IDIV BX}, \reg{BX} = $2^n$ & \asm{SAR AX, n} & Nilai bertanda
    (komplemen dua); pembulatan ke bawah \\
    \addlinespace
    \asm{MOV AX, 0} & \asm{XOR AX, AX} & Hasil harus nol; \eng{flag}
    memang boleh berubah \\
    \addlinespace
    \asm{CMP AX, 0} & \asm{TEST AX, AX} & Uji dilakukan hanya terhadap
    nol \\
    \addlinespace
    \asm{ADD SI, 2} & \asm{LEA SI, [SI+2]} & \eng{Flag} harus
    dipertahankan agar pembandingan sebelumnya tetap sahih \\
    \addlinespace
    \asm{DEC CX} + \asm{JNZ} & \asm{LOOP} & Penghitung di \reg{CX} dan
    jarak lompatan kurang dari $\pm$128 byte \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XVI, Strength Reduction}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pergeseran sebagai Pengganti \asm{MUL} dan \asm{DIV}}
\small \asm{MUL BX} memerlukan 118--133 siklus, sedangkan
\asm{SHL AX, 1} hanya 2 siklus. Untuk mengalikan dengan delapan, tiga
kali pergeseran satu bit sudah cukup.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
    ; --- varian cepat: 3 instruksi, 6 byte, 6 siklus ---
    SHL AX, 1           ; AX = AX x 2
    SHL AX, 1           ; AX = AX x 4
    SHL AX, 1           ; AX = AX x 8

    ; --- varian kecil: 2 instruksi, 4 byte, 20 siklus ---
    MOV CL, 3           ; CL = banyak pergeseran
    SHL AX, CL          ; 8 + 4 x 3 = 20 siklus
\end{kodeasm}

\vspace{0.1em}
\begin{alertblock}{Syarat yang tidak boleh dilupakan}
\footnotesize Pergeseran hanya setara perkalian bila nilai yang digeser
tak bertanda dan bit yang keluar boleh hilang. \asm{MUL} memerhatikan
\flag{CF} dan \flag{OF} untuk mendeteksi luapan, \asm{SHL} tidak.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{\asm{LEA} untuk Aritmatika Tanpa Mengubah \eng{Flag}}
\small \asm{ADD} mengubah \flag{ZF} sehingga lompatan bersyarat menjadi
salah; \asm{LEA} menghitung alamat tanpa menyentuh \eng{flag}.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
    CMP AX, 0           ; ZF = 1 bila AX = 0
    ADD SI, 2           ; SALAH: ADD mengubah ZF
    JE  kosong          ; lompatan tidak lagi dapat dipercaya

    CMP AX, 0           ; ZF = 1 bila AX = 0
    LEA SI, [SI+2]      ; BENAR: LEA tidak mengubah flag
    JE  kosong          ; lompatan masih sahih
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\asm{LEA SI, [BX+SI+4]} menjumlahkan \reg{BX}, \reg{SI},
dan konstanta 4 sekaligus dalam satu instruksi. Namun \asm{LEA} bukan
pilihan yang lebih cepat: pada 8086 ia 10 siklus, sedangkan
\asm{ADD SI, 2} hanya 4 siklus --- keunggulannya keringkasan dan sifat
tidak mengubah \eng{flag}.\par}
\Sumber{Bab XVI, LEA untuk Aritmatika Tanpa Mengubah Flag}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Mengurangi Lompatan dan \eng{Loop Unrolling}}
\small \asm{LOOP} memerlukan 17 siklus dan menghalangi \eng{prefetch
queue} mengisi instruksi berikutnya. \eng{Unrolling} menggandakan isi
perulangan sehingga overhead \asm{LOOP} berkurang separuh.

\vspace{0.15em}
\begin{kodeasm}[numbers=none]
    ; --- perulangan biasa: 1 elemen per iterasi ---
    MOV CX, 10
ULANG:
    ADD AX, [SI]
    ADD SI, 2
    LOOP ULANG

    ; --- setelah unrolling 2x: 2 elemen per iterasi ---
    MOV CX, 5           ; cukup 5 iterasi
ULANG2:
    ADD AX, [SI]        ; elemen ke-1
    ADD AX, [SI+2]      ; elemen ke-2
    ADD SI, 4           ; maju dua word
    LOOP ULANG2
\end{kodeasm}

{\footnotesize\color{obeGray}Harganya jelas: kode menjadi lebih
panjang.\par}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Mengosongkan Register dengan \asm{XOR reg, reg}}
\small Cara ini benar dan mudah dibaca, tetapi bukan yang terpendek.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
    MOV AX, 0           ; 3 byte (B8 00 00), 4 siklus
    XOR AX, AX          ; 2 byte (31 C0),    3 siklus
    SUB AX, AX          ; 2 byte (29 C0),    3 siklus
\end{kodeasm}

\vspace{0.15em}
{\footnotesize \asm{XOR AX, AX} lebih pendek satu byte dan lebih cepat
satu siklus karena tidak perlu memuat nilai segera; ia juga
mengosongkan register 16 bit sekaligus.\par}

\vspace{0.15em}
\begin{alertblock}{Harganya adalah \eng{flag}}
\footnotesize \asm{XOR} mengubah \flag{ZF}, \flag{SF}, \flag{PF} serta
menghapus \flag{CF} dan \flag{OF}. Bila dipakai di antara sebuah
pembandingan dan lompatan bersyaratnya, program justru rusak ---
tempatkan pada awal blok, bukan di tengah alur keputusan.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pengukuran yang Adil}
\footnotesize
\begin{enumerate}
    \item \textbf{Masukan sama dan jumlah iterasi sama.} Kedua versi
          memakai larik yang sama dan mengerjakan pekerjaan yang sama
          banyaknya; penghitung iterasi dicatat untuk membuktikannya.
    \item \textbf{Batas pengukuran sama.} Waktu mulai dicatat tepat
          sebelum pekerjaan dan berhenti tepat sesudahnya, tanpa
          pemuatan program, pencetakan keluaran, atau penekanan tombol.
    \item \textbf{Diulang beberapa kali.} Jalankan minimal lima kali,
          buang percobaan pertama (pemanasan), lalu ambil nilai tengah.
\end{enumerate}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
    MOV AH, 00h         ; baca pencacah tick BIOS
    INT 1Ah             ; CX:DX = tick sejak tengah malam
    MOV tick_awal, DX   ; simpan titik awal

    ; ... pekerjaan yang diukur ...

    INT 1Ah             ; baca lagi
    SUB DX, tick_awal   ; selisih tick (18,2 tick = 1 detik)
\end{kodeasm}
\Sumber{Bab XVI, Pengukuran yang Adil}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Etika dan Batas Optimasi}
\small
\begin{itemize}
    \item \textbf{Benar lebih dahulu, cepat kemudian.} Program yang
          cepat tetapi salah tidak berguna; uji kebenaran setiap versi
          dengan masukan yang sama sebelum membandingkan kecepatannya.

    \vspace{0.25em}
    \item \textbf{Jangan mengoptimasi dini.} Optimasi hanya layak pada
          bagian yang sudah terbukti menjadi hambatan melalui
          pengukuran, bukan pada bagian yang hanya \emph{diperkirakan}
          lambat.

    \vspace{0.25em}
    \item \textbf{Jaga keterbacaan dan dokumentasi.} Setiap trik diberi
          komentar yang menjelaskan alasan dan syarat berlakunya, dan
          versi naif disimpan sebagai pembanding.
\end{itemize}

\vspace{0.25em}
\begin{block}{Optimasi yang dapat dipertanggungjawabkan}
\small Ada pengukuran sebelum, ada penjelasan mengapa perubahan dipilih,
dan ada pengukuran sesudah. Bila salah satunya tidak ada, yang
dikerjakan bukan optimasi melainkan sekadar perubahan.
\end{block}
\Sumber{Bab XVI, Etika dan Batas Optimasi}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item Teknik paling mendasar adalah menyimpan nilai yang sering
          dipakai di register, bukan di memori: 150 siklus menjadi 44
          siklus pada sepuluh iterasi.

    \vspace{0.22em}
    \item \eng{Strength reduction} menggantikan operasi mahal dengan
          operasi murah, tetapi hanya sahih bila syaratnya dipenuhi
          --- misalnya pengali harus pangkat dua dan nilai harus tak
          bertanda.

    \vspace{0.22em}
    \item \asm{LEA} menghitung alamat dalam satu instruksi tanpa
          mengubah \eng{flag}, sehingga pembandingan sebelumnya tetap
          sahih.

    \vspace{0.22em}
    \item \eng{Unrolling} mengurangi separuh overhead \asm{LOOP} dengan
          harga kode yang lebih panjang; \asm{XOR reg, reg} lebih
          pendek satu byte dan lebih cepat satu siklus daripada
          \asm{MOV reg, 0}.

    \vspace{0.22em}
    \item Optimasi ukuran dan optimasi kecepatan sering bertentangan;
          pilihan ditentukan batasan proyek dan dibuktikan dengan
          pengukuran yang adil.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Contoh 1: Versi Naif}
\scriptsize Ketiga contoh mengerjakan pekerjaan sama: menjumlahkan
sepuluh elemen larik $10, 20, \dots, 100$, lalu mengalikan jumlahnya
($550$) dengan delapan sehingga hasilnya \hex{1130}.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.57\textwidth}
\begin{kodeasm}[numbers=none]
    MOV CX, 10          ; 10 elemen
    MOV SI, 0           ; indeks byte
    MOV AX, 0           ; akumulator = 0
ULANG:
    ADD AX, larik[SI]   ; ambil dari memori
    ADD SI, 2           ; maju satu word
    LOOP ULANG
    MOV jumlah, AX      ; simpan ke memori
    MOV AX, jumlah      ; muat kembali lagi
    MOV BX, 8           ; pengali = 8
    MUL BX              ; 118-133 siklus
    MOV hasil, AX       ; simpan hasil akhir
\end{kodeasm}
\end{column}
\begin{column}{0.4\textwidth}
{\color{obeBlue}\bfseries Empat pemborosan}
\begin{enumerate}\footnotesize
    \item \asm{ADD AX, larik[SI]} memakai \eng{displacement} sehingga
          alamat efektifnya 8 siklus, bukan 5.
    \item \asm{MOV AX, 0} tiga byte, padahal ada cara dua byte.
    \item Hasil ditulis ke memori lalu dibaca kembali --- dua akses
          memori yang tidak perlu.
    \item \asm{MUL BX} untuk mengalikan delapan adalah pilihan paling
          mahal: 118--133 siklus.
\end{enumerate}
\vspace{0.2em}
{\footnotesize\color{obeGray}\file{naif.asm}\par}
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh 2: Versi Teroptimasi}
\scriptsize Empat perbaikan diterapkan tanpa mengubah hasil akhir:
\reg{hasil} tetap bernilai \hex{1130}.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.57\textwidth}
\begin{kodeasm}[numbers=none]
    MOV CX, 10          ; penghitung iterasi
    MOV SI, OFFSET larik ; SI ke larik
    XOR AX, AX          ; akumulator = 0
ULANG:
    ADD AX, [SI]        ; tanpa displacement
    ADD SI, 2           ; maju satu word
    LOOP ULANG
    SHL AX, 1           ; AX = AX x 2
    SHL AX, 1           ; AX = AX x 4
    SHL AX, 1           ; AX = AX x 8
    MOV hasil, AX       ; satu akses memori
\end{kodeasm}
\end{column}
\begin{column}{0.4\textwidth}
{\color{obeTeal}\bfseries Empat perbaikan}
\begin{itemize}\footnotesize
    \item \reg{SI} menunjuk langsung ke larik, sehingga
          \asm{ADD AX, [SI]} hanya 5 siklus alamat efektif.
    \item \asm{XOR AX, AX} mengosongkan akumulator dalam dua byte dan
          tiga siklus.
    \item Nilai antara tidak lagi bolak-balik ke memori; \reg{AX}
          dipakai sebagai akumulator sampai akhir.
    \item \asm{MUL BX} diganti tiga \asm{SHL AX, 1} --- enam siklus,
          sekitar dua puluh kali lebih cepat.
\end{itemize}
\vspace{0.2em}
{\footnotesize\color{obeGray}\file{teroptimasi.asm}\par}
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh 3: Teroptimasi dengan \eng{Loop Unrolling}}
\scriptsize Isi perulangan digandakan sehingga satu iterasi mengerjakan
dua elemen; jumlah iterasi turun dari sepuluh menjadi lima.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.57\textwidth}
\begin{kodeasm}[numbers=none]
    MOV CX, 5           ; cukup 5 iterasi
    MOV SI, OFFSET larik
    XOR AX, AX          ; akumulator = 0
ULANG:
    ADD AX, [SI]        ; elemen ke-1
    ADD AX, [SI+2]      ; elemen ke-2
    ADD SI, 4           ; maju dua word
    LOOP ULANG
    SHL AX, 1           ; AX = AX x 2
    SHL AX, 1           ; AX = AX x 4
    SHL AX, 1           ; AX = AX x 8
    MOV hasil, AX       ; satu akses memori
\end{kodeasm}
\end{column}
\begin{column}{0.4\textwidth}
{\color{obeAmber}\bfseries Yang diperoleh dan harganya}
\begin{itemize}\footnotesize
    \item Overhead \asm{LOOP} (17 siklus) berkurang separuh karena
          iterasi hanya lima kali.
    \item Dua elemen diambil per iterasi: \asm{ADD AX, [SI]} dan
          \asm{ADD AX, [SI+2]}.
    \item Alamat maju empat byte sekaligus dengan \asm{ADD SI, 4}.
    \item Harganya: ukuran kode bertambah tiga byte dibandingkan versi
          teroptimasi.
\end{itemize}
\vspace{0.2em}
{\footnotesize\color{obeGray}\file{teroptimasi-unroll.asm}\par}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Perbandingan Hasil Disassembly: Naif vs Teroptimasi}
\scriptsize Keduanya dibongkar dengan asumsi program dimuat pada alamat
\hex{1D2F}:0100, larik pada offset \hex{01F4}, \code{jumlah} pada
\hex{0208}, dan \code{hasil} pada \hex{020A}.

\vspace{0.15em}
\begin{columns}[t]
\begin{column}{0.48\textwidth}
{\footnotesize\color{obeMaroon}\bfseries Versi naif}\\[0.1em]
\begin{keluaran}

1D2F:0106 B80000   MOV  AX,0000
1D2F:0109 0384F401 ADD  AX,[SI+01F4]
1D2F:0110 E2F7     LOOP 0109
1D2F:0112 89060802 MOV  [0208],AX
1D2F:0116 8B060802 MOV  AX,[0208]
1D2F:011A BB0800   MOV  BX,0008
1D2F:011D F7E3     MUL  BX
\end{keluaran}
{\scriptsize\color{obeGray}11 instruksi, 35 byte\par}
\end{column}
\begin{column}{0.48\textwidth}
{\footnotesize\color{obeTeal}\bfseries Versi teroptimasi}\\[0.1em]
\begin{keluaran}

1D2F:0106 31C0     XOR  AX,AX
1D2F:0108 0304     ADD  AX,[SI]
1D2F:010D E2F9     LOOP 0108
1D2F:010F D1E0     SHL  AX,1
1D2F:0111 D1E0     SHL  AX,1
1D2F:0113 D1E0     SHL  AX,1
1D2F:0115 89060A02 MOV  [020A],AX

\asm{XOR AX, AX} (2 byte); \asm{MUL BX} menjadi tiga \asm{SHL AX, 1}.
Hasil \eng{disassembly} tidak memberi tahu \emph{mengapa} perubahan itu
dilakukan --- penjelasannya hanya ada pada perbandingan metrik.\par}

% LATIHAN
% =========================================================
\begin{frame}{Latihan}
\small
\begin{enumerate}
    \item Jelaskan mengapa \eng{disassembly} tidak dapat mengembalikan
          kode sumber asli. Sebutkan tiga informasi yang hilang selama
          perakitan dan mengapa informasi itu tidak tersimpan di dalam
          berkas \file{.exe}.

    \vspace{0.25em}
    \item Byte \hex{48} dapat berarti instruksi \asm{DEC AX} atau
          karakter \code{'H'}. Jelaskan bagaimana keadaan itu
          menyulitkan \eng{disassembly}, lalu sebutkan dua cara praktis
          menentukan batas daerah kode dan daerah data.

    \vspace{0.2em}
    \item Diberikan potongan hasil \eng{disassembly} berisi
          \asm{MOV CX,000A}, \asm{XOR BX,BX}, \asm{ADD AX,[SI]},
          \asm{ADD SI,+02}, dan \asm{LOOP 0205}. Pola apa yang tampak,
          berapa kali \asm{ADD AX, [SI]} dieksekusi, dan berapa jarak
          relatif yang disimpan pada byte \hex{E2} \hex{F9}?

    \vspace{0.2em}
    \item Perulangan 20 kali berisi \asm{ADD AX, [SI]},
          \asm{ADD SI, 2}, dan \asm{LOOP}. Hitung perkiraan jumlah
          siklusnya, lalu sebutkan dua alasan mengapa angka itu belum
          tentu sama dengan waktu sesungguhnya.

    \vspace{0.2em}
    \item Tuliskan hasil \eng{strength reduction} untuk \asm{MOV AX, 0},
          \asm{ADD AX, nilai}, \asm{MOV BX, 16}, \asm{MUL BX},
          \asm{CMP AX, 0}, dan \asm{JE kosong}, beserta syarat sahihnya.
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item \eng{Disassembly} mengubah kode mesin kembali menjadi
          \eng{mnemonic}; pekerjaan ini tidak dapat membalikkan
          \eng{assembly} sepenuhnya karena nama simbolik, komentar, dan
          batas data hilang saat perakitan.

    \vspace{0.18em}
    \item Perintah \code{U} pada DEBUG menampilkan alamat, byte mesin,
          dan \eng{mnemonic} sekaligus; perkakas lain: Emu8086, GUI
          Turbo Assembler, \code{ndisasm}, dan \code{objdump}.

    \vspace{0.18em}
    \item Pola hasil \eng{disassembly} dapat dikenali: pembandingan dan
          percabangan, perulangan, \eng{prologue} dan \eng{epilogue}
          prosedur, inisialisasi segmen, dan pemanggilan interupsi.

    \vspace{0.18em}
    \item Empat metrik optimasi: jumlah instruksi yang dieksekusi,
          perkiraan siklus, ukuran kode dalam byte, dan jumlah akses
          memori. Tanpa pengukuran, optimasi hanyalah dugaan.

    \vspace{0.18em}
    \item \eng{Strength reduction} hanya sahih bila syaratnya dipenuhi.
          Optimasi ukuran dan kecepatan sering bertentangan, sehingga
          pilihan ditentukan batasan proyek.

    \vspace{0.18em}
    \item Batas etika optimasi: kebenaran lebih dahulu, larangan
          optimasi dini, serta kewajiban menjaga keterbacaan.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Perintah MS-DEBUG untuk membongkar kode mesin menjadi
          \eng{mnemonic} adalah \dots
          \begin{enumerate}
              \item \code{T} \quad \item \code{U} \quad
              \item \code{D} \quad \item \code{G}
          \end{enumerate}

    \vspace{0.18em}
    \item Nama variabel \code{larik} berubah menjadi \asm{[SI+01F4]}
          pada hasil \eng{disassembly} karena \dots
          \begin{enumerate}
              \item nama simbolik tidak disimpan di berkas biner
              \item DEBUG selalu menghapus nama simbolik
              \item linker mengubahnya menjadi alamat relatif
              \item disassembler hanya membaca register
          \end{enumerate}

    \vspace{0.18em}
    \item Instruksi tunggal pengganti \asm{MUL BX} untuk mengalikan
          \reg{AX} (tak bertanda) dengan delapan adalah \dots
          \begin{enumerate}
              \item \asm{SHR AX, 3} \quad \item \asm{SHL AX, 3} \quad
              \item \asm{SAR AX, 3} \quad \item \asm{DIV AX, 8}
          \end{enumerate}
\end{enumerate}
\end{frame}
