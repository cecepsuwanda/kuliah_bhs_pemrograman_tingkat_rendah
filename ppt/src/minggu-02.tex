\begin{frame}{Mengapa 8086 Dipakai sebagai Model}
\small
\begin{itemize}
    \item 8086 memiliki bus data 16 bit dan bus alamat 20 bit
          sehingga menjangkau memori 1 MB.

    \vspace{0.18em}
    \item Susunan internalnya cukup sederhana untuk dipelajari,
          tetapi sudah memuat konsep utama sebuah prosesor.

    \vspace{0.18em}
    \item Register, segmentasi, bus, dan urutan pengambilan
          instruksi tetap menjadi fondasi prosesor sesudahnya.

    \vspace{0.18em}
    \item Varian 8088 memakai inti yang sama dengan bus luar 8 bit
          dan dipakai pada IBM PC awal.
\end{itemize}

\vspace{0.25em}

\small 8086 dipakai sebagai model pembelajaran, bukan sebagai

\begin{frame}{Kerja Bersamaan BIU dan EU}
\small
\begin{itemize}
    \item BIU dapat mengambil instruksi berikutnya sementara EU
          masih mengerjakan instruksi yang sekarang.

    \vspace{0.18em}
    \item EU tidak berhubungan langsung dengan bus luar.

    \vspace{0.18em}
    \item Bila instruksi memerlukan \eng{operand} memori, EU
          mengirim alamat ke BIU; BIU menjalankan siklus bus dan
          menyerahkan hasilnya.

    \vspace{0.18em}
    \item \eng{Queue} enam byte menjadi penyeimbang antara
          pengambilan dan eksekusi.
\end{itemize}

\vspace{0.25em}
\begin{block}{Akibat pembagian tugas}
\small Pengambilan instruksi dan eksekusi dapat berjalan tumpang
tindih, sehingga prosesor tidak selalu menunggu memori.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item 8086 memakai bus data 16 bit dan bus alamat 20 bit
          sehingga menjangkau memori 1 MB.

    \vspace{0.15em}
    \item 8088, 80286, dan 80386 memperluas bus atau ruang alamat,
          tetapi konsep dasar 8086 tetap dipakai.

    \vspace{0.15em}
    \item Arsitektur internal 8086 dibagi menjadi BIU dan EU.

    \vspace{0.15em}
    \item BIU mengurus hubungan dengan memori dan port; EU
          menerjemahkan dan menjalankan instruksi.

    \vspace{0.15em}
    \item Pembagian tugas memungkinkan pengambilan instruksi
          berjalan bersamaan dengan eksekusi.
\end{itemize}
\end{frame}

\begin{frame}{Perhitungan Alamat Fisik}
\small
Register segmen hanya 16 bit, sedangkan bus alamat 20 bit. BIU
menjembatani keduanya dengan rumus:

\[
\text{alamat fisik} = (\text{nilai segmen} \times 16)
+ \text{offset}
\]

Contoh: \reg{CS} = \hex{1234} dan \reg{IP} = \hex{0010}.

\[
\hex{1234} \times 16 + \hex{0010}
= \hex{12340} + \hex{0010} = \hex{12350}
\]

\begin{itemize}
    \item Perkalian dengan 16 sama dengan menggeser nilai segmen
          satu digit heksadesimal ke kiri.
    \item Perhitungan ini dikerjakan oleh BIU.
\end{itemize}
\Sumber{Bab IV, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item BIU mengambil instruksi, membaca dan menulis data, serta
          menangani pengalamatan dan transfer data.

    \vspace{0.15em}
    \item Komponen BIU: bus, \eng{queue} enam byte, register segmen
          CS/DS/SS/ES, dan \eng{Instruction Pointer}.

    \vspace{0.15em}
    \item \eng{Queue} memungkinkan pengambilan instruksi berikutnya
          berjalan sementara EU mengeksekusi.

    \vspace{0.15em}
    \item Tiga kelompok bus: alamat 20 bit, data 16 bit, dan kontrol.

    \vspace{0.15em}
    \item Alamat fisik = (segmen $\times$ 16) + offset; contoh
          \hex{1234}:\hex{0010} menghasilkan \hex{12350}.
\end{itemize}
\end{frame}

\begin{frame}{Tugas Execution Unit}
\small
\begin{enumerate}
    \item Menerjemahkan perintah yang diterima dari BIU.

    \vspace{0.18em}
    \item Mengeksekusi perintah tersebut.

\end{enumerate}

\vspace{0.3em}

\small EU tidak berhubungan langsung dengan bus luar. Bila
\eng{operand} berada di memori, EU meminta BIU membacanya.
\end{block}

\begin{frame}{ALU, Flag, dan Operand}
\small
\begin{itemize}
    \item \textbf{ALU} menjalankan operasi aritmetika dan logika:
          penjumlahan, pengurangan, perkalian, pembagian, AND, OR,
          dan NOT.

    \vspace{0.18em}
    \item \textbf{Register \eng{flag}} menandai hasil operasi,
          misalnya hasil nol, hasil negatif, atau \eng{carry}.

    \vspace{0.18em}
    \item \textbf{\eng{Operand}} adalah data yang dioperasikan;
          dapat berupa nilai langsung, register, atau isi memori.
\end{itemize}

\vspace{0.25em}
\begin{block}{Contoh}
\small Pada \code{ADD AL, 5}, \code{AL} dan \code{5} adalah
\eng{operand}, sedangkan \code{ADD} menentukan operasinya.
\end{block}
\Sumber{Bab IV, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Fetch, Decode, Execute}
\small
\begin{enumerate}
    \item \textbf{Fetch.} BIU menghitung alamat fisik dari
          \reg{CS}:\reg{IP}, mengambil byte instruksi melalui bus,
          memasukkannya ke \eng{queue}, lalu menambah \reg{IP}
          sesuai panjang instruksi.

    \vspace{0.15em}
    \item \textbf{Decode.} EU mengambil byte dari \eng{queue} dan
          menerjemahkannya: operasi apa yang dilakukan dan
          \eng{operand} apa yang terlibat.

    \vspace{0.15em}
    \item \textbf{Execute.} EU menjalankan operasi melalui ALU atau
          memindahkan data. Bila \eng{operand} di memori, EU
          meminta BIU membacanya.
\end{enumerate}

\vspace{0.15em}
{\footnotesize\color{obeGray}Urutan ini tidak berarti BIU dan EU
selalu bergantian; BIU dapat mengisi \eng{queue} sementara EU
mengeksekusi.\par}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Hasil ke Register vs Hasil ke Memori}
\small
\begin{columns}[t]
\begin{column}{0.47\textwidth}
    {\color{obeTeal}\bfseries Hasil ke register}\\[0.25em]
    \begin{itemize}
        \item Ditulis di dalam chip.
        \item Tidak memerlukan bus luar.
        \item Contoh: \code{MOV AX, 1234h}.
    \end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
    {\color{obeMaroon}\bfseries Hasil ke memori}\\[0.25em]
    \begin{itemize}
        \item Memerlukan siklus bus tulis.
        \item Dikerjakan melalui BIU.
        \item Waktu eksekusi lebih lama.
    \end{itemize}
\end{column}
\end{columns}

\vspace{0.5em}
\begin{block}{Sumber perbedaan waktu}
\small Perbedaan hasil akhir tidak selalu terlihat, tetapi jumlah
langkah bus yang dikerjakan BIU berbeda.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item EU menerjemahkan dan menjalankan instruksi serta memberi
          tahu BIU alamat \eng{operand} yang diperlukan.

    \vspace{0.15em}
    \item Komponen EU: ALU, register \eng{flag}, \eng{operand},
          decoder, register umum, serta register penunjuk dan indeks.

    \vspace{0.15em}
    \item ALU menjalankan operasi aritmetika dan logika; hasilnya
          ditandai melalui \eng{flag}.

    \vspace{0.15em}
    \item Urutan pemrosesan: \eng{fetch} oleh BIU, lalu
          \eng{decode} dan \eng{execute} oleh EU.

    \vspace{0.15em}
    \item Hasil ke register tidak memerlukan bus luar; hasil ke
          memori memerlukan siklus bus melalui BIU.
\end{itemize}
\end{frame}

ke EU.

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item 8086 memakai bus data 16 bit dan bus alamat 20 bit;
          ruang alamatnya 1 MB.

    \vspace{0.15em}
    \item Arsitektur internal dibagi menjadi BIU dan EU agar
          pengambilan instruksi dan eksekusi dapat berjalan
          bersamaan.

    \vspace{0.15em}
    \item BIU mengambil instruksi, melayani data, menangani
          pengalamatan, dan mengisi \eng{queue} enam byte.

    \vspace{0.15em}
    \item Bus alamat 20 bit, bus data 16 bit, dan bus kontrol
          menghubungkan prosesor dengan memori serta port.

    \vspace{0.15em}
    \item Alamat fisik = (segmen $\times$ 16) + offset.

    \vspace{0.15em}
    \item Urutan instruksi: \eng{fetch} oleh BIU, \eng{decode} dan
          \eng{execute} oleh EU; hasil ke memori memerlukan siklus
          bus tambahan.
\end{itemize}
\end{frame}
