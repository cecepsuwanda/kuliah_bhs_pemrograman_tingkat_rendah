% ---------------------------------------------------------
\begin{frame}{AND, OR, XOR, dan NOT}
\small
\begin{itemize}
    \item \textbf{AND}: 1 hanya bila kedua bit 1. Dipakai untuk
          mempertahankan bit tertentu dan memadamkan bit lain.

    \vspace{0.18em}
    \item \textbf{OR}: 1 bila salah satu atau kedua bit 1. Dipakai
          untuk menyalakan bit tertentu.

    \vspace{0.18em}
    \item \textbf{XOR}: 1 bila kedua bit berbeda. Dipakai untuk
          membalik bit tertentu.

    \vspace{0.18em}
    \item \textbf{NOT}: membalik seluruh bit operand; tidak
          memengaruhi \eng{flag}.
\end{itemize}

\vspace{0.2em}
\begin{block}{Bitwise}
\small Operasi bekerja pada 8 bit atau 16 bit, bergantung pada
ukuran operand.
\end{block}
\Sumber{Bab XI, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Pengaruh terhadap Flag}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.6cm}
    >{\raggedright\arraybackslash}p{5.0cm}
    >{\raggedright\arraybackslash}p{6.2cm}@{}}
    \toprule
    \textbf{Flag} & \textbf{Nilai setelah operasi} &
    \textbf{Alasan} \\
    \midrule
    CF & Selalu 0 & Tidak ada carry dari bit paling kiri \\
    \addlinespace
    OF & Selalu 0 & Tidak ada luapan bertanda \\
    \addlinespace
    ZF & 1 bila seluruh bit hasil 0 & Memeriksa bidang bit kosong \\
    \addlinespace
    SF & Menyalin bit paling kiri hasil & Hasil dianggap negatif
    bila bertanda \\
    \addlinespace
    PF & 1 bila jumlah bit 1 pada byte rendah genap & Pemeriksaan
    paritas \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XI, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Pengecualian NOT}
\small
\code{NOT} membalik seluruh bit, tetapi tidak menyentuh
\eng{flag}. Bila hasilnya perlu diuji, tambahkan \code{TEST AL, AL}
sesudahnya.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV AL, 0A5h
NOT AL              ; AL = 5Ah
TEST AL, AL         ; perbarui ZF, SF, PF
\end{kodeasm}

\vspace{0.15em}
\begin{alertblock}{Jangan mengandalkan CF}
\small Setelah AND, OR, atau XOR, CF selalu 0; nilai CF dari operasi
sebelumnya tidak dapat dipakai.
\end{alertblock}
\Sumber{Bab XI, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 1}
\small
\begin{itemize}
    \item Operasi logika bekerja per bit tanpa carry antarbit.

    \vspace{0.15em}
    \item AND: kedua bit 1; OR: salah satu 1; XOR: bit berbeda;
          NOT: membalik semua bit.

    \vspace{0.15em}
    \item AND, OR, dan XOR mengosongkan CF dan OF.

    \vspace{0.15em}
    \item ZF, SF, dan PF mengikuti hasil operasi.

    \vspace{0.15em}
    \item NOT tidak mengubah \eng{flag}.
\end{itemize}
\end{frame}

\PemisahBagian{Mask dan Instruksi TEST}

% ---------------------------------------------------------
\begin{frame}{Mask: Menyentuh Satu Bit}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.6cm}
    >{\raggedright\arraybackslash}p{4.0cm}
    >{\raggedright\arraybackslash}p{2.4cm}
    >{\raggedright\arraybackslash}p{3.4cm}@{}}
    \toprule
    \textbf{Kehendak} & \textbf{Instruksi} & \textbf{Mask} &
    \textbf{Hasil pada A5h, bit 3} \\
    \midrule
    Menyalakan bit & \code{OR AL, mask} & \hex{08} &
    \hex{AD} \\
    \addlinespace
    Memadamkan bit & \code{AND AL, NOT mask} & \hex{F7} &
    \hex{A5} \\
    \addlinespace
    Membalik bit & \code{XOR AL, mask} & \hex{08} & \hex{AD} \\
    \addlinespace
    Menguji bit & \code{TEST AL, mask} & \hex{08} &
    ZF = 0 bila bit menyala \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XI, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh Mask}
\small
\begin{kodeasm}[numbers=none]
OR  AL, 08h         ; nyalakan bit 3
AND AL, 0F7h        ; padamkan bit 3 (F7h = NOT 08h)
XOR AL, 08h         ; balikkan bit 3
TEST AL, 08h        ; uji bit 3 tanpa mengubah AL
JNZ BIT_MENYALA     ; lompat bila bit 3 = 1
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}Mask penghapus diperoleh dari
\code{NOT 08h}; bila dihitung saat program berjalan, simpan mask ke
register lalu balik dengan NOT.\par}
\Sumber{Bab XI, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{TEST vs CMP}
\footnotesize
\begin{center}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.2cm}
    >{\raggedright\arraybackslash}p{4.6cm}
    >{\raggedright\arraybackslash}p{5.8cm}@{}}
    \toprule
    \textbf{Aspek} & \textbf{TEST} & \textbf{CMP} \\
    \midrule
    Operasi internal & AND tanpa menyimpan hasil &
    SUB tanpa menyimpan hasil \\
    \addlinespace
    CF dan OF & Selalu 0 & Sesuai hasil pengurangan \\
    \addlinespace
    Pemakaian & Memeriksa bit tertentu &
    Membandingkan nilai utuh \\
    \addlinespace
    Lompatan lazim & JZ/JNZ, JP/JNP & JE/JNE, JA/JB \\
    \addlinespace
    Contoh & \code{TEST AL, 01h} lalu JNZ &
    \code{CMP AL, BL} lalu JA \\
    \bottomrule
\end{tabular}
\end{center}
\Sumber{Bab XI, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{TEST AL, AL}
\small
\code{TEST AL, AL} melakukan AND antara AL dengan dirinya sendiri.
Isi AL tidak berubah, tetapi ZF, SF, dan PF diperbarui.

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
TEST AL, 80h        ; periksa bit tanda
JNZ NEGATIF

TEST BL, 04h        ; periksa bit ke-2
JZ  BIT_NOL

TEST AL, AL         ; periksa apakah AL = 0
JZ  SELESAI
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Gunakan TEST untuk bit tertentu dan CMP
untuk nilai keseluruhan.\par}
\Sumber{Bab XI, Bagian 2}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item Mask adalah pola bit yang dirancang untuk menyentuh posisi
          tertentu.

    \vspace{0.15em}
    \item OR menyalakan bit; AND dengan mask terbalik memadamkan
          bit; XOR membalik bit.

    \vspace{0.15em}
    \item TEST menguji bit tanpa mengubah operand.

    \vspace{0.15em}
    \item TEST memakai AND; CMP memakai SUB.

    \vspace{0.15em}
    \item \code{TEST AL, AL} dipakai untuk memeriksa apakah AL nol.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Pengaruh Pergeseran terhadap Flag}
\small
\begin{itemize}
    \item CF menerima bit terakhir yang keluar dari operand.

    \vspace{0.18em}
    \item ZF, SF, dan PF mengikuti hasil pergeseran.

    \vspace{0.18em}
    \item OF hanya bermakna untuk pergeseran satu bit.

    \vspace{0.18em}
    \item Untuk pergeseran lebih dari satu bit, nilai OF tidak dapat
          diandalkan.
\end{itemize}

\vspace{0.25em}
\begin{block}{Contoh}
\small SHL pada \hex{A5} satu bit menghasilkan \hex{4A} dengan
CF = 1.
\end{block}
\Sumber{Bab XI, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Perkalian dan Pembagian dengan Shift}
\small
\begin{kodeasm}[numbers=none]
; 5 x 16 = 80
MOV AX, 5
MOV CL, 4           ; 16 = 2^4
SHL AX, CL          ; AX = 80

; 1000 / 8 = 125
MOV AX, 1000
MOV CL, 3           ; 8 = 2^3
SHR AX, CL          ; AX = 125
\end{kodeasm}

\vspace{0.15em}
{\footnotesize\color{obeGray}SHL setara dengan perkalian pangkat dua;
SHR setara dengan pembagian pangkat dua untuk bilangan tidak
bertanda.\par}
\Sumber{Bab XI, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{SHR untuk Bilangan Bertanda}
\small
\begin{itemize}
    \item SHL dan SHR tepat untuk bilangan tidak bertanda.

    \vspace{0.18em}
    \item SHR mengisi bit paling kiri dengan 0, termasuk bit tanda.

    \vspace{0.18em}
    \item \hex{FFFF} ($-1$) digeser satu bit menjadi \hex{7FFF}
          ($+32767$).

    \vspace{0.18em}
    \item Pembagian bilangan bertanda memerlukan penanganan tanda
          tersendiri.
\end{itemize}

\vspace{0.25em}
\begin{alertblock}{Jangan memakai SHR untuk negatif}
\small Hasilnya tidak lagi mewakili pembagian yang benar.
\end{alertblock}
\Sumber{Bab XI, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item SHL menggeser bit ke kiri; bit kanan diisi 0.

    \vspace{0.15em}
    \item SHR menggeser bit ke kanan; bit kiri diisi 0.

    \vspace{0.15em}
    \item Bit yang keluar ditampung CF.

    \vspace{0.15em}
    \item Pergeseran kiri setara perkalian pangkat dua; pergeseran
          kanan setara pembagian untuk bilangan tidak bertanda.

    \vspace{0.15em}
    \item SHR tidak tepat untuk bilangan bertanda negatif.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Contoh 1: Memeriksa Paritas}
\begin{columns}[t]
\begin{column}{0.53\textwidth}
\begin{kodeasm}
.DATA
    nilai   DB 10110110b
    pesan1  DB 'Byte 10110110b berparitas: $'
    genap   DB 'GENAP', 0Dh, 0Ah, '$'
    ganjil  DB 'GANJIL', 0Dh, 0Ah, '$'
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AL, nilai
    MOV BL, 0
    MOV CX, 8
ULANG:
    SHR AL, 1
    ADC BL, 0
    LOOP ULANG
\end{kodeasm}
\end{column}
\begin{column}{0.44\textwidth}
\footnotesize
{\color{obeTeal}\bfseries Keluaran}\\[0.15em]
\begin{keluaran}
Byte 10110110b berparitas: GANJIL
\end{keluaran}

\vspace{0.3em}
\begin{itemize}
    \item SHR memindahkan bit terkanan ke CF.
    \item ADC BL, 0 menambahkan CF ke penghitung.
    \item TEST BL, 1 memeriksa bit terendah jumlah bit.
    \item Lima bit 1 berarti paritas ganjil.
\end{itemize}
\end{column}
\end{columns}
\Sumber{Bab XI, Contoh 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Contoh 2: XOR Cipher}
\begin{columns}[t]
\begin{column}{0.55\textwidth}
\begin{kodeasm}
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    LEA SI, asli
    LEA DI, buffer
ENKRIPSI:
    MOV AL, [SI]
    CMP AL, '$'
    JE TAMPIL
    XOR AL, kunci
    MOV [DI], AL
    INC SI
    INC DI
    JMP ENKRIPSI
\end{kodeasm}
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
{\color{obeTeal}\bfseries Keluaran}\\[0.15em]
\begin{keluaran}
Teks asli    : BELAJARASM
Terenkripsi  : belajarasm
Terdekripsi  : BELAJARASM
\end{keluaran}

\vspace{0.2em}
{\footnotesize\color{obeGray}\code{asli}, \code{buffer}, dan
\code{kunci} didefinisikan di \dir{.DATA}.\par}

\vspace{0.2em}
Sifat $A \oplus K \oplus K = A$ membuat XOR yang sama dipakai untuk
enkripsi dan dekripsi. Kunci \hex{20} membalik bit ke-5.
\end{column}
\end{columns}

\begin{frame}{Latihan}
\small
\begin{enumerate}
    \item Jelaskan hasil AND, OR, XOR, dan NOT untuk pasangan bit
          (1,0), (1,1), dan (0,0).

    \vspace{0.15em}
    \item Jelaskan pengaruh AND, OR, dan XOR terhadap CF, OF, ZF,
          dan SF. Mengapa NOT berbeda?

    \vspace{0.15em}
    \item Diketahui AL = \hex{A5}. Tuliskan instruksi untuk
          menyalakan, memadamkan, membalik, dan menguji bit 4.

    \vspace{0.15em}
    \item Jelaskan perbedaan TEST dan CMP. Apa fungsi
          \code{TEST AL, AL}?

    \vspace{0.15em}
    \item Diketahui AL = \bin{10100101}. Tentukan hasil dan CF
          untuk SHL 1, SHR 1, dan SHL 2.
\end{enumerate}
\end{frame}

% =========================================================
% RANGKUMAN
% =========================================================
\begin{frame}{Rangkuman}
\small
\begin{itemize}
    \item Operasi logika bekerja per bit tanpa carry antarbit.

    \vspace{0.15em}
    \item AND, OR, XOR, dan NOT memiliki tabel kebenaran berbeda.

    \vspace{0.15em}
    \item AND/OR/XOR mengosongkan CF dan OF; NOT tidak mengubah
          flag.

    \vspace{0.15em}
    \item Mask dipakai untuk menyalakan, memadamkan, membalik, dan
          menguji bit.

    \vspace{0.15em}
    \item TEST memakai AND; CMP memakai SUB.

    \vspace{0.15em}
    \item SHL/SHR menggeser bit; CF menampung bit yang keluar.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Cek Pemahaman}
\footnotesize
\begin{enumerate}
    \item Hasil AND antara bit 1 dan bit 0 adalah \dots
          \begin{enumerate}
              \item 0 \quad \item 1 \quad
              \item bergantung pada CF \quad
              \item tidak terdefinisi
          \end{enumerate}

    \vspace{0.2em}
    \item Instruksi yang dipakai untuk membalik bit tertentu
          adalah \dots
          \begin{enumerate}
              \item AND \quad \item OR \quad
              \item XOR \quad \item NOT
          \end{enumerate}

    \vspace{0.2em}
    \item Diketahui AL = \bin{10100101}. Setelah SHL AL, 1, isi AL
          dan CF adalah \dots
          \begin{enumerate}
              \item \bin{01001010}, CF = 1
              \item \bin{01001010}, CF = 0
              \item \bin{11010010}, CF = 1
              \item \bin{01010010}, CF = 1
          \end{enumerate}
\end{enumerate}
\end{frame}
