    \item menjelaskan mekanisme \code{INT} dan \code{IRET};
    \item menggunakan \code{INT 10h} dan \code{INT 16h};
    \item menggunakan \code{INT 21h} serta membedakan

\begin{frame}{Tabel Vektor Interupsi}
\small
\begin{itemize}
    \item Terletak di awal memori, alamat \hex{00000}--\hex{003FF};
          ukurannya 1 KB.

    \vspace{0.18em}
    \item Terdiri atas 256 entri; setiap entri 4 byte.

    \vspace{0.18em}
    \item Dua byte pertama: \eng{offset} handler; dua byte berikutnya:
          nomor segmen handler.

    \vspace{0.18em}
    \item Alamat entri interupsi $n$ adalah $n \times 4$.
\end{itemize}

\vspace{0.25em}
\begin{block}{Contoh}
\small \code{INT 21h} berada pada \hex{21} $\times 4 = \hex{84}$,
yaitu \hex{0000}:\hex{0084}.
\end{block}
\Sumber{Bab IX, Bagian 1}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Mekanisme INT dan IRET}
\small
\begin{enumerate}
    \item Menyimpan \reg{FLAGS}, \reg{CS}, dan \reg{IP} ke
          \eng{stack}.

    \vspace{0.15em}
    \item Membaca entri vektor ke-$n$ dari tabel vektor.

    \vspace{0.15em}
    \item Memuat \reg{IP} dan \reg{CS} dengan alamat handler.

    \vspace{0.15em}
    \item Menjalankan handler.

    \vspace{0.15em}
    \item Handler mengakhiri dengan \code{IRET}, yang memulihkan
          \reg{IP}, \reg{CS}, dan \reg{FLAGS}.
\end{enumerate}

\vspace{0.2em}
\begin{alertblock}{IRET bukan RET}
\small \code{RET} hanya mengambil alamat kembali; \code{IRET} juga
memulihkan \reg{FLAGS}.
\end{alertblock}
\Sumber{Bab IX, Bagian 1}
\end{frame}

\begin{frame}{Blocking dan Non-Blocking}
\small
\begin{itemize}
    \item Fungsi \hex{00} bersifat \eng{blocking}: program berhenti
          sampai pengguna menekan tombol.

    \vspace{0.18em}
    \item Fungsi \hex{01} bersifat \eng{non-blocking}: program dapat
          terus berjalan sambil memeriksa tombol.

    \vspace{0.18em}
    \item \code{INT 16h} membaca tombol pada level BIOS;
          \code{INT 21h} fungsi \hex{01}/\hex{0A} membaca melalui
          DOS.

    \vspace{0.18em}
    \item Keduanya tidak dicampur tanpa alasan karena memengaruhi
          penyangga \eng{keyboard}.
\end{itemize}

\vspace{0.2em}
\begin{block}{Pilihan}
\small Gunakan \hex{00} bila program harus menunggu; gunakan
\hex{01} bila program tetap berjalan.
\end{block}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Rangkuman Bagian 2}
\small
\begin{itemize}
    \item BIOS menyediakan layanan video dan keyboard sejak komputer
          menyala.

    \vspace{0.15em}
    \item \code{INT 10h} memakai AH untuk memilih fungsi video.

    \vspace{0.15em}
    \item Fungsi \hex{00}, \hex{02}, \hex{09}, dan \hex{13} dipakai
          untuk mode, kursor, karakter, dan string.

    \vspace{0.15em}
    \item \code{INT 16h} menyediakan fungsi baca tombol, cek tombol,
          dan status shift.

    \vspace{0.15em}
    \item Fungsi \hex{00} menunggu; fungsi \hex{01} tidak menunggu.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Membaca Masukan}
\small
\begin{itemize}
    \item Fungsi \hex{01}: membaca satu karakter dengan
          \eng{echo}; hasil di AL.
    \item Fungsi \hex{0A}: membaca satu baris ke penyangga.
\end{itemize}

\vspace{0.15em}
{\footnotesize Susunan penyangga fungsi \hex{0A}: byte 0 kapasitas;
byte 1 jumlah terisi (diisi DOS); byte 2 dan seterusnya isi
masukan.\par}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
nama DB 20          ; kapasitas
     DB 0           ; jumlah terisi
     DB 20 DUP(0)   ; isi masukan
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Setelah masukan, tambahkan
\code{'\$'} setelah karakter terakhir agar dapat dicetak dengan
fungsi \hex{09}.\par}
\Sumber{Bab IX, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}[fragile]{Mengakhiri Program}
\small
\begin{itemize}
    \item \code{INT 20h} mengakhiri program \file{.COM} dan
          mengembalikan kendali ke DOS.
    \item \code{INT 21h} fungsi \hex{4C} mengakhiri \file{.EXE}
          atau \file{.COM} dengan kode keluar di AL.
\end{itemize}

\vspace{0.2em}
\begin{kodeasm}[numbers=none]
MOV AH, 4Ch
MOV AL, 00h
INT 21h
\end{kodeasm}

\vspace{0.1em}
{\footnotesize\color{obeGray}Fungsi \hex{4C} lebih umum karena
dapat mengembalikan kode keluar.\par}
\Sumber{Bab IX, Bagian 3}
\end{frame}

% ---------------------------------------------------------
\begin{frame}{Kesalahan Umum Layanan DOS}
\small
\begin{itemize}
    \item String untuk fungsi \hex{09} tidak diakhiri
          \code{'\$'}, sehingga pencetakan tidak berhenti.

    \vspace{0.18em}
    \item Penyangga fungsi \hex{0A} tidak menyediakan byte
          kapasitas dan byte jumlah.

    \vspace{0.18em}
    \item Program tidak diakhiri, sehingga kendali tidak kembali
          ke DOS.

    \vspace{0.18em}
    \item Mencampur \code{INT 16h} dan \code{INT 21h} untuk
          masukan tanpa alasan yang jelas.
\end{itemize}

\vspace{0.2em}
\begin{alertblock}{Periksa sebelum menjalankan}
\small Pastikan string, penyangga, dan instruksi keluar sudah
lengkap.
\end{alertblock}
\end{frame}

\begin{frame}{Rangkuman Bagian 3}
\small
\begin{itemize}
    \item \code{INT 21h} memakai AH untuk memilih fungsi DOS.

    \vspace{0.15em}
    \item Fungsi \hex{01}, \hex{02}, \hex{09}, \hex{0A}, dan

\end{itemize}

    \item \code{INT 10h} untuk video, \code{INT 16h} untuk
          keyboard, \code{INT 21h} untuk layanan DOS.

    \vspace{0.15em}
    \item \code{INT 20h} khusus \file{.COM}; fungsi \hex{4C}
          mengembalikan kode keluar.
\end{itemize}
\end{frame}
