    \def\BookRoot{../..}%

}

% ---------------------------------------------------------
% CATATAN ARSITEKTUR
%
% Bab dimuat dengan \subfile, bukan \include.
%
% Alasannya: package subfiles menetapkan direktori berkas yang
% sedang dibaca (melalui \import@path). Dengan \include, direktori
% itu tidak ditetapkan, sehingga \subfile{section-01} di dalam
% chapter.tex akan dicari relatif terhadap direktori main.tex dan
% gagal ditemukan. Dengan \subfile, seluruh rantai
% chapter -> section -> input kecil dapat diselesaikan.
%
% Konsekuensi: fitur \includeonly tidak tersedia. Bila diperlukan,
% kompilasi per bab tetap dapat dilakukan langsung dari direktori
% bab (lihat scripts/build-chapter.bat).
% ---------------------------------------------------------
