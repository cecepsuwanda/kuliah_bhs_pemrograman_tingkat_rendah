% =========================================================
% config/styles.tex
% Konfigurasi tampilan buku: warna, format judul, header/footer,
% spasi, dan keterangan gambar/tabel.
% =========================================================

% ---------------------------------------------------------
% 1. Palet warna buku
% ---------------------------------------------------------
\definecolor{obeBlue}{HTML}{1F3864}
\definecolor{obeBlueLight}{HTML}{DCE4F5}
\definecolor{obeTeal}{HTML}{0F6E6E}
\definecolor{obeTealLight}{HTML}{D9F0EE}
\definecolor{obeGreen}{HTML}{2E6B34}
\definecolor{obeGreenLight}{HTML}{DEEFDF}
\definecolor{obeAmber}{HTML}{8A5A00}
\definecolor{obeAmberLight}{HTML}{FDF1D6}
\definecolor{obeMaroon}{HTML}{7A2233}
\definecolor{obeMaroonLight}{HTML}{F7E1E5}
\definecolor{obeGray}{HTML}{4A4A4A}
\definecolor{obeGrayLight}{HTML}{F0F0F0}
\definecolor{codeBack}{HTML}{F7F7F4}
\definecolor{codeFrame}{HTML}{C9C9C0}

% ---------------------------------------------------------
% 2. Spasi
% ---------------------------------------------------------
\onehalfspacing
\setlength{\parindent}{1.2em}
\setlength{\parskip}{0.35em}
\setlength{\headheight}{15pt}
\setlength{\emergencystretch}{2em}
\sloppy

% ---------------------------------------------------------
% 3. Format judul bab
% ---------------------------------------------------------
\titleformat{\chapter}[display]
    {\normalfont\sffamily\bfseries\color{obeBlue}}
    {\filright\normalsize\MakeUppercase{\chaptertitlename}~\Huge\thechapter}
    {0.6ex}
    {\titlerule[1pt]\vspace{1.4ex}\LARGE\filright}

\titlespacing*{\chapter}{0pt}{-10pt}{28pt}

% ---------------------------------------------------------
% 4. Format judul section dan subsection
% ---------------------------------------------------------
\titleformat{\section}
    {\normalfont\large\sffamily\bfseries\color{obeBlue}}
    {\thesection}{0.7em}{}

\titleformat{\subsection}
    {\normalfont\normalsize\sffamily\bfseries\color{obeTeal}}
    {\thesubsection}{0.6em}{}

\titleformat{\subsubsection}
    {\normalfont\normalsize\sffamily\bfseries\color{obeGray}}
    {\thesubsubsection}{0.6em}{}

\titlespacing*{\section}{0pt}{2.4ex plus 0.6ex minus 0.2ex}{1.1ex}
\titlespacing*{\subsection}{0pt}{2.0ex plus 0.5ex minus 0.2ex}{0.9ex}
\titlespacing*{\subsubsection}{0pt}{1.6ex plus 0.4ex minus 0.2ex}{0.7ex}

% ---------------------------------------------------------
% 5. Format judul tanpa nomor (\section*)
% ---------------------------------------------------------
\newcommand{\obeheading}[1]{%
    \section*{\color{obeBlue}#1}%
    \addcontentsline{toc}{section}{#1}%
    \vspace{-0.4em}%
    \noindent\textcolor{obeBlue}{\rule{\textwidth}{0.8pt}}%
    \vspace{0.6em}\par
}

% ---------------------------------------------------------
% 6. Header dan footer
% ---------------------------------------------------------
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\small\sffamily\color{obeGray}\nouppercase{\leftmark}}
\fancyhead[R]{\small\sffamily\color{obeGray}\thepage}
\fancyfoot[C]{}

\fancypagestyle{plain}{%
    \fancyhf{}%
    \renewcommand{\headrulewidth}{0pt}%
    \fancyfoot[C]{\small\sffamily\thepage}%
}

% ---------------------------------------------------------
% 7. Keterangan gambar dan tabel
% ---------------------------------------------------------
\captionsetup{
    font=small,
    labelfont={bf,color=obeBlue},
    labelsep=colon,
    skip=6pt
}

\captionsetup[table]{position=top}

% ---------------------------------------------------------
% 8. Daftar
% ---------------------------------------------------------
\setlist[itemize]{topsep=0.4em, itemsep=0.15em, parsep=0.15em}
\setlist[enumerate]{topsep=0.4em, itemsep=0.2em, parsep=0.15em}

% ---------------------------------------------------------
% 9. Jarak tabel
%
% \tabcolsep sedikit dirapatkan (6pt -> 5pt) agar tabel dengan
% banyak kolom tidak melewati lebar teks.
% ---------------------------------------------------------
\renewcommand{\arraystretch}{1.25}
\setlength{\tabcolsep}{5pt}

% ---------------------------------------------------------
% 10. Tautan silang
% ---------------------------------------------------------
\hypersetup{
    colorlinks=false,
    pdfborder={0 0 0},
    bookmarksnumbered=true,
    bookmarksopen=true
}
