\section{Evaluasi}
\label{sec:evaluasi-bab15}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Elemen ketiga array word \code{deret} berada pada offset
          \dots
          \begin{enumerate}
              \item \code{deret+1} \item \code{deret+2}
              \item \code{deret+4} \item \code{deret+6}
          \end{enumerate}

    \item Offset elemen baris $r$ kolom $c$ pada matriks word dengan
          $K$ kolom adalah \dots
          \begin{enumerate}
              \item $(r + c) \times 2$
              \item $(r \times K + c) \times 2$
              \item $(r \times c + K) \times 2$
              \item $(K \times c + r) \times 2$
          \end{enumerate}

    \item Penanda akhir string untuk layanan DOS fungsi \hex{09}
          adalah \dots
          \begin{enumerate}
              \item \hex{00} \item \code{'\$'}
              \item \hex{0D} \item \hex{FF}
          \end{enumerate}

    \item Konversi bilangan desimal menjadi digit memakai pembagian
          berulang oleh \dots
          \begin{enumerate}
              \item 2 \item 8
              \item 10 \item 16
          \end{enumerate}

    \item Nilai heksadesimal 10 diubah menjadi karakter \code{'A'}
          dengan menambahkan \dots{} pada nilai digit.
          \begin{enumerate}
              \item \hex{30} \item \hex{37}
              \item \hex{41} \item \hex{07}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan cara mendeklarasikan array satu dimensi dan
          menghitung offset elemennya. Jelaskan pula perhitungan
          offset array multi dimensi.

    \item Tuliskan perulangan untuk menyalin string dari \reg{SI} ke
          \reg{DI} sampai penanda \code{'\$'}. Jelaskan pula cara
          membandingkan dua string.

    \item Jelaskan algoritme konversi bilangan desimal. Mengapa sisa
          pembagian disimpan pada \eng{stack}, dan bagaimana urutan
          pengambilannya?

    \item Jelaskan algoritme konversi bilangan heksadesimal. Tuliskan
          potongan program yang mencetak \reg{AX} sebagai empat digit
          heksadesimal.
\end{enumerate}
