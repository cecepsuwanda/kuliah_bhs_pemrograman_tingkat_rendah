\section{Praktikum}
\label{sec:praktikum-bab15}

\begin{praktikumbox}
\vspace{1.1em}
\textbf{Judul:} Mengolah array/string dan mencetak bilangan.\\[0.3em]
\textbf{Sub-CPMK:} 3.6 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu mengakses array,
menyalin string, serta mengubah dan mencetak bilangan desimal dan
heksadesimal.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Komputer dengan emu8086 terpasang.
    \item DOSBox dengan TASM, TLINK, dan DEBUG.
    \item Berkas \file{array-string.asm} dan
          \file{cetak-bilangan.asm} dari \file{code/asm/}.
    \item Lembar pengamatan memori dan keluaran.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Menguji array dan string.}
    \begin{enumerate}
        \item Rakit \file{array-string.asm} dengan TASM, lalu link
              dengan TLINK.
        \item Jalankan program. Program tidak mencetak keluaran.
        \item Jalankan \file{debug array-string.exe}, lalu periksa
              segmen data dengan \lstasm|D DS:0000 L 30|.
        \item Catat nilai \code{jumlah}, isi \code{tujuan}, dan
              \code{panjang}.
        \item Bandingkan dengan Tabel~\ref{tab:hasil-array}.
    \end{enumerate}

    \item \textbf{Mengubah array.}
    \begin{enumerate}
        \item Ubah array menjadi \code{5, 10, 15, 20, 25}.
        \item Rakit ulang dan periksa \code{jumlah}.
        \item Pastikan hasilnya 75.
    \end{enumerate}

    \item \textbf{Mengubah string.}
    \begin{enumerate}
        \item Ubah \code{sumber} menjadi \code{'MIKROPROSESOR\$'}.
        \item Perbesar buffer \code{tujuan} bila diperlukan.
        \item Rakit ulang dan pastikan salinan serta panjangnya
              benar.
    \end{enumerate}

    \item \textbf{Menguji pencetakan bilangan.}
    \begin{enumerate}
        \item Rakit dan jalankan \file{cetak-bilangan.asm}.
        \item Catat keluarannya.
        \item Ubah \code{nilai} menjadi 255, lalu rakit ulang.
        \item Pastikan keluaran desimal 255 dan heksadesimal 00FF.
    \end{enumerate}

    \item \textbf{Menelusuri konversi desimal.}
    \begin{enumerate}
        \item Jalankan program pada emu8086 dengan \eng{single step}.
        \item Masuk ke \code{CETAK\_DESIMAL}.
        \item Catat hasil bagi dan sisa setiap \asm{DIV BX}.
        \item Jelaskan mengapa sisa disimpan pada \eng{stack}.
    \end{enumerate}

    \item \textbf{Menelusuri konversi heksadesimal.}
    \begin{enumerate}
        \item Masuk ke \code{CETAK\_HEX}.
        \item Catat nilai \reg{AL} setelah \asm{ROL AX, 4} dan
              \asm{AND DL, 0Fh}.
        \item Jelaskan mengapa nilai 10--15 perlu ditambah 7 sebelum
              diubah menjadi karakter.
    \end{enumerate}
\end{enumerate}

\subsection{Hasil yang Dikumpulkan}

\begin{enumerate}
    \item Berkas \file{array-string.asm} dan
          \file{cetak-bilangan.asm} versi modifikasi.
    \item Tangkapan layar isi segmen data.
    \item Tangkapan layar keluaran program pencetakan.
    \item Tabel hasil konversi desimal dan heksadesimal.
    \item Penjelasan penggunaan \eng{stack} pada konversi desimal.
\end{enumerate}
