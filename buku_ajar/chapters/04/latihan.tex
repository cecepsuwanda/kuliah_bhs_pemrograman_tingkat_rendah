\section{Latihan}
\label{sec:latihan-bab4}

\begin{enumerate}
    \item \textbf{Konseptual.} Mengapa 8086 membagi prosesor menjadi
          BIU dan EU? Jelaskan keuntungan pembagian tersebut
          dibandingkan prosesor yang mengambil dan mengeksekusi
          instruksi secara bergantian.

    \item \textbf{BIU.} Sebutkan tugas utama BIU dan empat komponen
          utamanya. Jelaskan peran register segmen dan
          \eng{Instruction Pointer} dalam pengambilan instruksi.

    \item \textbf{EU.} Sebutkan tiga tugas utama EU. Jelaskan pula
          peran ALU, register \eng{flag}, dan \eng{operand} dalam
          eksekusi sebuah instruksi.

    \item \textbf{Alamat fisik.} Hitung alamat fisik untuk pasangan
          berikut.
          \begin{enumerate}
              \item \reg{CS} $=$ \hex{1000}, \reg{IP} $=$ \hex{0025}
              \item \reg{DS} $=$ \hex{2000}, \reg{BX} $=$ \hex{0100}
              \item \reg{SS} $=$ \hex{3000}, \reg{SP} $=$ \hex{FFFE}
          \end{enumerate}

    \item \textbf{Instruction queue.} Jelaskan mengapa BIU memerlukan
          \eng{queue} enam byte. Apa yang terjadi ketika isi antrean
          berkurang dua byte? Mengapa ukuran antrean tidak dibuat
          sangat besar?

    \item \textbf{Penelusuran.} Instruksi \lstasm|MOV AX, 1234h|
          berada pada \reg{CS}:\reg{IP} $=$ \hex{1000}:\hex{0005}.
          Instruksi tersebut berukuran tiga byte.
          \begin{enumerate}
              \item Tentukan alamat fisik byte pertama.
              \item Tentukan nilai \reg{IP} setelah seluruh byte
                    instruksi diambil.
              \item Sebutkan unit yang mengambil instruksi dan unit
                    yang mengeksekusinya.
              \item Jelaskan mengapa hasil instruksi tidak memerlukan
                    siklus bus tulis.
          \end{enumerate}

    \item \textbf{Perbandingan.} Bandingkan \lstasm|MOV AX, BX| dan
          \lstasm|MOV AX, [BX]| dari sudut pandang pekerjaan BIU.
          Instruksi mana yang memerlukan siklus bus tambahan?
          Jelaskan alasannya.

    \item \textbf{Analisis.} Sebuah program banyak memakai operand
          memori, sedangkan program lain banyak memakai operand
          register. Program mana yang lebih sedikit menunggu BIU?
          Kaitkan jawaban Anda dengan fungsi \eng{queue} dan siklus
          bus.
\end{enumerate}

\begin{refleksibox}
\vspace{1.1em}
Jawab pertanyaan berikut pada jurnal praktikum Anda.
\begin{enumerate}
    \item Setelah menelusuri program dengan \eng{single step},
          bagian mana dari kerja BIU yang paling sulit diamati?
          Mengapa?
    \item Apakah pembagian BIU/EU membantu Anda memahami mengapa
          program yang sama dapat memiliki waktu eksekusi berbeda?
          Jelaskan.
    \item Konsep apa yang perlu Anda ulangi sebelum mempelajari
          register dan \eng{flag} pada bab berikutnya?
\end{enumerate}
\end{refleksibox}
