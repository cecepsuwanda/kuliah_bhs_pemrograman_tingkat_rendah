\section{Evaluasi}
\label{sec:evaluasi-bab4}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Unit yang bertanggung jawab menghitung alamat fisik dari
          pasangan \eng{segment:offset} pada 8086 adalah \dots
          \begin{enumerate}
              \item \eng{Execution Unit}
              \item \eng{Bus Interface Unit}
              \item \eng{Arithmetic Logic Unit}
              \item register \eng{flag}
          \end{enumerate}

    \item Ukuran \eng{instruction queue} pada mikroprosesor 8086
          adalah \dots
          \begin{enumerate}
              \item 2 byte \item 4 byte
              \item 6 byte \item 16 byte
          \end{enumerate}

    \item Bila \reg{CS} $=$ \hex{1000} dan \reg{IP} $=$ \hex{0025},
          maka instruksi berikutnya diambil dari alamat fisik \dots
          \begin{enumerate}
              \item \hex{10025} \item \hex{10250}
              \item \hex{1002} \item \hex{100025}
          \end{enumerate}

    \item Operasi aritmetika dan logika dijalankan oleh \dots
          \begin{enumerate}
              \item \eng{Bus Interface Unit}
              \item \eng{Instruction Pointer}
              \item ALU di dalam EU
              \item register segmen
          \end{enumerate}

    \item Register yang menyimpan \eng{offset} instruksi berikutnya
          adalah \dots
          \begin{enumerate}
              \item \reg{CS} \item \reg{DS}
              \item \reg{IP} \item \reg{SP}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Gambarkan diagram blok 8086 yang memuat EU, BIU, bus
          internal, dan bus eksternal. Sebutkan komponen yang berada
          di dalam masing-masing unit dan jelaskan mengapa EU tidak
          mengakses bus luar secara langsung.

    \item Jelaskan peran \eng{instruction queue} enam byte. Gunakan
          contoh tiga instruksi untuk menunjukkan bagaimana
          pengambilan instruksi berikutnya dapat berjalan sementara
          EU mengeksekusi instruksi sekarang. Jelaskan pula kapan
          keuntungan antrean berkurang.

    \item Instruksi \lstasm|MOV AX, 1234h| berada pada
          \reg{CS}:\reg{IP} $=$ \hex{1000}:\hex{0005} dan berukuran
          tiga byte. Uraikan tahap \eng{fetch}, \eng{decode}, dan
          \eng{execute}, sertakan nilai \reg{IP} sebelum dan sesudah
          pengambilan, serta jelaskan mengapa tidak ada siklus bus
          tulis.

    \item Bandingkan \lstasm|MOV AX, BX| dan \lstasm|MOV AX, [BX]|
          dari sudut pandang pekerjaan BIU dan EU. Tunjukkan pada
          tahap mana perbedaan waktu muncul dan mengapa.
\end{enumerate}

\vspace{1.1em}
\textbf{Checklist Pencapaian Kompetensi
Bab~\ref{chap:arsitektur-8086}}

Seorang mahasiswa dinyatakan kompeten pada Sub-CPMK 1.2 apabila:
\begin{itemize}
    \item[\checked] dapat menjelaskan pembagian BIU dan EU beserta
          alasannya;
    \item[\checked] dapat menyebutkan tugas dan komponen BIU;
    \item[\checked] dapat menyebutkan tugas dan komponen EU;
    \item[\checked] dapat menjelaskan peran \eng{queue} enam byte;
    \item[\checked] dapat menghitung alamat fisik dari
          \eng{segment:offset};
    \item[\checked] dapat menelusuri urutan
          \eng{fetch}--\eng{decode}--\eng{execute};
    \item[\checked] dapat membedakan instruksi register dan memori
          dari sisi siklus bus.
\end{itemize}
\end{checklistbox}
