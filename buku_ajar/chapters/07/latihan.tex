\section{Latihan}
\label{sec:latihan-bab7}

\begin{enumerate}
    \item \textbf{Rantai proses.} Jelaskan tahapan dari berkas
          \file{.ASM} sampai program berjalan. Sebutkan peran
          editor, TASM, TLINK, dan \eng{loader}.

    \item \textbf{COM dan EXE.} Sebutkan empat perbedaan penting
          antara program \file{.COM} dan \file{.EXE}. Mengapa
          program \file{.COM} tidak memerlukan inisialisasi
          \reg{DS}, sedangkan program \file{.EXE} memerlukannya?

    \item \textbf{Struktur program.} Jelaskan fungsi direktif
          \dir{.MODEL}, \dir{.STACK}, \dir{.DATA}, \dir{.CODE},
          \dir{ORG 100h}, dan \dir{END}. Jelaskan pula aturan
          penulisan label dan komentar.

    \item \textbf{Direktif data.} Jelaskan perbedaan \dir{DB},
          \dir{DW}, \dir{DD}, \dir{EQU}, dan \dir{DUP}. Berapa byte
          memori yang disediakan oleh masing-masing contoh berikut?
          \begin{enumerate}
              \item \code{nilai DB 25}
              \item \code{kata DW 1234h}
              \item \code{besar DD 12345678h}
              \item \code{ruang DB 10 DUP(?)}
          \end{enumerate}

    \item \textbf{Penulisan data.} Tulislah potongan segmen data
          yang memuat:
          \begin{enumerate}
              \item string \code{'Selamat belajar'} diakhiri pindah
                    baris dan \code{'\$'};
              \item konstanta \code{PANJANG} bernilai 20;
              \item ruang kosong 16 byte.
          \end{enumerate}

    \item \textbf{Analisis kerangka.} Perhatikan kerangka program
          berikut.
          \begin{lstlisting}[language=MASM, numbers=none]
.MODEL SMALL
.STACK 100h
.DATA
    nilai DW 1234h
.CODE
START:
    MOV AX, @DATA
    MOV DS, AX
    MOV AX, nilai
    MOV AH, 4Ch
    INT 21h
END START
          \end{lstlisting}
          Jelaskan mengapa \lstasm|MOV AX, @DATA| dan
          \lstasm|MOV DS, AX| diperlukan, dan apa akibatnya bila
          kedua baris itu dihapus.

    \item \textbf{Analisis kesalahan.} Sebutkan penyebab dan cara
          memperbaiki setiap kasus berikut.
          \begin{enumerate}
              \item TASM melaporkan \file{Undefined symbol: PESAN}.
              \item TASM melaporkan \file{Operand types do not
                    match}.
              \item Program \file{.EXE} memakai \dir{ORG 100h}.
              \item Berkas tidak memiliki direktif \dir{END}.
          \end{enumerate}

    \item \textbf{Penulisan program.} Tulislah program \file{.COM}
          lengkap yang mencetak nama dan NIM Anda pada dua baris
          terpisah. Gunakan \dir{ORG 100h}, layanan DOS
          \code{INT 21h} fungsi \hex{09}, dan akhiri dengan
          \code{INT 20h}.
\end{enumerate}

\begin{refleksibox}
\vspace{1.1em}
Jawab pertanyaan berikut pada jurnal praktikum Anda.
\begin{enumerate}
    \item Tahap mana yang paling sering menimbulkan kesalahan saat
          Anda merakit program: penulisan, perakitan, atau
          pengaitan? Mengapa?
    \item Setelah membandingkan program \file{.COM} dan \file{.EXE},
          bentuk mana yang lebih mudah Anda pahami? Jelaskan.
    \item Konsep apa yang perlu Anda ulangi sebelum mempelajari
          instruksi transfer data pada bab berikutnya?
\end{enumerate}
\end{refleksibox}
