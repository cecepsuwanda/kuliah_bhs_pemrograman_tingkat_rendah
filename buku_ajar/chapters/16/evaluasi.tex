\section{Evaluasi}
\label{sec:evaluasi-bab16}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Perintah DEBUG untuk menjalankan satu instruksi adalah
          \dots
          \begin{enumerate}
              \item \file{R} \item \file{T}
              \item \file{D} \item \file{G}
          \end{enumerate}

    \item Perintah yang menjalankan satu instruksi tetapi melewati
          prosedur adalah \dots
          \begin{enumerate}
              \item \file{T} \item \file{P}
              \item \file{D} \item \file{R}
          \end{enumerate}

    \item Perintah DEBUG untuk menampilkan isi memori adalah \dots
          \begin{enumerate}
              \item \file{R} \item \file{T}
              \item \file{D} \item \file{G}
          \end{enumerate}

    \item Penyebab \eng{hang} pada \file{bug-loop.asm} adalah \dots
          \begin{enumerate}
              \item \reg{CX} dinaikkan di dalam perulangan
              \item \asm{LOOP} tidak pernah dijalankan
              \item \asm{ADD AX, CX} salah operand
              \item program tidak memiliki \dir{.DATA}
          \end{enumerate}

    \item Penyebab \eng{hang} pada \file{bug-stack.asm} adalah \dots
          \begin{enumerate}
              \item terlalu banyak \asm{POP}
              \item satu \asm{POP} tidak ada
              \item \asm{CALL} tidak menyimpan alamat kembali
              \item prosedur tidak memiliki \asm{RET}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan perbedaan kesalahan sintaks, kesalahan logika, dan
          \eng{hang}. Sebutkan alat dan perintah yang dipakai untuk
          masing-masing.

    \item Jelaskan hasil penelusuran \file{bug-loop.asm}. Mengapa
          \reg{CX} tidak pernah mencapai nol, dan bagaimana
          perbaikannya?

    \item Jelaskan hasil pemeriksaan stack pada \file{bug-stack.asm}.
          Mengapa \asm{RET} mengambil alamat yang salah, dan
          bagaimana perbaikannya?

    \item Tuliskan langkah debugging yang sistematis untuk program
          yang berhenti. Sebutkan informasi yang harus dicatat pada
          laporan debugging.
\end{enumerate}
