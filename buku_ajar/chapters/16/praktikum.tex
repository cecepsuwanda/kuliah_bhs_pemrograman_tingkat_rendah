\section{Praktikum}
\label{sec:praktikum-bab16}

\begin{praktikumbox}
\vspace{1.1em}
\textbf{Judul:} Menemukan dan memperbaiki kesalahan dengan
\eng{single step} dan \eng{dump}.\\[0.3em]
\textbf{Sub-CPMK:} 4.1--4.3 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu menelusuri
program per instruksi, memeriksa register, \eng{flag}, memori, dan
stack, menemukan penyebab \eng{hang}, serta memperbaiki kesalahan.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Komputer dengan emu8086 terpasang.
    \item DOSBox dengan TASM, TLINK, dan DEBUG.
    \item Berkas \file{bug-loop.asm}, \file{bug-loop-fix.asm},
          \file{bug-stack.asm}, dan \file{bug-stack-fix.asm}.
    \item Lembar laporan debugging.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Reproduksi kesalahan perulangan.}
    \begin{enumerate}
        \item Rakit \file{bug-loop.asm} dan jalankan.
        \item Amati gejalanya. Hentikan program bila tidak selesai.
        \item Buka program pada emu8086 dan jalankan
              \eng{single step} maksimal sepuluh langkah.
        \item Catat \reg{AX} dan \reg{CX} pada setiap langkah.
        \item Tentukan baris yang menyebabkan \reg{CX} tidak pernah
              nol.
    \end{enumerate}

    \item \textbf{Memperbaiki kesalahan perulangan.}
    \begin{enumerate}
        \item Hapus \lstasm|INC CX| dari program.
        \item Rakit ulang dan jalankan.
        \item Pastikan keluaran menjadi \code{Jumlah 1..5 = 15}.
        \item Bandingkan hasil Anda dengan \file{bug-loop-fix.asm}.
    \end{enumerate}

    \item \textbf{Reproduksi kesalahan stack.}
    \begin{enumerate}
        \item Rakit \file{bug-stack.asm} dan jalankan.
        \item Amati gejalanya.
        \item Jalankan di bawah DEBUG, lalu masuk ke prosedur
              \code{HITUNG} dengan \file{T}.
        \item Catat \reg{SP} sebelum \asm{CALL}, setelah \asm{CALL},
              setelah tiga \asm{PUSH}, dan setelah dua \asm{POP}.
        \item Tampilkan stack dengan \lstasm|D SS:SP L 10|.
        \item Tentukan word mana yang tersisa sebelum \asm{RET}.
    \end{enumerate}

    \item \textbf{Memperbaiki kesalahan stack.}
    \begin{enumerate}
        \item Tambahkan \lstasm|POP AX| sebelum \asm{RET}.
        \item Rakit ulang dan jalankan.
        \item Pastikan pesan \code{Selesai.} tercetak.
        \item Bandingkan hasil Anda dengan \file{bug-stack-fix.asm}.
    \end{enumerate}

    \item \textbf{Memeriksa memori dan flag.}
    \begin{enumerate}
        \item Pada salah satu program, catat \reg{DS} dengan
              \file{R}.
        \item Tampilkan variabel dengan \lstasm|D DS:0000 L 10|.
        \item Catat \flag{ZF}, \flag{CF}, dan \flag{SF} setelah
              instruksi \asm{CMP} atau \asm{LOOP}.
    \end{enumerate}

    \item \textbf{Menulis laporan debugging.}
    \begin{enumerate}
        \item Tuliskan gejala, dugaan, langkah penelusuran, penyebab,
              dan perbaikan untuk setiap kasus.
        \item Sertakan tabel penelusuran register/stack.
    \end{enumerate}
\end{enumerate}

\subsection{Hasil yang Dikumpulkan}

\begin{enumerate}
    \item Berkas perbaikan \file{bug-loop.asm} dan
          \file{bug-stack.asm}.
    \item Tabel penelusuran \reg{AX}, \reg{CX}, \reg{SP}, dan isi
          stack.
    \item Tangkapan layar keluaran setelah perbaikan.
    \item Laporan debugging yang memuat gejala, penyebab, dan
          perbaikan.
\end{enumerate}
