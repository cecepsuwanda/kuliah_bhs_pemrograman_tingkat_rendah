\chapter{Debugging Level Instruksi}
\label{chap:debugging}

% =========================================================
% OUTCOME PEMBELAJARAN KHUSUS BAB
% =========================================================

\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{4.1} --- Melakukan penelusuran program per instruksi
(\textit{single step}/\textit{trace}) dan memantau perubahan register
serta \textit{flag}.

\vspace{0.4em}
\SubCPMK{4.2} --- Memeriksa isi memori dan stack (\textit{dump}) untuk
menemukan kesalahan sintaks dan kesalahan logika.

\vspace{0.4em}
\SubCPMK{4.3} --- Menganalisis penyebab program berhenti
(\textit{hang}) dan memperbaiki kesalahan berdasarkan hasil
penelusuran.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{04} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{03}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item membedakan kesalahan sintaks, kesalahan logika, dan
          program yang berhenti (\eng{hang});
    \item menggunakan DEBUG, emu8086, dan Turbo Debugger untuk
          menelusuri program;
    \item mencatat perubahan register dan \eng{flag} pada setiap
          instruksi;
    \item memeriksa isi memori dan stack dengan perintah
          \eng{dump};
    \item menemukan penyebab perulangan tak berujung;
    \item menemukan ketidakseimbangan \asm{PUSH} dan \asm{POP};
    \item memperbaiki kesalahan berdasarkan hasil penelusuran;
    \item menyusun laporan debugging yang sistematis.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Jenis kesalahan: sintaks, logika, dan \eng{hang}.
    \item Alat debugging: DEBUG, emu8086, dan Turbo Debugger.
    \item Perintah debugging: register, \eng{single step},
          \eng{proceed}, \eng{dump}, dan \eng{go}.
    \item Penelusuran register dan \eng{flag}.
    \item Pemeriksaan memori dan stack.
    \item Analisis perulangan tak berujung.
    \item Analisis ketidakseimbangan stack.
    \item Perbaikan dan verifikasi hasil.
\end{enumerate}
\end{materibox}
