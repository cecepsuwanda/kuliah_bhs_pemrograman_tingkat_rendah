\section{Evaluasi}
\label{sec:evaluasi-bab13}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Instruksi yang memindahkan kendali tanpa memeriksa
          \eng{flag} adalah \dots
          \begin{enumerate}
              \item \asm{JE} \item \asm{JNE}
              \item \asm{JMP} \item \asm{LOOP}
          \end{enumerate}

    \item Instruksi \asm{CMP AX, BX} \dots
          \begin{enumerate}
              \item mengubah isi \reg{AX} dan \reg{BX}
              \item hanya mengubah \eng{flag}
              \item menyimpan hasil di \reg{AX}
              \item menyimpan hasil di memori
          \end{enumerate}

    \item Setelah \lstasm|CMP AX, BX|, lompatan untuk
          \reg{AX} $>$ \reg{BX} pada bilangan bertanda adalah
          \dots
          \begin{enumerate}
              \item \asm{JA} \item \asm{JG}
              \item \asm{JB} \item \asm{JL}
          \end{enumerate}

    \item Instruksi \asm{LOOP} memakai register \dots{} sebagai
          pencacah.
          \begin{enumerate}
              \item \reg{AX} \item \reg{BX}
              \item \reg{CX} \item \reg{DX}
          \end{enumerate}

    \item \asm{LOOPE} melompat bila \reg{CX} belum nol dan \dots
          \begin{enumerate}
              \item \flag{ZF} $=1$ \item \flag{ZF} $=0$
              \item \flag{CF} $=1$ \item \flag{SF} $=1$
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan aturan penulisan label. Jelaskan pula perbedaan
          \asm{JMP} dan lompatan bersyarat, serta peran \asm{CMP}
          dalam percabangan.

    \item Setelah \lstasm|CMP AX, BX|, sebutkan lompatan yang tepat
          untuk perbandingan $>$, $\geq$, $<$, dan $\leq$ pada
          bilangan bertanda dan tidak bertanda.

    \item Terjemahkan pola \code{if--else} berikut ke assembly,
          lengkap dengan label dan lompatan.
          \begin{lstlisting}[language=MASM, numbers=none]
if (AX == BX) {
    hasil = 1;
} else {
    hasil = 0;
}
          \end{lstlisting}

    \item Jelaskan cara kerja \asm{LOOP} dan \asm{LOOPE}. Jelaskan
          pula mengapa perulangan bersarang memerlukan penyimpanan
          pencacah luar.
\end{enumerate}
