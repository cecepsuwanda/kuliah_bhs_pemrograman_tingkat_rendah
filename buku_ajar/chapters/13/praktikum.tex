\section{Praktikum}
\label{sec:praktikum-bab13}

\begin{praktikumbox}
\textbf{Judul:} Menulis prosedur berparameter dan memeriksa isi stack
dengan \eng{debugger}.\\[0.3em]
\textbf{Sub-CPMK:} 3.4 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu \quad\textbullet\quad
\textbf{Tugas:} Tugas 6 (dikumpulkan pada pertemuan ini)
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu menulis prosedur
berparameter melalui stack, mengembalikan nilai melalui register
\reg{AX}, serta membuktikan kebenaran \eng{stack frame} dengan
mengamati isi stack pada \eng{debugger}.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Emu8086 dengan jendela register, memori, dan stack, atau
          GUI Turbo Assembler dengan \eng{debugger} terpadu.
    \item DOSBox beserta perangkat \file{DEBUG}, TASM, dan TLINK.
    \item Berkas \file{prosedur-luas.asm} dan \file{faktorial.asm}
          dari direktori \file{code/asm/}.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Menjalankan prosedur berparameter.}
    \begin{enumerate}
        \item Rakit dan jalankan \file{prosedur-luas.asm}. Periksa
              isi \reg{AX} dan variabel \code{luas}; pastikan
              nilainya \hex{0032}.
        \item Ubah \code{lebar} menjadi \dec{7} dan \code{tinggi}
              menjadi \dec{6}, lalu pastikan hasilnya \dec{42}
              (\hex{002A}).
        \item Jelaskan mengapa urutan instruksi \asm{PUSH} tidak
              boleh ditukar tanpa mengubah badan prosedur.
    \end{enumerate}

    \item \textbf{Menelusuri isi stack per langkah.}
    \begin{enumerate}
        \item Jalankan program dengan \file{Single Step}, mulai dari
              \lstasm|PUSH tinggi|.
        \item Catat nilai \reg{SP}, \reg{BP}, dan \reg{SS} pada
              setiap langkah ke dalam tabel pengamatan.
        \item Bandingkan hasilnya dengan Tabel~\ref{tab:trace-stack}
              dan Gambar~\ref{fig:isi-stack-call}; pastikan alamat
              setiap sel cocok.
        \item Setelah \lstasm|MOV BP, SP|, buktikan bahwa
              \code{[BP+4]} berisi \hex{0005} dan \code{[BP+6]}
              berisi \hex{000A}.
    \end{enumerate}

    \item \textbf{Membedah prosedur rekursif.}
    \begin{enumerate}
        \item Jalankan \file{faktorial.asm} sampai pemanggilan
              \code{Faktorial} yang pertama.
        \item Catat nilai \reg{SP} sebelum dan sesudah pemanggilan
              rekursif. Berapa byte yang berkurang pada setiap
              tingkat?
        \item Hitung kedalaman rekursi dan perkirakan pemakaian stack
              untuk $n = 5$ serta $n = \dec{8}$.
    \end{enumerate}

    \item \textbf{Menguji kesalahan stack yang disengaja.}
    \begin{enumerate}
        \item Pada salinan \file{prosedur-luas.asm}, ubah
              \asm{RET} \code{4} menjadi \asm{RET} biasa, lalu
              rakit ulang dan jalankan.
        \item Jelaskan mengapa kesalahan ini tidak dilaporkan
              assembler, dan bagaimana \eng{debugger} membantu
              menemukannya.
    \end{enumerate}
\end{enumerate}

\subsection{Hasil yang Dikumpulkan (Tugas 6)}

\begin{enumerate}
    \item Berkas \file{prosedur-luas.asm} yang dimodifikasi memakai
          tiga parameter (lebar, tinggi, dan satuan harga), dengan
          prosedur yang benar-benar mengakses \code{[BP+8]}.
    \item Berkas \file{prosedur-ganjil.asm} berisi prosedur
          \code{HitungJumlahGanjil} yang menjumlahkan bilangan ganjil
          dari \dec{1} sampai $n$; parameter dikirim melalui stack
          dan hasil dikembalikan pada \reg{AX}.
    \item Tabel pengamatan nilai \reg{SP}, \reg{BP}, dan \reg{SS}
          pada setiap langkah pemanggilan prosedur.
    \item Tangkapan layar jendela stack yang memperlihatkan isi sel
          \code{[BP+2]}, \code{[BP+4]}, dan \code{[BP+6]}.
    \item Jawaban tertulis atas pertanyaan pada langkah kerja nomor
          1(c), 3(b), dan 4(b).
\end{enumerate}
