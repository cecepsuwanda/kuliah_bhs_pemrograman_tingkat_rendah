\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{3.4} --- Menyusun percabangan dan perulangan dengan label,
JMP, CMP, lompatan bersyarat, dan LOOP.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{03} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{03}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item menulis label dan menggunakannya sebagai tujuan lompatan;
    \item menggunakan \asm{JMP} untuk lompatan tanpa syarat;
    \item menggunakan \asm{CMP} untuk menyiapkan \eng{flag};
    \item memilih lompatan bersyarat yang tepat untuk bilangan
          bertanda dan tidak bertanda;
    \item menyusun percabangan \code{if} dan \code{if--else};
    \item menyusun perulangan dengan \asm{LOOP};
    \item menerjemahkan pola \code{while} dan \code{for} ke assembly;
    \item menyusun perulangan bersarang untuk pengolahan array.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Label dan aturan penulisannya.
    \item Lompatan tanpa syarat \asm{JMP}.
    \item \asm{CMP} dan \eng{flag} sebagai dasar keputusan.
    \item Lompatan bersyarat \asm{Jcc}.
    \item Perulangan \asm{LOOP}.
    \item Pola \code{if}, \code{if--else}, \code{while}, dan
          \code{for}.
    \item Perulangan bersarang dan pengolahan array.
\end{enumerate}
\end{materibox}
