\section{Evaluasi}
\label{sec:evaluasi-bab9}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Ukuran tabel vektor interupsi pada 8086 adalah \dots
          \begin{enumerate}
              \item 256 byte \item 512 byte
              \item 1 KB \item 64 KB
          \end{enumerate}

    \item Entri vektor untuk \code{INT 21h} berada pada alamat
          \dots
          \begin{enumerate}
              \item \hex{0000}:\hex{0021}
              \item \hex{0000}:\hex{0042}
              \item \hex{0000}:\hex{0084}
              \item \hex{0000}:\hex{2100}
          \end{enumerate}

    \item String yang dicetak dengan \code{INT 21h} fungsi \hex{09}
          harus diakhiri \dots
          \begin{enumerate}
              \item \hex{00} \item \code{'\$'}
              \item \hex{0D} \item \hex{FF}
          \end{enumerate}

    \item Fungsi \code{INT 10h} untuk menetapkan posisi kursor
          adalah \dots
          \begin{enumerate}
              \item \hex{00} \item \hex{02}
              \item \hex{09} \item \hex{13}
          \end{enumerate}

    \item Instruksi yang memulihkan \reg{FLAGS}, \reg{CS}, dan
          \reg{IP} setelah interupsi adalah \dots
          \begin{enumerate}
              \item \code{RET} \item \code{IRET}
              \item \code{INT} \item \code{JMP}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan perbedaan interupsi perangkat keras dan perangkat
          lunak. Jelaskan pula susunan tabel vektor interupsi
          beserta pembagian nomor BIOS dan DOS.

    \item Uraikan langkah prosesor ketika menjalankan \code{INT n}.
          Jelaskan mengapa handler harus diakhiri \code{IRET}.

    \item Sebutkan fungsi \code{INT 21h} \hex{01}, \hex{02},
          \hex{09}, \hex{0A}, dan \hex{4C}. Untuk setiap fungsi,
          tuliskan register yang harus diisi dan hasil yang
          diperoleh.

    \item Jelaskan perbedaan \code{INT 10h} dan \code{INT 16h}.
          Berikan satu contoh pemakaian masing-masing beserta
          register yang diisi.
\end{enumerate}

    \item[\checked] dapat menggunakan \code{INT 10h};
