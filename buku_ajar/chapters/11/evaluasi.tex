\section{Evaluasi}
\label{sec:evaluasi-bab11}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Hasil \asm{AND} antara bit 1 dan bit 0 adalah \dots
          \begin{enumerate}
              \item 0 \item 1
              \item bergantung pada \flag{CF}
              \item tidak terdefinisi
          \end{enumerate}

    \item Instruksi yang dipakai untuk membalik bit tertentu
          adalah \dots
          \begin{enumerate}
              \item \asm{AND} \item \asm{OR}
              \item \asm{XOR} \item \asm{NOT}
          \end{enumerate}

    \item \asm{TEST} melakukan operasi internal yang sama dengan
          \dots
          \begin{enumerate}
              \item \asm{ADD} \item \asm{AND}
              \item \asm{OR} \item \asm{SUB}
          \end{enumerate}

    \item Diketahui \reg{AL} = \bin{10100101}. Setelah
          \lstasm|SHL AL, 1|, isi \reg{AL} dan \flag{CF} adalah
          \dots
          \begin{enumerate}
              \item \bin{01001010}, \flag{CF} = 1
              \item \bin{01001010}, \flag{CF} = 0
              \item \bin{11010010}, \flag{CF} = 1
              \item \bin{01010010}, \flag{CF} = 1
          \end{enumerate}

    \item Pergeseran \asm{SHR} satu bit pada bilangan tidak bertanda
          setara dengan \dots
          \begin{enumerate}
              \item menambah 2
              \item mengalikan 2
              \item membagi 2
              \item membalik tanda
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan tabel kebenaran \asm{AND}, \asm{OR}, \asm{XOR},
          dan \asm{NOT}. Jelaskan pula pengaruh masing-masing
          operasi terhadap \flag{CF}, \flag{OF}, \flag{ZF}, dan
          \flag{SF}.

    \item Diketahui \reg{AL} = \hex{A5}. Tuliskan instruksi untuk
          menyalakan, memadamkan, membalik, dan menguji bit 4.
          Jelaskan mask yang dipakai pada setiap instruksi.

    \item Jelaskan perbedaan \asm{TEST} dan \asm{CMP}. Jelaskan
          pula kegunaan \lstasm|TEST AL, AL| dan mengapa instruksi
          itu tidak mengubah isi \reg{AL}.

    \item Jelaskan cara kerja \asm{SHL} dan \asm{SHR}. Tuliskan
          program yang menghitung $5 \times 16$ dengan \asm{SHL}
          dan $1000 \div 8$ dengan \asm{SHR}.
\end{enumerate}

\begin{checklistbox}
\vspace{1.1em}
\textbf{Checklist Pencapaian Kompetensi
Bab~\ref{chap:logika}}

Seorang mahasiswa dinyatakan kompeten pada Sub-CPMK 3.2 apabila:
\begin{itemize}
    \item[\checked] dapat menjelaskan tabel kebenaran operasi
          logika;
    \item[\checked] dapat menjelaskan pengaruh operasi logika
          terhadap \eng{flag};
    \item[\checked] dapat menggunakan mask untuk manipulasi bit;
    \item[\checked] dapat membedakan \asm{TEST} dan \asm{CMP};
    \item[\checked] dapat menggunakan \asm{SHL} dan \asm{SHR};
    \item[\checked] dapat memakai pergeseran untuk perkalian dan
          pembagian pangkat dua;
    \item[\checked] dapat menganalisis program paritas dan
          \eng{XOR cipher}.
\end{itemize}
