\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{1.4} --- Menghitung alamat fisik dari pasangan
\eng{segment:offset} serta menjelaskan pembagian memori 1 MB dan
\eng{overlapping} segmen.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{01} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{02}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item menjelaskan alasan 8086 memakai register segmen 16 bit
          walaupun bus alamatnya 20 bit;
    \item membedakan alamat logis (\eng{segment:offset}) dan alamat
          fisik;
    \item menghitung alamat fisik dari pasangan
          \eng{segment:offset};
    \item menjelaskan paragraf 16 byte dan struktur segmen 64 KB;
    \item menjelaskan \eng{overlapping} segmen dan banyaknya pasangan
          yang menunjuk satu alamat fisik;
    \item menyebutkan pembagian memori 1 MB;
    \item memverifikasi hasil perhitungan alamat dengan DEBUG atau
          emu8086.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Masalah pengalamatan: register 16 bit dan bus alamat 20 bit.
    \item Alamat logis dan alamat fisik.
    \item Rumus alamat fisik dan perhitungan bertingkat.
    \item Paragraf 16 byte dan segmen 64 KB.
    \item \eng{Overlapping} segmen.
    \item Pembagian memori 1 MB.
    \item Verifikasi alamat dengan DEBUG dan emu8086.
\end{enumerate}
\end{materibox}
