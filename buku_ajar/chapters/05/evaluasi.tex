\section{Evaluasi}
\label{sec:evaluasi-bab5}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Setelah instruksi \lstasm|MOV AL, 20h| dijalankan pada
          register \reg{AX} yang semula berisi \hex{1234}, isi
          register \reg{AX} menjadi \dots
          \begin{enumerate}
              \item \hex{1220} \item \hex{2034}
              \item \hex{0020} \item \hex{2020}
          \end{enumerate}

    \item Register yang selalu berpasangan dengan \reg{CS} untuk
          menentukan alamat instruksi berikutnya adalah \dots
          \begin{enumerate}
              \item \reg{SP} \item \reg{IP}
              \item \reg{DS} \item \reg{DI}
          \end{enumerate}

    \item Instruksi \lstasm|ADD AL, BL| dengan \reg{AL} = \hex{7F}
          dan \reg{BL} = \hex{01} menghasilkan \dots
          \begin{enumerate}
              \item \flag{CF} = 1 dan \flag{OF} = 0
              \item \flag{CF} = 0 dan \flag{OF} = 1
              \item \flag{CF} = 1 dan \flag{OF} = 1
              \item \flag{CF} = 0 dan \flag{OF} = 0
          \end{enumerate}

    \item Instruksi berikut yang \textbf{tidak} mengubah nilai
          \eng{flag} adalah \dots
          \begin{enumerate}
              \item \asm{ADD AX, BX} \item \asm{CMP AX, BX}
              \item \asm{MOV AX, BX} \item \asm{INC AX}
          \end{enumerate}

    \item Register yang tidak boleh diubah sembarangan karena
          menentukan letak \eng{stack} adalah \dots
          \begin{enumerate}
              \item \reg{AX} dan \reg{BX} \item \reg{SI} dan \reg{DI}
              \item \reg{DS} dan \reg{ES} \item \reg{SS} dan \reg{SP}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan mengapa \lstasm|MOV AL, 5| dan \lstasm|MOV AX, 5|
          menghasilkan keadaan register yang berbeda bila sebelumnya
          \reg{AH} tidak bernilai nol. Sertakan contoh angka.

    \item Sebutkan peran khas \reg{BX}, \reg{CX}, dan \reg{DX},
          lalu jelaskan mengapa peran tersebut muncul walaupun
          keempat register umum sebenarnya sama-sama dapat menampung
          nilai apa pun.

    \item Diberikan keadaan awal \reg{AL} = \hex{80} dan \reg{BL} =
          \hex{80}. Tentukan hasil \lstasm|ADD AL, BL| beserta nilai
          \flag{CF}, \flag{OF}, \flag{SF}, dan \flag{ZF}. Jelaskan
          mengapa \flag{CF} dan \flag{OF} bernilai demikian, dan
          mengapa hal itu tidak bertentangan.

    \item Seorang mahasiswa mengubah isi \reg{SS} dan \reg{SP}
          secara manual di dalam \eng{debugger} ketika program
          sedang berjalan. Jelaskan akibat yang mungkin terjadi pada
          instruksi \asm{RET} dan mengapa hal itu berbahaya.
\end{enumerate}

\begin{checklistbox}
\textbf{Checklist Pencapaian Kompetensi Bab~\ref{chap:register}}

Seorang mahasiswa dinyatakan kompeten pada Sub-CPMK 1.2 apabila:
\begin{itemize}
    \item[\checked] dapat menyebutkan register umum beserta byte
          rendah dan byte tingginya;
    \item[\checked] dapat menjelaskan peran khas \reg{AX},
          \reg{BX}, \reg{CX}, dan \reg{DX};
    \item[\checked] dapat menjelaskan peran \reg{CS}, \reg{DS},
          \reg{SS}, dan \reg{ES} pada pengalamatan memori;
    \item[\checked] dapat menjelaskan tugas \reg{SP}, \reg{BP},
          \reg{SI}, \reg{DI}, dan \reg{IP};
    \item[\checked] dapat menyebutkan sembilan \eng{flag} beserta
          posisi bitnya;
    \item[\checked] dapat memprediksi perubahan \eng{flag} pada
          operasi \asm{ADD} dan menjelaskan akibatnya bagi alur
          program.
\end{itemize}
\end{checklistbox}
