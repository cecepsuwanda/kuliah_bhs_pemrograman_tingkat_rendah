\section{Praktikum}
\label{sec:praktikum-bab5}

\begin{praktikumbox}
\vspace{1.1em}
\textbf{Judul:} Pengamatan register dan \eng{flag} dengan
\eng{single step}.\\[0.3em]
\textbf{Sub-CPMK:} 1.3 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu mengamati isi
register 8086, membaca keadaan \eng{flag} setelah operasi
aritmatika, dan menjelaskan pengaruh \eng{flag} terhadap jalannya
program.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Komputer dengan emu8086 terpasang.
    \item DOSBox dengan TASM, TLINK, dan DEBUG.
    \item Berkas \file{prediksi-flag.asm} dan \file{pesan-flag.asm}
          dari \file{code/asm/}.
    \item Lembar kerja pengamatan register dan \eng{flag}.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Menyiapkan program pengamatan.}
    \begin{enumerate}
        \item Buka \file{prediksi-flag.asm} pada emu8086.
        \item Pilih \file{Compile and Emulate}, lalu tampilkan
              jendela \file{Registers} dan \file{Flags}.
    \end{enumerate}

    \item \textbf{Mengamati register dengan \eng{single step}.}
    \begin{enumerate}
        \item Catat nilai awal \reg{AX}, \reg{BX}, \reg{CX},
              \reg{DX}, \reg{DS}, \reg{SP}, dan \reg{IP}.
        \item Tekan \file{Single Step} satu kali untuk setiap
              instruksi. Catat perubahan register pada
              Tabel~\ref{tab:pengamatan-register}.
        \item Berhenti sebelum \lstasm|ADD AX, BX| dan catat keadaan
              \eng{flag}.
        \item Jalankan \lstasm|ADD AX, BX| satu langkah, lalu catat
              \flag{CF}, \flag{OF}, \flag{SF}, dan \flag{ZF}.
        \item Lanjutkan sampai kasus kedua selesai.
    \end{enumerate}

    \item \textbf{Mengisi tabel prediksi.}
    \begin{enumerate}
        \item Isi kolom prediksi pada Tabel~\ref{tab:prediksi-flag}
              sebelum menjalankan setiap kasus.
        \item Bandingkan prediksi dengan hasil pengamatan.
        \item Jelaskan setiap prediksi yang tidak sesuai.
    \end{enumerate}

    \item \textbf{Mengamati pengaruh \eng{flag} pada keputusan.}
    \begin{enumerate}
        \item Jalankan \file{pesan-flag.asm} dengan
              \eng{single step}.
        \item Catat keadaan \flag{CF}, \flag{ZF}, dan \flag{SF}
              setelah \lstasm|ADD AL, BL|.
        \item Catat pesan yang tercetak dan jelaskan mengapa pesan
              itu yang dipilih.
    \end{enumerate}

    \item \textbf{Mengubah nilai operand.}
    \begin{enumerate}
        \item Pada \file{prediksi-flag.asm}, ubah kasus pertama
              menjadi \hex{7FFE} $+$ \hex{0002}.
        \item Rakit ulang, lalu prediksikan \eng{flag} sebelum
              menjalankan.
        \item Periksa hasil prediksi dengan \eng{single step}.
        \item Ulangi untuk kasus kedua dengan nilai \hex{7FFF} dan
              \hex{0001}.
    \end{enumerate}

    \item \textbf{Memeriksa dengan DEBUG.}
    \begin{enumerate}
        \item Dari DOSBox, rakit program dengan
              \file{tasm prediksi-flag.asm}, lalu link dengan
              \file{tlink prediksi-flag.obj}.
        \item Jalankan \file{debug prediksi-flag.exe}, lalu ketik
              \file{r} untuk melihat register dan \eng{flag}.
        \item Ketik \file{t} untuk menjalankan satu instruksi dan
              bandingkan hasilnya dengan emu8086.
    \end{enumerate}
\end{enumerate}

    \item Tabel~\ref{tab:prediksi-flag} yang telah terisi prediksi
