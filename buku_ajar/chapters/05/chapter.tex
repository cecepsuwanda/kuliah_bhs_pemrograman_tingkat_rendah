\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{1.3} --- Mengidentifikasi register umum (AX--DX), register
segmen (CS, DS, SS, ES), register \eng{pointer}/\eng{index} (SP, BP,
SI, DI), IP, dan \eng{flag} beserta pengaruh \eng{flag} terhadap
keputusan program.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{01} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{02}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item mengidentifikasi keempat belas register 8086 dan
          kelompoknya;
    \item menjelaskan register umum AX, BX, CX, DX beserta akses byte
          tinggi dan byte rendah;
    \item menjelaskan peran khas setiap register umum;
    \item menjelaskan register segmen CS, DS, SS, ES dan pasangan
          alamat yang dibentuknya;
    \item menjelaskan register \eng{pointer}/\eng{index} SP, BP, SI,
          DI serta register IP;
    \item mengidentifikasi sembilan \eng{flag} dan menjelaskan kapan
          masing-masing bernilai 1;
    \item memprediksi perubahan \eng{flag} akibat operasi aritmatika
          dan menjelaskan pengaruhnya terhadap keputusan program.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Register umum AX, BX, CX, DX dan akses byte tinggi/rendah.
    \item Peran khas setiap register umum.
    \item Register segmen CS, DS, SS, ES dan pasangan alamatnya.
    \item Register \eng{pointer}/\eng{index} SP, BP, SI, DI.
    \item Register \eng{Instruction Pointer} (IP).
    \item Register \eng{flag}: CF, PF, AF, ZF, SF, TF, IF, DF, OF.
    \item Pengaruh \eng{flag} terhadap keputusan program.
\end{enumerate}
\end{materibox}
