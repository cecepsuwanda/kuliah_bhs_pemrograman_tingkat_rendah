\section{Praktikum}
\label{sec:praktikum-bab14}

\begin{praktikumbox}
\vspace{1.1em}
\textbf{Judul:} Menelusuri stack, prosedur berparameter, dan macro.\\[0.3em]
\textbf{Sub-CPMK:} 3.5 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu menelusuri
perubahan \reg{SP} dan \reg{BP} pada pemanggilan prosedur, membaca
parameter melalui \eng{stack frame}, serta membuat dan memakai macro
berparameter.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Komputer dengan emu8086 terpasang.
    \item DOSBox dengan TASM, TLINK, dan DEBUG.
    \item Berkas \file{prosedur-luas.asm} dan \file{param-bp.asm} dari
          \file{code/asm/}.
    \item Lembar pengamatan stack.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Menguji prosedur berparameter stack.}
    \begin{enumerate}
        \item Rakit \file{prosedur-luas.asm} dengan TASM, lalu link
              dengan TLINK.
        \item Jalankan program. Program tidak mencetak keluaran;
              hasilnya disimpan pada variabel \code{luas}.
        \item Buka program pada emu8086 dan jalankan
              \eng{single step} sampai \asm{CALL HitungLuas}.
        \item Catat \reg{SP} sebelum dan sesudah \asm{PUSH tinggi},
              \asm{PUSH lebar}, dan \asm{CALL}.
        \item Masuk ke prosedur dan catat \reg{SP} serta \reg{BP}
              setelah \asm{PUSH BP} dan \asm{MOV BP, SP}.
        \item Catat nilai \code{[BP+4]} dan \code{[BP+6]}. Cocokkan
              dengan \code{lebar} dan \code{tinggi}.
    \end{enumerate}

    \item \textbf{Mengubah parameter.}
    \begin{enumerate}
        \item Ubah \code{lebar} menjadi 7 dan \code{tinggi} menjadi 6.
        \item Rakit ulang dan jalankan program.
        \item Periksa nilai \code{luas}; hasilnya harus 42.
    \end{enumerate}

    \item \textbf{Menguji akses parameter dengan BP.}
    \begin{enumerate}
        \item Rakit dan jalankan \file{param-bp.asm}.
        \item Catat keluarannya.
        \item Telusuri \asm{PUSH} dua parameter dan \asm{MOV BP, SP}.
        \item Jelaskan mengapa \code{[BP+0]} berisi parameter pertama
              dan \code{[BP+2]} berisi parameter kedua.
        \item Jelaskan mengapa dua \asm{POP AX} diperlukan sebelum
              program selesai.
    \end{enumerate}

    \item \textbf{Membuat macro.}
    \begin{enumerate}
        \item Tulis macro \code{Cetak\_Kar} seperti pada
              Kode~\ref{lst:macro-demo}.
        \item Panggil macro dengan argumen \code{'A'}.
        \item Rakit dan jalankan program; pastikan keluarannya
              \code{AAA}.
        \item Panggil macro dua kali dengan argumen berbeda. Bila
              muncul kesalahan label ganda, jelaskan penyebabnya dan
              perbaiki.
    \end{enumerate}

    \item \textbf{Membandingkan prosedur dan macro.}
    \begin{enumerate}
        \item Catat ukuran berkas hasil program prosedur.
        \item Catat ukuran berkas hasil program macro.
        \item Jelaskan mengapa program dengan macro dapat lebih besar.
    \end{enumerate}

    \item \textbf{Memeriksa stack dengan DEBUG.}
    \begin{enumerate}
        \item Jalankan \file{debug prosedur-luas.exe}.
        \item Catat \reg{SS} dan \reg{SP} dengan perintah \file{r}.
        \item Tampilkan isi stack dengan
              \lstasm|D SS:SP L 10|.
        \item Cocokkan isi stack dengan parameter dan alamat kembali.
    \end{enumerate}
\end{enumerate}

\subsection{Hasil yang Dikumpulkan}

\begin{enumerate}
    \item Berkas \file{prosedur-luas.asm} versi modifikasi.
    \item Tabel pengamatan \reg{SP}, \reg{BP}, dan isi stack.
    \item Tangkapan layar keluaran \file{param-bp.asm}.
    \item Berkas program macro dan tangkapan layar keluarannya.
    \item Penjelasan perbedaan prosedur dan macro.
\end{enumerate}
