\begin{subcpmkbox}

\SubCPMK{3.5} --- Menggunakan stack (PUSH/POP), prosedur (PROC/ENDP,
CALL/RET), dan macro dalam program yang modular.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{03} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{03}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item menjelaskan sifat \eng{last in, first out} pada
          \eng{stack};
    \item menggunakan \asm{PUSH} dan \asm{POP} serta menjelaskan
          perubahan \reg{SP};
    \item menjelaskan peran \reg{SS}:\reg{SP} dan \reg{SS}:\reg{BP};
    \item menulis prosedur dengan \asm{PROC} dan \asm{ENDP};
    \item menggunakan \asm{CALL} dan \asm{RET} serta menjelaskan
          alamat kembali;
    \item menyampaikan data ke prosedur melalui register,
          \eng{stack}, atau variabel;
    \item mendefinisikan dan menggunakan macro dengan parameter;
    \item membedakan prosedur dan macro.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Pengertian dan sifat \eng{stack}.
    \item Instruksi \asm{PUSH} dan \asm{POP}.
    \item Prosedur \asm{PROC}/\asm{ENDP}.
    \item \asm{CALL} dan \asm{RET}.
    \item Penyampaian data melalui register, \eng{stack}, dan
          variabel.
    \item \eng{Stack frame} dan register \reg{BP}.
    \item Macro \asm{MACRO}/\asm{ENDM} dan parameter.
    \item Perbedaan prosedur dan macro.
\end{enumerate}
\end{materibox}
