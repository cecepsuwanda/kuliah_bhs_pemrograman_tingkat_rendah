\section{Evaluasi}
\label{sec:evaluasi-bab8}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Instruksi yang \textbf{tidak} dapat dirakit karena
          memindahkan memori ke memori adalah \dots
          \begin{enumerate}
              \item \lstasm|MOV AX, BX|
              \item \lstasm|MOV AX, nilai|
              \item \lstasm|MOV nilai, AX|
              \item \lstasm|MOV [BX], [SI]|
          \end{enumerate}

    \item Setelah \lstasm|LEA SI, nilai| dijalankan, \reg{SI}
          berisi \dots
          \begin{enumerate}
              \item isi variabel \code{nilai}
              \item alamat offset variabel \code{nilai}
              \item nomor segmen variabel \code{nilai}
              \item ukuran variabel \code{nilai}
          \end{enumerate}

    \item Diketahui \code{ptr DD 12345678h}. Setelah
          \lstasm|LDS SI, ptr|, isi \reg{SI} dan \reg{DS} adalah
          \dots
          \begin{enumerate}
              \item \reg{SI} = \hex{1234}, \reg{DS} = \hex{5678}
              \item \reg{SI} = \hex{5678}, \reg{DS} = \hex{1234}
              \item \reg{SI} = \hex{12345678}, \reg{DS} = \hex{0000}
              \item \reg{SI} = \hex{0000}, \reg{DS} = \hex{1234}
          \end{enumerate}

    \item Diketahui \reg{BX} menunjuk awal tabel dan \reg{AL} = 4.
          Setelah \asm{XLATB}, \reg{AL} berisi \dots
          \begin{enumerate}
              \item indeks 4
              \item alamat tabel
              \item elemen tabel ke-4
              \item panjang tabel
          \end{enumerate}

    \item Diketahui \reg{AL} = \hex{80}. Setelah \asm{CBW}, isi
          \reg{AX} adalah \dots
          \begin{enumerate}
              \item \hex{0080} \item \hex{FF80}
              \item \hex{8000} \item \hex{FFFF}
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan aturan operand \asm{MOV}. Mengapa pemindahan
          memori ke memori tidak diperbolehkan? Tuliskan contoh
          penggantinya.

    \item Jelaskan perbedaan \asm{LEA} dan \asm{MOV} dengan
          operator \code{OFFSET}. Berikan satu situasi ketika
          \asm{LEA} lebih tepat digunakan.

    \item Diketahui far pointer \code{ptr DD 12345678h}. Uraikan
          empat byte yang tersimpan, lalu tentukan hasil
          \lstasm|LDS SI, ptr| dan \lstasm|LES DI, ptr|. Jelaskan
          pula risiko penggunaan \asm{LDS}.

    \item Diketahui \reg{AL} = \hex{80} dan \reg{AX} = \hex{8000}.
          Tentukan hasil \asm{CBW} dan \asm{CWD}. Jelaskan mengapa
          hasilnya bukan \hex{0080} dan \hex{0000}.
\end{enumerate}
