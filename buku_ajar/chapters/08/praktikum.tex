\section{Praktikum}
\label{sec:praktikum-bab8}

\begin{praktikumbox}
\vspace{1.1em}
\textbf{Judul:} Menelusuri instruksi transfer data dengan
\eng{single step}.\\[0.3em]
\textbf{Sub-CPMK:} 2.2 \quad\textbullet\quad
\textbf{Alokasi waktu:} 100 menit \quad\textbullet\quad
\textbf{Bentuk:} individu
\end{praktikumbox}

\subsection{Tujuan Praktikum}

Setelah menyelesaikan praktikum ini, mahasiswa mampu menjalankan
instruksi \asm{MOV}, \asm{LEA}, \asm{XCHG}, \asm{CMP}, \asm{XLAT},
\asm{CBW}, dan \asm{CWD}, lalu menjelaskan perubahan register dan
\eng{flag} yang dihasilkan.

\subsection{Alat dan Bahan}

\begin{itemize}
    \item Komputer dengan emu8086 terpasang.
    \item DOSBox dengan TASM, TLINK, dan DEBUG.
    \item Berkas \file{transfer-data.asm} dari \file{code/asm/}.
    \item Lembar pengamatan register dan \eng{flag}.
\end{itemize}

\subsection{Langkah Kerja}

\begin{enumerate}
    \item \textbf{Merakit program.}
    \begin{enumerate}
        \item Buka DOSBox, lalu masuk ke direktori kerja.
        \item Rakit dengan \file{tasm transfer-data.asm}.
        \item Link dengan \file{tlink transfer-data.obj}.
        \item Jalankan \file{transfer-data}. Program tidak mencetak
              keluaran; hasilnya diperiksa pada register.
    \end{enumerate}

    \item \textbf{Menelusuri dengan emu8086.}
    \begin{enumerate}
        \item Buka \file{transfer-data.asm} pada emu8086.
        \item Pilih \file{Compile and Emulate}, lalu tampilkan
              jendela \file{Registers} dan \file{Flags}.
        \item Tekan \file{Single Step} satu kali untuk setiap
              instruksi.
        \item Catat \reg{AX}, \reg{BX}, \reg{CX}, \reg{DX},
              \reg{SI}, dan \eng{flag} pada
              Tabel~\ref{tab:pengamatan-transfer}.
    \end{enumerate}

    \item \textbf{Menguji LEA dan MOV.}
    \begin{enumerate}
        \item Bandingkan hasil \lstasm|LEA SI, nilai1| dengan
              \lstasm|MOV SI, nilai1|.
        \item Jelaskan mengapa keduanya menghasilkan nilai berbeda.
        \item Kembalikan program ke bentuk semula.
    \end{enumerate}

    \item \textbf{Menguji XLAT.}
    \begin{enumerate}
        \item Sebelum \asm{XLATB}, catat \reg{AL} dan \reg{BX}.
        \item Setelah \asm{XLATB}, catat \reg{AL}.
        \item Ubah indeks dari 4 menjadi 9, rakit ulang, lalu
              pastikan \reg{AL} berisi karakter \code{'9'}.
    \end{enumerate}

    \item \textbf{Menguji CBW dan CWD.}
    \begin{enumerate}
        \item Catat \reg{AX} setelah \asm{CBW}.
        \item Catat \reg{DX} setelah \asm{CWD}.
        \item Ulangi dengan nilai \hex{7F} dan \hex{7FFF}. Jelaskan
              perbedaan hasil untuk bilangan positif dan negatif.
    \end{enumerate}

    \item \textbf{Menguji CMP.}
    \begin{enumerate}
        \item Catat \eng{flag} setelah \lstasm|CMP AX, nilai2|.
        \item Tentukan apakah \reg{AX} lebih kecil, sama, atau lebih
              besar dari \code{nilai2} secara tidak bertanda.
    \end{enumerate}

    \item \textbf{Memodifikasi data.}
    \begin{enumerate}
        \item Ubah \code{nilai1} menjadi \hex{5678} dan
              \code{nilai2} menjadi \hex{1234}.
        \item Rakit ulang dan ulangi penelusuran.
        \item Jelaskan perubahan hasil \asm{XCHG} dan \asm{CMP}.
    \end{enumerate}
\end{enumerate}
