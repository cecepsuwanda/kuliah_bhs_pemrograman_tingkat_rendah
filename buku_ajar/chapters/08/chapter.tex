\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{2.2} --- Menerapkan instruksi transfer data (MOV, LEA,
LDS/LES, XCHG, CMP, XLAT, CBW, CWD) untuk memindahkan data antar
register, memori, dan segmen.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{02} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{02}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item menjelaskan aturan operand pada instruksi \asm{MOV};
    \item memindahkan data antarregister, antara register dan memori,
          serta antara register dan segmen;
    \item menggunakan \asm{LEA} untuk memperoleh alamat offset;
    \item menggunakan \asm{LDS} dan \asm{LES} untuk memuat
          \eng{far pointer};
    \item menggunakan \asm{XCHG} dan \asm{CMP};
    \item menggunakan \asm{XLAT} untuk menerjemahkan isi tabel;
    \item menggunakan \asm{CBW} dan \asm{CWD} untuk konversi ukuran
          bilangan bertanda.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Aturan dan bentuk instruksi \asm{MOV}.
    \item Pemindahan data antarregister dan memori.
    \item \asm{LEA} dan pemuatan alamat offset.
    \item \asm{LDS} dan \asm{LES} untuk memuat \eng{far pointer}.
    \item \asm{XCHG} untuk menukar isi operand.
    \item \asm{CMP} untuk membandingkan operand.
    \item \asm{XLAT} untuk menerjemahkan isi tabel.
    \item \asm{CBW} dan \asm{CWD} untuk konversi bilangan bertanda.
\end{enumerate}
\end{materibox}
