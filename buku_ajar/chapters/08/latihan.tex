\section{Latihan}
\label{sec:latihan-bab8}

\begin{enumerate}
    \item \textbf{Konseptual.} Jelaskan mengapa mode pengalamatan
          diperlukan, padahal satu instruksi \asm{MOV} sudah cukup
          untuk memindahkan data. Kaitkan dengan keterbatasan byte
          \eng{ModR/M} pada Bab~\ref{chap:format-instruksi}.

    \item \textbf{Klasifikasi mode.} Tentukan mode pengalamatan
          operand sumber pada setiap penulisan berikut, lalu sebutkan
          segmen \eng{default} yang dipakai.
          \begin{enumerate}
              \item \lstasm|MOV AX, 1234h|
              \item \lstasm|MOV AX, CX|
              \item \lstasm|MOV AX, nilai| dengan \code{nilai}
                    adalah variabel \dir{DW}
              \item \lstasm|MOV AX, [BX]|
              \item \lstasm|MOV AX, [BP+2]|
              \item \lstasm|MOV AX, data1[SI]|
              \item \lstasm|MOV AX, [BX+DI+6]|
              \item \lstasm|MOV AX, ES:[BX]|
          \end{enumerate}

    \item \textbf{Perhitungan alamat.} Diketahui \reg{DS} =
          \hex{1234}, \reg{SS} = \hex{2000}, \reg{BX} = \hex{0050},
          \reg{BP} = \hex{0010}, \reg{SI} = \hex{0008}, dan
          \reg{DI} = \hex{0002}. Hitung alamat efektif dan alamat
          fisik untuk setiap instruksi berikut.
          \begin{enumerate}
              \item \lstasm|MOV AX, [SI+4]|
              \item \lstasm|MOV AX, [BP+DI]|
              \item \lstasm|MOV AX, [BX+SI+2]|
              \item \lstasm|MOV AX, [200h]|
          \end{enumerate}

    \item \textbf{Segmen default.} Jelaskan apa yang terjadi pada
          setiap pasangan instruksi berikut. Apakah keduanya membaca
          lokasi yang sama? Tuliskan bentuk \eng{override} yang
          membuat keduanya menunjuk lokasi yang sama.
          \begin{enumerate}
              \item \lstasm|MOV AX, [BX]| dan \lstasm|MOV AX, [BP]|
                    dengan \reg{BX} = \reg{BP} = \hex{0100}
              \item \lstasm|MOV AX, [SI+4]| dan
                    \lstasm|MOV AX, DS:[BP+4]|
          \end{enumerate}

    \item \textbf{Legalitas operand.} Manakah penulisan berikut yang
          ditolak assembler? Jelaskan alasan setiap penolakan dengan
          merujuk pada daftar \eng{encoding} 8086.
          \begin{enumerate}
              \item \lstasm|MOV AX, [AX]|
              \item \lstasm|MOV AX, [BX+BP]|
              \item \lstasm|MOV AX, [SI+DI]|
              \item \lstasm|MOV AX, [BX+SI+DI]|
              \item \lstasm|MOV AX, [BP+DI]|
              \item \lstasm|MOV AX, [SP+2]|
          \end{enumerate}

    \item \textbf{Efisiensi.} Urutkan empat instruksi berikut dari
          yang paling ringkas ke yang paling panjang. Sertakan
          perkiraan jumlah byte dan perkiraan jumlah siklusnya, lalu
          jelaskan mengapa urutannya demikian.
          \begin{enumerate}
              \item \lstasm|MOV AX, BX|
              \item \lstasm|MOV AX, [BX]|
              \item \lstasm|MOV AX, [BX+SI+2]|
              \item \lstasm|MOV AX, [1234h]|
          \end{enumerate}

    \item \textbf{Penulisan program.} Tulislah program lengkap yang
          menjumlahkan delapan elemen array \dir{DB} bernama
          \code{data1} dengan mode indexed, lalu menyimpan hasilnya
          pada variabel \code{total}. Jelaskan mengapa langkah
          penambahan pada register indeks berbeda dengan array
          \dir{DW}.

    \item \textbf{Analisis program.} Seorang mahasiswa menulis
          prosedur berikut untuk membaca parameter kedua. Program
          dapat dirakit, tetapi nilai yang dibaca salah.

\begin{lstlisting}[language=MASM, numbers=none]
    PUSH nilai1
    PUSH nilai2
    MOV BP, SP
    MOV AX, [BX+2]      ; membaca parameter kedua?
\end{lstlisting}

          \begin{enumerate}
              \item Jelaskan penyebab kesalahan tersebut.
              \item Tuliskan perbaikan yang tepat dengan mengubah
                    hanya satu instruksi.
              \item Sebutkan alamat efektif yang seharusnya dipakai
                    bila \reg{SS} = \hex{3000} dan \reg{SP} =
                    \hex{0100}.
          \end{enumerate}
\end{enumerate}

\begin{refleksibox}
Jawab pertanyaan berikut pada jurnal praktikum Anda.
\begin{enumerate}
    \item Mode pengalamatan mana yang paling sulit Anda pahami?
          Bagian mana yang masih membingungkan?
    \item Setelah menelusuri program dengan \eng{debugger}, apakah
          dugaan Anda tentang alamat efektif terbukti benar? Bagian
          mana yang paling mengejutkan?
    \item Konsep apa yang perlu Anda ulangi sebelum mempelajari
          direktif assembler pada bab berikutnya?
\end{enumerate}
\end{refleksibox}
