\section{Evaluasi}
\label{sec:evaluasi-bab12}

\subsection{Soal Pilihan Ganda}

\begin{enumerate}
    \item Instruksi yang menyiapkan \eng{flag} melalui pengurangan
          tanpa menyimpan hasilnya adalah \dots
          \begin{enumerate}
              \item \asm{SUB} \item \asm{CMP}
              \item \asm{TEST} \item \asm{AND}
          \end{enumerate}

    \item Setelah \lstasm|CMP AX, BX|, lompatan yang tepat untuk
          menyatakan bahwa \reg{AX} lebih kecil dari \reg{BX} pada
          perbandingan \emph{tak bertanda} adalah \dots
          \begin{enumerate}
              \item \asm{JL} \item \asm{JG}
              \item \asm{JB} \item \asm{JA}
          \end{enumerate}

    \item Instruksi \asm{LOOP} bekerja dengan cara \dots
          \begin{enumerate}
              \item mengurangi \reg{CX}, lalu melompat bila belum nol
              \item menambah \reg{CX}, lalu melompat bila belum nol
              \item menguji \reg{CX} lebih dahulu, lalu melompat
              \item menguji \flag{ZF} tanpa mengubah \reg{CX}
          \end{enumerate}

    \item Instruksi lompatan bersyarat pada 8086 selalu berbentuk
          \dots
          \begin{enumerate}
              \item \eng{short}, dengan jangkauan $\pm$ 127 byte
              \item \eng{near}, dengan jangkauan satu segmen
              \item \eng{far}, antar segmen kode
              \item bebas dipilih sesuai keinginan programmer
          \end{enumerate}

    \item Pada tabel lompatan yang dideklarasikan dengan
          \dir{DW OFFSET}, indeks nomor kasus harus dikalikan
          \dots\ sebelum dipakai untuk mengindeks tabel.
          \begin{enumerate}
              \item satu \item dua
              \item tiga \item empat
          \end{enumerate}
\end{enumerate}

\subsection{Soal Uraian}

\begin{enumerate}
    \item Jelaskan mengapa lompatan bersyarat pada 8086 tidak dapat
          membandingkan dua bilangan secara langsung, serta uraikan
          dua langkah pengambilan keputusan beserta instruksi yang
          berperan.

    \item Diberikan potongan program berikut.

\begin{lstlisting}[language=MASM, numbers=none]
MOV AL, 0FEh
MOV BL, 03h
CMP AL, BL
JB  KECIL
JA  BESAR
KECIL:
    ; blok KECIL
BESAR:
    ; blok BESAR
\end{lstlisting}

          \begin{enumerate}
              \item Tentukan keadaan \flag{ZF}, \flag{CF}, dan
                    \flag{SF} setelah \asm{CMP}.
              \item Lompatan mana yang diambil? Jelaskan.
              \item Perbaiki program tersebut agar hanya satu blok
                    yang dijalankan, dan jelaskan perbaikannya.
          \end{enumerate}

    \item Terjemahkan pseudocode berikut ke assembly 8086 beserta
          komentar, dengan \asm{CMP}, \asm{Jcc}, dan \asm{LOOP}.

\begin{pseudocode}[caption={Pseudocode untuk soal uraian nomor 3.}]
hitung = 0
i = 1
lakukan
    hitung = hitung + i
    i = i + 1
selama i <= 5
\end{pseudocode}

    \item Seorang mahasiswa menerima pesan \file{Symbol already
          defined} dan \file{Relative jump out of range}. Jelaskan
          penyebab masing-masing dan cara memperbaikinya; untuk pesan
          kedua, tuliskan potongan kode perbaikannya.
\end{enumerate}

\begin{checklistbox}
\textbf{Checklist Pencapaian Kompetensi Bab~\ref{chap:kontrol}}

Seorang mahasiswa dinyatakan kompeten pada Sub-CPMK 3.3 apabila:
\begin{itemize}
    \item[\checked] dapat menjelaskan hubungan \eng{flag} dengan
          instruksi lompatan bersyarat;
    \item[\checked] dapat memilih \asm{Jcc} yang benar untuk
          perbandingan bertanda maupun tak bertanda;
    \item[\checked] dapat menjelaskan pengaruh jarak lompatan
          terhadap ukuran kode mesin;
    \item[\checked] dapat menyusun perulangan dengan \asm{LOOP}
          maupun dengan \asm{DEC} dan \asm{JNZ};
    \item[\checked] dapat menerjemahkan \code{if}, \code{if--else},
          dan \code{while} dari pseudocode ke assembly;
    \item[\checked] berhasil menjalankan dan menelusuri program
          pencarian maksimum serta pengurutan sederhana.
\end{itemize}
\end{checklistbox}
