\begin{subcpmkbox}
\vspace{1.1em}
\SubCPMK{3.3} --- Menerapkan tujuh mode pengalamatan 8086
(\textit{register}, \textit{immediate}, \textit{direct memory},
\textit{indirect memory}, \textit{base relative}, \textit{direct
index}, \textit{based indexed}) pada instruksi transfer dan
aritmatika.

\vspace{0.4em}
\textbf{CPMK yang didukung:} \CPMK{03} \quad\textbullet\quad
\textbf{CPL yang didukung:} \CPL{03}
\end{subcpmkbox}

\begin{tujuanbox}
\vspace{1.1em}
Setelah mempelajari bab ini, mahasiswa mampu:
\begin{enumerate}
    \item menjelaskan peran mode pengalamatan dalam menentukan letak
          operand;
    \item mengenali mode pengalamatan dari penulisan operand;
    \item menerapkan mode \textit{register}, \textit{immediate}, dan
          \textit{direct memory};
    \item menerapkan mode \textit{indirect memory}, \textit{base
          relative}, \textit{direct index}, dan \textit{based
          indexed};
    \item menghitung alamat efektif dan alamat fisik operand;
    \item menentukan segmen \eng{default} setiap mode;
    \item memilih mode yang tepat untuk mengakses array dan
          \eng{record}.
\end{enumerate}
\end{tujuanbox}

\begin{materibox}
\vspace{1.1em}
\begin{enumerate}
    \item Konsep operand dan mode pengalamatan.
    \item Mode \textit{register}, \textit{immediate}, dan
          \textit{direct memory}.
    \item Mode \textit{indirect memory}.
    \item Mode \textit{base relative} dan \textit{direct index}.
    \item Mode \textit{based indexed}.
    \item Alamat efektif dan kombinasi register yang legal.
    \item Segmen \eng{default} dan penerapan pada array/\eng{record}.
\end{enumerate}
\end{materibox}
